1# Builds the file viewer pages of the showcase.
2# Every Markdown document, source or text file and folder that the showcase or a document links to gets a readable
3# page in showcase/pages/<path>.html (folders: showcase/pages/<path>/index.html): Markdown is rendered with its tables,
4# code has line numbers and colours, and a link such as SearchService.java#L63 marks that line. The pages use the
5# showcase style and script (assets/), so pictures open in the lightbox and "Show in File Explorer" works locally.
6# Run again after a document changes: python showcase/build_pages.py

20ROOT = Path(__file__).resolve().parent.parent
21OUT = ROOT / "showcase" / "pages"
22SEEDS = ["showcase/index.html", "showcase/index.vi.html", "README.md", "README.vi.md"] + sorted(
23 p.relative_to(ROOT).as_posix() for p in (ROOT / "docs").glob("*.md"))
24# A document X.md may have a Vietnamese translation X.vi.md next to it (same headings in the same order, same links).
25# Both get a page with an EN/VI switch; the Vietnamese page takes the heading ids of the English one, so every
26# #anchor works in both languages, and its links to other documents lead to their Vietnamese pages.
27TEXT = {".md", ".java", ".js", ".mjs", ".py", ".sql", ".ddl", ".txt", ".css", ".tsv", ".csv", ".json", ".html",

30# not project sources (lecturer's material, archive, built essay, run-time data): links to them stay plain links
31SKIP_TREES = ("De_bai", "Luu_tru", "Bai_nop", "app/data", "showcase/pages")
32MAX_TEXT = 1_000_000

37def kind(rel):
38 p = ROOT / rel
39 if p.is_dir():
40 return "d"
41 if p.is_file() and (p.suffix.lower() in TEXT or p.name in TEXT) and p.stat().st_size <= MAX_TEXT:
42 return "f"
43 return None

46def page_of(rel, k):
47 """Path of the viewer page of a project path, relative to the project folder."""
48 return f"showcase/pages/{rel}/index.html" if k == "d" else f"showcase/pages/{rel}.html"

51def url(from_page, target):
52 """Relative URL from one project file to another (both relative to the project folder)."""
53 r = os.path.relpath(ROOT / target, (ROOT / from_page).parent).replace(os.sep, "/")
54 return quote(r, safe="/._-~")

57def resolve(base_file, href):
58 """Project path that href (written in base_file) points to, or None if it leaves the project."""
59 parts = urlsplit(href)
60 if parts.scheme or parts.netloc or not parts.path:
61 return None, ""
62 target = (ROOT / base_file).parent / unquote(parts.path)
63 try:
64 rel = Path(os.path.normpath(target)).relative_to(ROOT).as_posix()
65 except ValueError:
66 return None, ""
67 return rel, ("#" + parts.fragment if parts.fragment else "")

70def links_in(rel, text):
71 if rel.endswith(".md"):
72 text = md.render(text)
73 return re.findall(r'(?:href|src)="([^"]+)"', text)

76def slug(text, seen):
77 s = re.sub(r"[^\w\- ]", "", text.strip().lower(), flags=re.U).replace(" ", "-")
78 n = seen.get(s, 0)
79 seen[s] = n + 1
80 return s if n == 0 else f"{s}-{n}"

83# ---------------------------------------------------------------- collect everything that needs a page
84pages = {}
85queue = [s for s in SEEDS if (ROOT / s).exists()]
86seen = set()
87while queue:
88 rel = queue.pop(0)
89 if rel in seen:
90 continue
91 seen.add(rel)
92 if any(rel == t or rel.startswith(t + "/") for t in SKIP_TREES) or not (ROOT / rel).exists():
93 continue
94 k = kind(rel)
95 if rel.startswith("showcase/index"):
96 k = "seed"
97 elif k is None:
98 continue
99 else:
100 pages[rel] = k
101 if k == "d":
102 for child in sorted((ROOT / rel).iterdir()):
103 if child.name not in SKIP_DIRS and not child.name.startswith("."):
104 queue.append(child.relative_to(ROOT).as_posix())
105 elif rel.endswith(".md") or k == "seed":
106 for href in links_in(rel, (ROOT / rel).read_text(encoding="utf-8")):
107 target, _ = resolve(rel, href)
108 if target and not target.startswith("showcase/pages"):
109 queue.append(target)

128def frame(rel, k, body, extra_css=""):
129 me = page_of(rel, k)
130 up = url(me, "")
131 up = "" if up == "." else up.rstrip("/") + "/"
132 name = rel.rsplit("/", 1)[-1]
133 folder = rel.rsplit("/", 1)[0] + "/" if "/" in rel else ""
134 lang = "vi" if is_vi(rel) else "en"
135 switch = ""
136 en, vi = english_of(rel), vietnamese_of(english_of(rel))
137 if vi: # a document in two languages
138 cur = ' aria-current="page"'
139 switch = (f'<span class="lang vlang" aria-label="Language"><a href="{url(me, page_of(en, "f"))}" lang="en"'
140 f'{cur if lang == "en" else ""}>EN</a><a href="{url(me, page_of(vi, "f"))}" lang="vi"'
141 f'{cur if lang == "vi" else ""}>VI</a></span>')
142 # no button to the raw file (a browser shows Markdown and code as plain text); online the page offers GitHub
143 github = f'<a class="vbtn" data-github="{html.escape(rel)}" hidden data-i18n="github">View on GitHub</a>'
144 return f"""<!doctype html>
145<html lang="{lang}">
146<head>
147<meta charset="utf-8">
148<meta name="viewport" content="width=device-width, initial-scale=1">
149<title>{html.escape(name)} · ABC Online Shop</title>
150<link rel="stylesheet" href="{up}app/src/main/resources/static/fonts/fonts.css">
151<link rel="stylesheet" href="{up}showcase/assets/showcase.css">{extra_css}
152<script src="{up}showcase/pages/manifest.js" defer></script>
153<script src="{up}showcase/assets/showcase.js" defer></script>
154</head>
155<body class="viewer" data-file="{html.escape(rel)}">
156<header class="vbar">
157 <a class="vback" href="{up}showcase/index.html" data-i18n="back">← Showcase</a>
158 <span class="vpath">{html.escape(folder)}<b>{html.escape(name)}{"/" if k == "d" else ""}</b></span>
159 <span class="vactions">{switch}
160 <button type="button" class="vbtn local-only" data-reveal="{html.escape(rel)}" data-i18n="reveal">Show in File Explorer</button>
161 {github}
162 </span>
163</header>
164{body}
165</body>
166</html>
167"""

170def fix_links(rel, k, rendered):
171 """Point the links of a rendered document to the viewer pages (or to the original files) from its new place."""
172 me = page_of(rel, k)
173
174 def sub(m):
175 attr, href = m.group(1), html.unescape(m.group(2))
176 if href.startswith("#"):
177 return m.group(0)
178 target, frag = resolve(rel, href)
179 if target is None:
180 return m.group(0)
181 if attr == "href" and is_vi(rel) and vietnamese_of(target): # Vietnamese documents lead to Vietnamese ones
182 target = vietnamese_of(target)
183 if attr == "href" and target in pages:
184 return f'{attr}="{url(me, page_of(target, pages[target]))}{html.escape(frag)}"'
185 return f'{attr}="{url(me, target)}{html.escape(frag)}"'
186
187 return re.sub(r'(href|src)="([^"]+)"', sub, rendered)

190def heading_ids(rel):
191 tokens = md.parse((ROOT / rel).read_text(encoding="utf-8"))
192 seen = {}
193 return [slug(tokens[i + 1].content, seen) for i, tok in enumerate(tokens) if tok.type == "heading_open"]

196def markdown_page(rel):
197 tokens = md.parse((ROOT / rel).read_text(encoding="utf-8"))
198 ids = heading_ids(rel)
199 if is_vi(rel):
200 english = heading_ids(english_of(rel))
201 if len(english) == len(ids):
202 ids = english
203 else:
204 print(f"WARNING {rel}: {len(ids)} headings, the English document has {len(english)}; anchors differ")
205 n = 0
206 for tok in tokens:
207 if tok.type == "heading_open":
208 tok.attrSet("id", ids[n])
209 n += 1
210 body = md.renderer.render(tokens, md.options, {})
211 body = re.sub(r"<table>", '<div class="tablewrap"><table>', body).replace("</table>", "</table></div>")
212 body = re.sub(r'(<img [^>]*src="([^"]+)"[^>]*>)', r'<a href="\2">\1</a>', body)
213 return frame(rel, "f", f'<main class="doc">\n{fix_links(rel, "f", body)}</main>')

219def code_page(rel):
220 data = (ROOT / rel).read_bytes()
221 text = data.decode("utf-8", errors="replace")
222 try:
223 lexer = get_lexer_for_filename(rel, stripnl=False, ensurenl=False)
224 except ClassNotFound:
225 lexer = TextLexer(stripnl=False, ensurenl=False)
226 src_lines = text.split("\n")
227 if src_lines and src_lines[-1] == "":
228 src_lines.pop()
229 out = highlight(text, lexer, FORMATTER).split("\n")
230 if out and out[-1] == "":
231 out.pop()
232 if len(out) != len(src_lines): # never shift the line numbers: fall back to plain text
233 out = [html.escape(s) for s in src_lines]
234 rows = "\n".join(f'<div class="ln" id="L{i}"><a href="#L{i}">{i}</a><span>{line or " "}</span></div>'
235 for i, line in enumerate(out, 1))
236 info = (f'<span data-num="{len(out)}" data-i18n="lines">{len(out)} lines</span> · {len(data):,} bytes · '
237 f'{html.escape(lexer.name)}')
238 body = f'<div class="codewrap"><p class="codeinfo">{info}</p><div class="code">\n{rows}\n</div></div>'
239 css = f'\n<link rel="stylesheet" href="{url(page_of(rel, "f"), "showcase/assets/code.css")}">'
240 return frame(rel, "f", body, css)

243def folder_page(rel):
244 me = page_of(rel, "d")
245 items = []
246 children = sorted((ROOT / rel).iterdir(), key=lambda p: (p.is_file(), p.name.lower()))
247 for child in children:
248 crel = child.relative_to(ROOT).as_posix()
249 if child.name in SKIP_DIRS or child.name.startswith(".") or any(crel == t or crel.startswith(t + "/") for t in SKIP_TREES):
250 continue
251 if child.is_dir():
252 n = sum(1 for _ in child.rglob("*") if _.is_file())
253 href = url(me, page_of(crel, "d")) if crel in pages else url(me, crel)
254 size = f'<span data-num="{n}" data-i18n="files">{n} files</span>'
255 label = child.name + "/"
256 else:
257 href = url(me, page_of(crel, "f")) if crel in pages else url(me, crel)
258 size, label = f"{child.stat().st_size:,} bytes", child.name
259 items.append(f'<tr><td><a href="{href}">{html.escape(label)}</a></td><td class="note">{size}</td></tr>')
260 body = (f'<main class="doc"><h1>{html.escape(rel)}/</h1>'
261 f'<div class="tablewrap"><table><tr><th data-i18n="name">Name</th><th data-i18n="size">Size</th></tr>'
262 f'{"".join(items)}</table></div></main>')
263 return frame(rel, "d", body)
