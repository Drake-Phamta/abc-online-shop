1# Handler of the link type abcshop: used by the showcase pages when they are opened from this computer.
2# abcshop:start run CHAY_DEMO.bat (the demo server) in its own window
3# abcshop:reveal?p=<path> show a file or folder of this project in File Explorer
4# abcshop:open?p=<path> open a Visual Paradigm project (.vpp) of this project
5# <path> is relative to the project folder. Anything outside the project folder, or any other action, is refused.
6# CHAY_DEMO.bat registers the handler for the current Windows user on every run (abcshop.ps1 -Register), so it always
7# points to this copy of the project. To remove it: reg delete "HKCU\Software\Classes\abcshop" /f
8# Keep this file ASCII only: Windows PowerShell 5.1 reads a script without BOM in the ANSI code page.
9# -DryRun prints what a link would do instead of doing it (used to check the handler).
10param([string]$Url = "", [switch]$Register, [switch]$DryRun)
11
12$root = [IO.Path]::GetFullPath((Join-Path $PSScriptRoot ".."))
13
14if ($Register) {
15 $exe = Join-Path $env:SystemRoot "System32\WindowsPowerShell\v1.0\powershell.exe"
16 $command = '"' + $exe + '" -NoProfile -ExecutionPolicy RemoteSigned -WindowStyle Hidden -File "' + $PSCommandPath + '" "%1"'
17 $key = [Microsoft.Win32.Registry]::CurrentUser.CreateSubKey("Software\Classes\abcshop")
18 $key.SetValue("", "URL:ABC Online Shop showcase")
19 $key.SetValue("URL Protocol", "")
20 $cmd = $key.CreateSubKey("shell\open\command")
21 $cmd.SetValue("", $command)
22 $cmd.Close()
23 $key.Close()
24 exit 0
25}
26
27function Refuse($why) { if ($DryRun) { "refused: $why" }; exit 1 }
28function Run($file, $arguments) {
29 if ($DryRun) { "run: $file $arguments"; return }
30 if ($arguments) { Start-Process -FilePath $file -ArgumentList $arguments } else { Start-Process -FilePath $file -WorkingDirectory $root }
31}
32
33if ($Url -notmatch '^abcshop:(?://)?(start|reveal|open)/?(?:\?p=([^&]*))?$') { Refuse "unknown link" }
34$action = $Matches[1]
35$rel = if ($Matches[2]) { [Uri]::UnescapeDataString($Matches[2]) } else { "" }
36
37if ($action -eq "start") { Run (Join-Path $root "CHAY_DEMO.bat") $null; exit 0 }
38
39# reveal and open: the path must stay inside the project folder and exist
40$rel = $rel.Replace("/", "\").TrimStart("\")
41if ($rel -eq "" -or $rel -match ':') { Refuse "empty or absolute path" }
42$full = [IO.Path]::GetFullPath((Join-Path $root $rel)).TrimEnd("\")
43if (-not $full.StartsWith($root.TrimEnd("\") + "\", [StringComparison]::OrdinalIgnoreCase)) { Refuse "outside the project folder" }
44if (-not (Test-Path -LiteralPath $full)) { Refuse "no such file" }
45
46if ($action -eq "reveal") {
47 if (Test-Path -LiteralPath $full -PathType Container) { Run "explorer.exe" ('"' + $full + '"') }
48 else { Run "explorer.exe" ('/select,"' + $full + '"') }
49} elseif ([IO.Path]::GetExtension($full) -ieq ".vpp") {
50 Run $full $null
51} else {
52 Refuse "only a .vpp project can be opened"
53}
54exit 0
