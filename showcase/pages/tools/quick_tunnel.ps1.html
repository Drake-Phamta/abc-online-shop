1# Public address for the demo shop through a Cloudflare quick tunnel: https://<random words>.trycloudflare.com forwards
2# to http://localhost:8080 on this computer for as long as this window stays open. No Cloudflare account is needed;
3# the address changes every time. Anyone who has the address can use the shop with the demo accounts; the data is
4# demo data, and CHAY_DEMO.bat recreates it. Started by CHAY_TUNNEL.bat. Keep this file ASCII only (PowerShell 5.1).
5$root = Split-Path -Parent $PSScriptRoot
6$shop = "http://localhost:8080/"
7
8function Answers { try { Invoke-WebRequest -UseBasicParsing -TimeoutSec 3 $shop | Out-Null; return $true } catch { return $false } }
9
10# 1. cloudflared: on the PATH, or the official binary in %LOCALAPPDATA%\cloudflared (downloaded once from GitHub)
11$exe = (Get-Command cloudflared -ErrorAction SilentlyContinue).Source
12if (-not $exe) { $exe = Join-Path $env:LOCALAPPDATA "cloudflared\cloudflared.exe" }
13if (-not (Test-Path $exe)) {
14 Write-Host "Downloading cloudflared from github.com/cloudflare/cloudflared (once) ..."
15 New-Item -ItemType Directory -Force (Split-Path $exe) | Out-Null
16 [Net.ServicePointManager]::SecurityProtocol = [Net.SecurityProtocolType]::Tls12
17 Invoke-WebRequest -UseBasicParsing "https://github.com/cloudflare/cloudflared/releases/latest/download/cloudflared-windows-amd64.exe" -OutFile $exe
18}
19if ((Get-AuthenticodeSignature $exe).Status -ne "Valid") { Write-Host "cloudflared.exe is not signed by Cloudflare: stopped."; exit 1 }
20
21# 2. the shop must answer first
22if (-not (Answers)) {
23 Write-Host "The shop is not running: starting CHAY_DEMO.bat in its own window ..."
24 Start-Process -FilePath (Join-Path $root "CHAY_DEMO.bat") -WorkingDirectory $root
25 for ($i = 0; $i -lt 120 -and -not (Answers); $i++) { Start-Sleep -Seconds 1 }
26 if (-not (Answers)) { Write-Host "The shop did not start; see the window of CHAY_DEMO.bat."; exit 1 }
27}
28
29# 3. the tunnel runs in this window; its log gives the public address. If cloudflared stops (network loss), it is
30# started again; every new address is written into the "Website" field of the public GitHub repository, where the
31# published showcase (GitHub Pages) reads it, so its "Open the shop" button always opens the current address.
32$repo = "Drake-Phamta/abc-online-shop"
33$gh = (Get-Command gh -ErrorAction SilentlyContinue).Source
34$log = Join-Path $env:TEMP "abc_shop_tunnel.log"
35$first = $true
36
37# The address also goes into live.txt of the branch "live" (not the published branch "main"): the showcase reads it from
38# raw.githubusercontent.com when the GitHub API is out of its hourly limit. Commits use the GitHub noreply address.
39function Publish-Live([string]$address) {
40 $who = "178000393+Drake-Phamta@users.noreply.github.com"
41 & $gh api "repos/$repo/branches/live" --silent 2>$null
42 if ($LASTEXITCODE -ne 0) {
43 $main = & $gh api "repos/$repo/git/ref/heads/main" --jq .object.sha 2>$null
44 & $gh api -X POST "repos/$repo/git/refs" -f "ref=refs/heads/live" -f "sha=$main" --silent 2>$null
45 }
46 $sha = & $gh api "repos/$repo/contents/live.txt?ref=live" --jq .sha 2>$null
47 $body = @("-X", "PUT", "repos/$repo/contents/live.txt", "-f", "message=Live shop address", "-f", "branch=live",
48 "-f", ("content=" + [Convert]::ToBase64String([Text.Encoding]::ASCII.GetBytes($address))),
49 "-f", "committer[name]=Drake-Phamta", "-f", "committer[email]=$who", "-f", "author[name]=Drake-Phamta", "-f", "author[email]=$who", "--silent")
50 if ($LASTEXITCODE -eq 0 -and $sha) { $body += @("-f", "sha=$sha") }
51 & $gh api @body 2>$null
52 return ($LASTEXITCODE -eq 0)
53}
54while ($true) {
55 if (Test-Path $log) { Clear-Content $log }
56 $p = Start-Process -FilePath $exe -ArgumentList "tunnel", "--no-autoupdate", "--url", $shop -NoNewWindow -PassThru -RedirectStandardError $log
57 $url = $null
58 for ($i = 0; $i -lt 60 -and -not $url -and -not $p.HasExited; $i++) {
59 Start-Sleep -Seconds 1
60 $m = Select-String -Path $log -Pattern "https://[a-z0-9-]+\.trycloudflare\.com" -ErrorAction SilentlyContinue | Select-Object -First 1
61 if ($m) { $url = $m.Matches[0].Value }
62 }
63 if ($url) {
64 try { Set-Clipboard -Value $url } catch { }
65 Write-Host ""
66 Write-Host (" " + (Get-Date -Format "HH:mm") + " ABC Online Shop is public at: $url") -ForegroundColor Green
67 Write-Host " (copied to the clipboard; the address works while this window stays open)"
68 Write-Host " Accounts: an.nguyen@example.com, sales@example.com, warehouse@example.com; password demo1234"
69 if ($gh) {
70 & $gh repo edit $repo --homepage $url 2>$null | Out-Null
71 if ($LASTEXITCODE -eq 0) { Write-Host " The published showcase now opens this address (GitHub repository website field)." }
72 else { Write-Host " Could not update the GitHub repository (gh auth status)." }
73 if (-not (Publish-Live $url)) { Write-Host " Could not write live.txt (branch live); the website field still works." }
74 }
75 Write-Host ""
76 if ($first) { Start-Process $url; $first = $false }
77 } else {
78 Write-Host "No public address after 60 s; the log is $log"
79 }
80 # while the tunnel runs, watch the shop too: if it has not answered for 3 minutes (its window was closed and not
81 # reopened, or the server stopped), start it again with the data kept. 3 minutes leave time for a manual restart.
82 $down = 0
83 while (-not $p.HasExited) {
84 Start-Sleep -Seconds 30
85 if (Answers) { $down = 0; continue }
86 $down++
87 if ($down -eq 6) {
88 Write-Host ((Get-Date -Format "HH:mm") + " the shop has not answered for 3 minutes: starting CHAY_DEMO.bat keep")
89 Start-Process -FilePath (Join-Path $root "CHAY_DEMO.bat") -ArgumentList "keep" -WorkingDirectory $root
90 }
91 if ($down -ge 12) { $down = 0 }
92 }
93 Write-Host ((Get-Date -Format "HH:mm") + " cloudflared stopped; starting it again in 10 s (close this window to stop for good)")
94 Start-Sleep -Seconds 10
95}
