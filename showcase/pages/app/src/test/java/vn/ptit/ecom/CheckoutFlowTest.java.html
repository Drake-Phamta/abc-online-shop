122 /**
123 * UC07 is one transaction (OrderService.placeOrder): the stock is reserved before the order row is written. When
124 * that write fails, the reservation must be rolled back too: same stock, same cart, no new order.
125 */
126 @Test
127 void aFailedOrderWriteRollsBackTheStockReservation() throws Exception { // BR05, commit / rollback
128 MockHttpSession minh = login("minh.le@example.com");
129 int pid = productWithStock(3);
130 mvc.perform(post("/api/cart/items").session(minh).contentType(MediaType.APPLICATION_JSON)
131 .content("{\"productId\":" + pid + ",\"quantity\":2}")).andExpect(status().isOk());
132 int stockBefore = stockOf(pid), ordersBefore = read(minh, "/api/orders").size();
133 JsonNode co = read(minh, "/api/checkout");
134 exec("CREATE TRIGGER test_fail_order BEFORE INSERT ON customer_order BEGIN SELECT RAISE(ABORT, 'simulated failure'); END");
135 try {
136 mvc.perform(post("/api/orders").session(minh).contentType(MediaType.APPLICATION_JSON)
137 .content("{\"addressId\":" + co.get("addresses").get(0).get("id").asInt() + ",\"paymentMethod\":\"COD\"}"))
138 .andExpect(status().isInternalServerError());
139 } finally {
140 exec("DROP TRIGGER test_fail_order");
141 }
142 assertEquals(stockBefore, stockOf(pid));
143 assertEquals(2, read(minh, "/api/cart").get("count").asInt());
144 assertEquals(ordersBefore, read(minh, "/api/orders").size());
145 }
