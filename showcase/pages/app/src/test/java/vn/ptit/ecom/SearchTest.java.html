28/** The AI search through the controller on the demo catalogue (docs/SPEC.md AT11 to AT17). */
29@SpringBootTest(properties = {"abc.db.file=target/test-db/abc_search_test.db", "abc.db.reset=true"})
30@AutoConfigureMockMvc
31@DirtiesContext
32class SearchTest {
33 @Autowired MockMvc mvc;
34 private final ObjectMapper json = new ObjectMapper();
35
36 private JsonNode search(String q) throws Exception {
37 return json.readTree(mvc.perform(get("/api/search").param("q", q)).andExpect(status().isOk())
38 .andReturn().getResponse().getContentAsString(java.nio.charset.StandardCharsets.UTF_8)); // JSON is UTF-8
39 }
40
41 private static List<String> skus(JsonNode r) {
42 List<String> out = new ArrayList<>();
43 for (JsonNode f : r.get("results")) out.add(f.get("product").get("sku").asText());
44 return out;
45 }

47 @Test
48 void synonymsAndColourFindTheRedSneakers() throws Exception { // AT11, AI02
49 JsonNode r = search("red sneakers");
50 assertEquals("FA-SHO-003", skus(r).get(0));
51 assertTrue(r.get("understood").get("related").toString().contains("shoe"));
52 assertFalse(r.get("results").get(0).get("reasons").isEmpty());
53 }

104 @Test
105 void photoOfAProductFindsIt() throws Exception { // AT15, FR-S5
106 BufferedImage shoe = ImageIO.read(getClass().getResourceAsStream("/static/img/products/fa-sho-002.jpg"));
107 int w = shoe.getWidth(), h = shoe.getHeight();
108 BufferedImage photo = new BufferedImage((int) (w * 0.8), (int) (h * 0.8), BufferedImage.TYPE_INT_RGB);
109 Graphics2D g = photo.createGraphics();
110 g.drawImage(shoe, -(int) (w * 0.1), -(int) (h * 0.1), null); // the middle 80 %
111 g.dispose();
112 photo = new RescaleOp(0.75f, 0, null).filter(photo, null); // a darker room
113 ByteArrayOutputStream jpeg = new ByteArrayOutputStream();
114 ImageIO.write(photo, "jpg", jpeg);
115 String body = mvc.perform(multipart("/api/search/photo").file(new MockMultipartFile("photo", "shoe.jpg", "image/jpeg", jpeg.toByteArray())))
116 .andExpect(status().isOk()).andExpect(jsonPath("$.mode").value("PHOTO")).andReturn().getResponse().getContentAsString();
117 assertEquals("FA-SHO-002", skus(json.readTree(body)).get(0));
118 }
