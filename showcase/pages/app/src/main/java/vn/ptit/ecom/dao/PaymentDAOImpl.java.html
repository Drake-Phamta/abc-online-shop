13 PaymentDAOImpl(Connection connection) {

17 @Override
18 public void saveAll(int orderId, List<Payment> payments) throws SQLException {
19 for (Payment p : payments) {
20 if (p.getId() != null) {
21 Jdbc.update(connection, "UPDATE payment SET status = ? WHERE payment_id = ?", p.getStatus().name(), p.getId());
22 continue;
23 }
24 int id = Jdbc.insert(connection, "INSERT INTO payment (order_id, method, amount, status, payment_date) VALUES (?, ?, ?, ?, ?)",
25 orderId, p.method(), p.getAmount(), p.getStatus().name(), Jdbc.ts(p.getPaymentDate()));
26 p.assignId(id);
27 if (p instanceof PayCredit c)
28 Jdbc.update(connection, "INSERT INTO pay_credit (payment_id, card_holder, card_last4, expiry, provider_token) "
29 + "VALUES (?, ?, ?, ?, ?)", id, c.getCardHolder(), c.getCardLast4(), c.getExpiry(), c.getProviderToken());
30 else if (p instanceof PayCash c)
31 Jdbc.update(connection, "INSERT INTO pay_cash (payment_id, cash_received, change_amount) VALUES (?, ?, ?)",
32 id, c.getCashReceived(), c.getChangeAmount());
33 }
34 }
