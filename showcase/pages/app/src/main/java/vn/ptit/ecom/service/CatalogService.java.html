10/** UC04 Browse / Search Products, UC05 View Product, UC15 Manage Inventory. */
11@Service
12public class CatalogService {
13 /** A product page: the product with its detail, and the parent of its category for the breadcrumb. */
14 public record ProductDetails(Product product, Category parentCategory) { }
15
16 private final Database db;
17
18 public CatalogService(Database db) {
19 this.db = db;
20 }
21
22 public List<Category> categories() {
23 return db.inTransaction(d -> d.products().categories());
24 }
25
26 /** UC04: ACTIVE products only; a parent category includes its sub-categories. */
27 public List<Product> search(String keyword, String categorySlug, ProductDAO.Sort sort) {
28 return db.inTransaction(d -> d.products().search(keyword, categorySlug, sort));
29 }
30
31 /** UC05: a hidden product is reported as missing (BR01). */
32 public ProductDetails product(int id) {
33 return db.inTransaction(d -> {
34 Product p = d.products().findById(id).filter(x -> x.getStatus() == Product.Status.ACTIVE)
35 .orElseThrow(() -> new NotFound("This product does not exist."));
36 Integer parentId = p.getCategory().parentId();
37 Category parent = parentId == null ? null
38 : d.products().categories().stream().filter(c -> c.id() == parentId).findFirst().orElse(null);
39 return new ProductDetails(p, parent);
40 });
41 }
42
43 /** UC15 step 1: every product, lowest stock first. */
44 public List<Product> stockList() {
45 return db.inTransaction(d -> d.products().listForStock());
46 }
47
48 /** UC15 steps 2 and 3: the new quantity is checked by the model (BR12) before it is stored. */
49 public Product updateStock(int productId, int quantity) {
50 return db.inTransaction(d -> {
51 Product p = d.products().findById(productId).orElseThrow(() -> new NotFound("This product does not exist."));
52 p.updateStock(quantity);
53 d.products().updateStock(productId, p.getStockQuantity());
54 return p;
55 });
56 }
