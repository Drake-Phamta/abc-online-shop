5/** Card data typed by the customer; checked before the provider is called (UC08 A2). Never stored. */
6public record CardData(String holder, String number, String expiry, String cvv) {

8 public CardData {

13 if (holder.length() < 2) throw new IllegalArgumentException("Enter the name printed on the card.");
14 if (!number.matches("\\d{16}") || !luhn(number)) throw new IllegalArgumentException("The card number is not valid.");
15 if (!expiry.matches("(0[1-9]|1[0-2])/\\d{2}")) throw new IllegalArgumentException("Expiry must look like MM/YY.");
16 YearMonth exp = YearMonth.of(2000 + Integer.parseInt(expiry.substring(3)), Integer.parseInt(expiry.substring(0, 2)));
17 if (exp.isBefore(YearMonth.now())) throw new IllegalArgumentException("The card has expired.");
18 if (!cvv.matches("\\d{3}")) throw new IllegalArgumentException("CVV must be 3 digits.");
19 }

23 private static boolean luhn(String digits) {
24 int sum = 0;
25 for (int i = 0; i < digits.length(); i++) {
26 int d = digits.charAt(digits.length() - 1 - i) - '0';
27 if (i % 2 == 1) { d *= 2; if (d > 9) d -= 9; }
28 sum += d;
29 }
30 return sum % 10 == 0;
31 }
