25 @Override
26 public Optional<Account> findAccount(String email) throws SQLException {
27 try (PreparedStatement ps = Jdbc.prepare(connection, "SELECT * FROM customer WHERE lower(email) = lower(?)", email);
28 ResultSet rs = ps.executeQuery()) {
29 if (!rs.next()) return Optional.empty();
30 return Optional.of(new Account(Role.CUSTOMER, rs.getInt("customer_id"), name(rs).display(),
31 rs.getString("password_hash")));
32 }

35 @Override
36 public Optional<Customer> findById(int customerId) throws SQLException {
37 String sql = "SELECT c.*, v.vip_level, v.discount_rate, v.reward_points, n.registered_date, n.welcome_voucher "
38 + "FROM customer c LEFT JOIN customer_vip v ON v.customer_id = c.customer_id "
39 + "LEFT JOIN customer_new n ON n.customer_id = c.customer_id WHERE c.customer_id = ?";
40 try (PreparedStatement ps = Jdbc.prepare(connection, sql, customerId); ResultSet rs = ps.executeQuery()) {
41 if (!rs.next()) return Optional.empty();
42 int id = rs.getInt("customer_id");
43 FullName name = name(rs);
44 String email = rs.getString("email"), phone = rs.getString("phone");
45 if (Jdbc.intOrNull(rs, "vip_level") != null)
46 return Optional.of(new CustomerVIP(id, name, email, phone, rs.getInt("vip_level"),
47 rs.getBigDecimal("discount_rate"), rs.getInt("reward_points")));
48 if (rs.getString("registered_date") != null)
49 return Optional.of(new CustomerNew(id, name, email, phone,
50 java.time.LocalDate.parse(rs.getString("registered_date")), rs.getString("welcome_voucher")));
51 return Optional.of(new Customer(id, name, email, phone));
52 }
