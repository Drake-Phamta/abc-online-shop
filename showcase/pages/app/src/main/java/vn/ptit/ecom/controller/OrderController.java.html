19/** UC07 Place Order (with UC21 Validate Cart), UC09 Track Order, UC14 Process Orders (status part). */

30 /** UC07 steps 1 to 4: the validated cart (UC21), the customer's addresses and the totals of BR04. */
31 @GetMapping("/api/checkout")
32 public Map<String, Object> checkout(HttpSession session) {
33 CurrentUser u = CurrentUser.require(session, Role.CUSTOMER);
34 OrderService.Checkout c = orders.checkout(u.id());
35 Map<String, Object> out = new LinkedHashMap<>();
36 out.put("cart", Views.CartView.of(c.cart()));
37 out.put("issues", c.issues().stream().map(Views.Issue::of).toList());
38 out.put("addresses", c.addresses().stream().map(Views.AddressView::of).toList());
39 out.put("customerKind", c.customer().kindLabel());
40 out.put("discountRate", c.customer().discountRate());
41 out.put("totals", c.totals());
42 return out;
43 }
44
45 /** UC07 steps 5 to 7. The request is checked here; the rules and the transaction are OrderService's. */
46 @PostMapping("/api/orders")
47 public Map<String, Object> placeOrder(@RequestBody PlaceOrder req, HttpSession session) {
48 CurrentUser u = CurrentUser.require(session, Role.CUSTOMER);
49 if (req.addressId() == null) throw new IllegalArgumentException("Choose a delivery address.");
50 Order order = orders.placeOrder(u.id(), req.addressId(), parseMethod(req.paymentMethod()), req.expectedTotal());
51 return Map.of("orderId", order.getId(), "code", order.code(), "status", order.getStatus(), "total", order.getTotalAmount());
52 }

54 /** UC09 step 2. */
55 @GetMapping("/api/orders")
56 public List<OrderDAO.OrderSummary> myOrders(HttpSession session) {
57 CurrentUser u = CurrentUser.require(session, Role.CUSTOMER);

61 /** UC09 step 4 for the customer (own orders only, BR08) and UC14 for sales staff. */

94 /** Customers see only their own orders (BR08); staff see all. */
95 static Integer ownerFilter(CurrentUser u) {
96 return u.role() == Role.CUSTOMER ? u.id() : null;
97 }
98
99 /** The order page, with the actions the caller may take next (BR09). */
100 static Views.OrderView view(OrderService.OrderDetails d, CurrentUser u) {
101 return Views.OrderView.of(d.order(), d.address(), d.customerName(),
102 d.order().getStatus().next(u.role() == Role.SALES ? Actor.SALES : Actor.CUSTOMER));
103 }

105 private static PaymentMethod parseMethod(String s) {
106 try {
107 return PaymentMethod.valueOf(s == null ? "" : s.trim().toUpperCase());
108 } catch (IllegalArgumentException e) {
109 throw new IllegalArgumentException("Choose card or cash on delivery.");
110 }
