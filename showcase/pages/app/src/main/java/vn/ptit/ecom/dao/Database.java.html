13/**
14 * Opens the SQLite database and runs one unit of work in one transaction. Checkout touches carts, products,
15 * orders and payments; running them on one connection makes it all-or-nothing (NFR07, NFR10).
16 */
17public class Database {
18
19 /** All DAOs of one unit of work, sharing its connection. */
20 public record Daos(ProductDAO products, CustomerDAO customers, StaffDAO staff, CartDAO carts, OrderDAO orders,
21 PaymentDAO payments, ShippingDAO shipping) { }
22
23 @FunctionalInterface
24 public interface Work<T> {
25 T run(Daos daos) throws SQLException;
26 }

28 private final Path file;
29
30 public Database(Path file) {
31 this.file = file;

34 public <T> T inTransaction(Work<T> work) {
35 try (Connection c = open()) {
36 c.setAutoCommit(false);
37 try {
38 T result = work.run(new Daos(new ProductDAOImpl(c), new CustomerDAOImpl(c), new StaffDAOImpl(c),
39 new CartDAOImpl(c), new OrderDAOImpl(c), new PaymentDAOImpl(c), new ShippingDAOImpl(c)));
40 c.commit();
41 return result;
42 } catch (RuntimeException | SQLException e) {
43 c.rollback();
44 throw e;
45 }
46 } catch (SQLException e) {
47 throw new DataAccessError(e);
48 }
49 }
50
51 /** Creates the database from the DDL generated from the physical model (db/schema.sql) and the seed data. */
52 public void create(boolean replaceExisting) {
53 try {
54 if (Files.exists(file) && !replaceExisting) return;
55 if (file.getParent() != null) Files.createDirectories(file.getParent());
56 Files.deleteIfExists(file);
57 try (Connection c = open(); Statement s = c.createStatement()) {
58 for (String script : new String[]{"/db/schema.sql", "/db/seed.sql"})
59 for (String sql : statements(resource(script))) s.execute(sql);
60 }
61 } catch (IOException | SQLException e) {
62 throw new DataAccessError(e);
63 }
64 }
65
66 private Connection open() throws SQLException {
67 SQLiteConfig config = new SQLiteConfig();
68 config.enforceForeignKeys(true);
69 config.setBusyTimeout(5000);
70 config.setTransactionMode(SQLiteConfig.TransactionMode.IMMEDIATE);
71 return config.createConnection("jdbc:sqlite:" + file.toAbsolutePath());
72 }
73
74 private static String resource(String name) throws IOException {
75 try (InputStream in = Database.class.getResourceAsStream(name)) {
76 if (in == null) throw new IOException("missing resource " + name);
77 return new String(in.readAllBytes(), StandardCharsets.UTF_8);
78 }
79 }

81 /** Splits a script on semicolons that end a line; comment lines are dropped. */
82 static String[] statements(String script) {
83 StringBuilder clean = new StringBuilder();
84 for (String line : script.split("\\R")) if (!line.trim().startsWith("--")) clean.append(line).append('\n');
85 return java.util.Arrays.stream(clean.toString().split(";\\s*\\n")).map(String::trim).filter(s -> !s.isEmpty())
86 .toArray(String[]::new);
87 }
