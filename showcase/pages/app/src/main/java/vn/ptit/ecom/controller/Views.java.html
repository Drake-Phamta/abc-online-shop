26 record ProductCard(int id, String sku, String name, String brand, long price, int stock, String stockState,
27 String image, String category, String categorySlug, String detailKind, List<String> chips,
28 String byline) {
29 static ProductCard of(Product p) {
30 List<String> chips = p.getDetail() == null ? List.of()
31 : p.getDetail().specs().stream()
32 .filter(s -> !s.label().equals("Warranty") && !s.label().equals("Author") && !s.label().equals("Publisher"))
33 .limit(3).map(ProductDetail.Spec::value).toList();
34 return new ProductCard(p.getId(), p.getSku(), p.getName(), p.getBrand(), p.getPrice(), p.getStockQuantity(),
35 Views.stockState(p), Views.image(p), p.getCategory().name(), p.getCategory().slug(),
36 p.getCategory().detailKind() == null ? null : p.getCategory().detailKind().name(), chips,
37 p.getBrand() != null ? p.getBrand()
38 : p.getDetail() instanceof ProductDetail.BookDetail b ? b.author() : null);
39 }
40 }

48 record CartView(int cartId, List<CartLine> lines, int count, long subtotal) {
49 static CartView of(Cart cart) {
50 List<CartLine> lines = cart.getItems().stream().map(i -> new CartLine(i.getProduct().getId(), i.getProduct().getName(),
51 i.getProduct().getBrand(), image(i.getProduct()), i.getCartPrice(), i.getProduct().getPrice(),
52 i.getQuantity(), i.getProduct().getStockQuantity(), i.subTotal())).toList();
53 return new CartView(cart.getId(), lines, lines.stream().mapToInt(CartLine::quantity).sum(), cart.calcTotal());
54 }
