15/** Maps failures to HTTP answers the views understand: {status, rule?, message, details?}. */
16@RestControllerAdvice
17class ApiErrors {
18 private static final Logger log = LoggerFactory.getLogger(ApiErrors.class);
19
20 static class Unauthorized extends RuntimeException {
21 Unauthorized() { this("Please log in first."); }
22 Unauthorized(String message) { super(message); }
23 }
24
25 static class Forbidden extends RuntimeException {
26 Forbidden() { super("Your account cannot use this function."); }
27 }
28
29 @ExceptionHandler(RuleViolation.class)
30 ResponseEntity<Map<String, Object>> rule(RuleViolation e) {
31 return body(HttpStatus.CONFLICT, Map.of("rule", e.getRule(), "message", e.getMessage(), "details", e.getDetails()));
32 }
33
34 @ExceptionHandler(IllegalArgumentException.class)
35 ResponseEntity<Map<String, Object>> badInput(IllegalArgumentException e) {
36 return body(HttpStatus.BAD_REQUEST, Map.of("message", e.getMessage()));
37 }
38
39 @ExceptionHandler(org.springframework.web.multipart.MaxUploadSizeExceededException.class) // AI06
40 ResponseEntity<Map<String, Object>> tooLarge(Exception e) {
41 return body(HttpStatus.BAD_REQUEST, Map.of("message", "The photo is larger than 5 MB."));
42 }
43
44 @ExceptionHandler(NotFound.class) // vn.ptit.ecom.service.NotFound: decided by the services (missing record, BR08)
45 ResponseEntity<Map<String, Object>> notFound(NotFound e) {
46 return body(HttpStatus.NOT_FOUND, Map.of("message", e.getMessage()));
47 }
48
49 @ExceptionHandler(Unauthorized.class)
50 ResponseEntity<Map<String, Object>> unauthorized(Unauthorized e) {
51 return body(HttpStatus.UNAUTHORIZED, Map.of("message", e.getMessage()));
52 }
53
54 @ExceptionHandler(Forbidden.class)
55 ResponseEntity<Map<String, Object>> forbidden(Forbidden e) {
56 return body(HttpStatus.FORBIDDEN, Map.of("message", e.getMessage()));
57 }
58
59 @ExceptionHandler(DataAccessError.class)
60 ResponseEntity<Map<String, Object>> database(DataAccessError e) {
61 log.error("database error", e);
62 return body(HttpStatus.INTERNAL_SERVER_ERROR, Map.of("message", "The shop could not save your request. Please try again."));
63 }
64
65 private static ResponseEntity<Map<String, Object>> body(HttpStatus status, Map<String, Object> fields) {
66 Map<String, Object> out = new java.util.LinkedHashMap<>(fields);
67 out.put("status", status.value());
68 if (out.get("details") instanceof List<?> l && l.isEmpty()) out.remove("details");
69 return ResponseEntity.status(status).body(out);
70 }
