2
3import jakarta.servlet.http.HttpSession;
4import org.springframework.web.bind.annotation.DeleteMapping;
5import org.springframework.web.bind.annotation.GetMapping;
6import org.springframework.web.bind.annotation.PathVariable;
7import org.springframework.web.bind.annotation.PostMapping;
8import org.springframework.web.bind.annotation.PutMapping;
9import org.springframework.web.bind.annotation.RequestBody;
10import org.springframework.web.bind.annotation.RequestMapping;
11import org.springframework.web.bind.annotation.RestController;
12import vn.ptit.ecom.model.customer.Role;
13import vn.ptit.ecom.service.CartService;
14
15/** UC06 Manage Shopping Cart. The customer id always comes from the session, never from the request (BR08). */
16@RestController
17@RequestMapping("/api/cart")
18public class CartController {
19 record AddItem(Integer productId, Integer quantity) { }
20
21 record ChangeQuantity(Integer quantity) { }
22
23 private final CartService carts;
24
25 public CartController(CartService carts) {
26 this.carts = carts;
27 }
28
29 @GetMapping
30 public Views.CartView viewCart(HttpSession session) {
31 CurrentUser u = CurrentUser.require(session, Role.CUSTOMER);
32 return Views.CartView.of(carts.cart(u.id()));
33 }
34
35 @PostMapping("/items")
36 public Views.CartView addItem(@RequestBody AddItem req, HttpSession session) {
37 CurrentUser u = CurrentUser.require(session, Role.CUSTOMER);
38 if (req.productId() == null || req.quantity() == null) throw new IllegalArgumentException("Choose a product and a quantity.");
39 return Views.CartView.of(carts.addItem(u.id(), req.productId(), req.quantity()));
40 }
41
42 @PutMapping("/items/{productId}")
43 public Views.CartView changeQuantity(@PathVariable int productId, @RequestBody ChangeQuantity req, HttpSession session) {
44 CurrentUser u = CurrentUser.require(session, Role.CUSTOMER);
45 if (req.quantity() == null) throw new IllegalArgumentException("Enter a quantity.");
46 return Views.CartView.of(carts.changeQuantity(u.id(), productId, req.quantity()));
47 }
48
49 @DeleteMapping("/items/{productId}")
50 public Views.CartView removeItem(@PathVariable int productId, HttpSession session) {
51 CurrentUser u = CurrentUser.require(session, Role.CUSTOMER);
52 return Views.CartView.of(carts.removeItem(u.id(), productId));
53 }
54}
