69 private static final Map<String, Set<String>> RELATED = new HashMap<>();
70 static {
71 for (String[] group : SYNONYMS)
72 for (String w : group)
73 for (String r : group)
74 if (!w.equals(r)) RELATED.computeIfAbsent(stem(w), k -> new TreeSet<>()).add(stem(r));
75 for (String[] pair : BROADER) { // both ways: "shoes" also finds boots, "boots" also finds other shoes
76 RELATED.computeIfAbsent(stem(pair[0]), k -> new TreeSet<>()).add(stem(pair[1]));
77 RELATED.computeIfAbsent(stem(pair[1]), k -> new TreeSet<>()).add(stem(pair[0]));
78 }

88 public SearchQuery interpret(String text) {
89 String original = text == null ? "" : text.trim();
90 if (original.length() > MAX_LENGTH) throw new IllegalArgumentException("A search can have at most " + MAX_LENGTH + " characters.");
91 String s = " " + normalize(original) + " ";
92 for (String[] vi : VIETNAMESE) s = s.replace(" " + vi[0] + " ", " " + vi[1] + " ");
93 Long min = null, max = null;
94 Matcher m = BETWEEN.matcher(s);
95 if (m.find()) {
96 min = amount(m.group(1), m.group(2));
97 max = amount(m.group(3), m.group(4) != null ? m.group(4) : m.group(2));
98 s = s.substring(0, m.start()) + " " + s.substring(m.end());
99 }
100 m = MAX.matcher(s);
101 if (m.find()) {
102 max = amount(m.group(1), m.group(2));
103 s = s.substring(0, m.start()) + " " + s.substring(m.end());
104 }
105 m = MIN.matcher(s);
106 if (m.find()) {
107 min = amount(m.group(1), m.group(2));
108 s = s.substring(0, m.start()) + " " + s.substring(m.end());
109 }
110 boolean cheap = false;
111 List<String> terms = new ArrayList<>();
112 Map<String, String> corrections = new LinkedHashMap<>();
113 for (String raw : s.split("[^a-z0-9]+")) {
114 if (raw.isEmpty() || STOP.contains(raw)) continue;
115 if (CHEAP.contains(raw)) { cheap = true; continue; }
116 String t = stem(raw);
117 if (!known(t) && !t.chars().allMatch(Character::isDigit)) {
118 String c = correct(t);
119 if (c != null) {
120 corrections.put(raw, c);
121 t = c;
122 }
123 }
124 if (!terms.contains(t)) terms.add(t);
125 }
126 Map<String, List<String>> related = new LinkedHashMap<>();
127 for (String t : terms) {
128 List<String> rel = RELATED.getOrDefault(t, Set.of()).stream()
129 .filter(r -> vocabulary.containsKey(r) && !terms.contains(r)).toList();
130 if (!rel.isEmpty()) related.put(t, rel);
131 }
132 return new SearchQuery(original, List.copyOf(terms), related, corrections, min, max, cheap);
133 }

135 /**
136 * Lower case, no accents (Vietnamese queries work with or without diacritics), no apostrophes ("Levi's" = "levis"),
137 * letters and digits apart ("16gb" = "16 gb", as the catalogue writes "16 GB").
138 */
139 static String normalize(String text) {
140 String s = Normalizer.normalize(text.toLowerCase().replace('đ', 'd'), Normalizer.Form.NFD).replaceAll("\\p{M}+", "");
141 s = s.replace("'", "").replace("’", "");
142 return s.replaceAll("(?<=[a-z])(?=\\d)|(?<=\\d)(?=[a-z])", " ");
143 }

145 /** The words of a text as the index stores them; the index and the query use the same function. */
146 public static List<String> tokens(String text) {
147 List<String> out = new ArrayList<>();
148 if (text == null) return out;
149 for (String raw : normalize(text).split("[^a-z0-9]+"))
150 if (!raw.isEmpty() && !STOP.contains(raw)) out.add(stem(raw));
151 return out;
152 }

154 /** Plural to singular, enough for product words: sneakers, laptops, stories. Numbers stay as they are. */
155 static String stem(String w) {
156 if (w.length() > 4 && w.endsWith("ies")) return w.substring(0, w.length() - 3) + "y";
157 if (w.length() > 3 && w.endsWith("s") && !w.endsWith("ss") && !w.endsWith("us") && !w.endsWith("is")
158 && !Character.isDigit(w.charAt(0))) return w.substring(0, w.length() - 1);
159 return w;
160 }

162 private boolean known(String t) {
163 return vocabulary.containsKey(t) || RELATED.containsKey(t);
164 }
165
166 /**
167 * The catalogue or lexicon word closest to a word nobody uses: one edit for words of 5 to 7 letters, two from 8.
168 * Shorter words are left alone: "dien" is one edit away from too many words to guess.
169 */
170 private String correct(String t) {
171 if (t.length() < 5) return null;
172 int limit = t.length() <= 7 ? 1 : 2;
173 Comparator<String> better = Comparator.<String>comparingInt(c -> distance(t, c))
174 .thenComparing(c -> -vocabulary.getOrDefault(c, 0)).thenComparing(c -> c);
175 String best = null;
176 for (String c : candidates()) {
177 if (Math.abs(c.length() - t.length()) > limit || c.chars().allMatch(Character::isDigit)) continue;
178 if (distance(t, c) <= limit && (best == null || better.compare(c, best) < 0)) best = c;
179 }
180 return best;
181 }
182
183 private Set<String> candidates() {
184 Set<String> all = new TreeSet<>(vocabulary.keySet());
185 all.addAll(RELATED.keySet());
186 return all;
187 }

189 /** Damerau-Levenshtein distance (a swap of two neighbouring letters counts as one edit). */
190 static int distance(String a, String b) {
191 int[][] d = new int[a.length() + 1][b.length() + 1];
192 for (int i = 0; i <= a.length(); i++) d[i][0] = i;
193 for (int j = 0; j <= b.length(); j++) d[0][j] = j;
194 for (int i = 1; i <= a.length(); i++)
195 for (int j = 1; j <= b.length(); j++) {
196 int cost = a.charAt(i - 1) == b.charAt(j - 1) ? 0 : 1;
197 d[i][j] = Math.min(Math.min(d[i - 1][j] + 1, d[i][j - 1] + 1), d[i - 1][j - 1] + cost);
198 if (i > 1 && j > 1 && a.charAt(i - 1) == b.charAt(j - 2) && a.charAt(i - 2) == b.charAt(j - 1))
199 d[i][j] = Math.min(d[i][j], d[i - 2][j - 2] + 1);
200 }
201 return d[a.length()][b.length()];
202 }

204 /**
205 * "10 million", "10tr", "500k", "1.5m", "2.000.000 vnd". A bare number after a price word is read as millions up to
206 * 100 ("under 10") and as thousands above ("under 500"); a bare number elsewhere stays a search term ("501").
207 */
208 static long amount(String number, String unit) {
209 double v = number.matches("\\d{1,3}([.,]\\d{3})+") ? Double.parseDouble(number.replaceAll("[.,]", ""))
210 : Double.parseDouble(number.replace(',', '.'));
211 String u = unit == null ? "" : unit;
212 switch (u) {
213 case "k", "nghin", "ngan", "thousand" -> v *= 1_000;
214 case "tr", "trieu", "m", "mil", "million", "millions" -> v *= 1_000_000;
215 case "" -> { if (v < 1000) v *= v <= 100 ? 1_000_000 : 1_000; }
216 default -> { }
217 }
218 return Math.round(v);
219 }
