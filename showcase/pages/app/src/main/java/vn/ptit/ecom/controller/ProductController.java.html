38 /** The category tree for the navigation (roots with their sub-categories). */
39 @GetMapping("/api/categories")
40 public List<CategoryNode> categories() {
41 List<Category> all = catalog.categories();
42 List<CategoryNode> roots = new ArrayList<>();
43 for (Category c : all)
44 if (c.parentId() == null) {
45 List<CategoryNode> children = all.stream().filter(k -> k.parentId() != null && k.parentId() == c.id())
46 .map(k -> new CategoryNode(k.id(), k.name(), k.slug(), List.of())).toList();
47 roots.add(new CategoryNode(c.id(), c.name(), c.slug(), children));
48 }
49 return roots;
50 }

62 Product p = pd.product();

64 return new Views.ProductPage(Views.ProductCard.of(p), p.getDescription(), p.getStatus().name(),
65 p.specs(),
66 parent == null ? null : parent.name(), parent == null ? null : parent.slug());
