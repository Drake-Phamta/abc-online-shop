12/**
13 * FR-S2, FR-S3: the text side of the AI search. Every ACTIVE product is a vector of its words; a word weighs more in
14 * the name than in the description, and less when many products use it (TF-IDF, weights learned from the catalogue).
15 * A query is a vector of its terms plus, at half weight, their related words; the score is the cosine of the two.
16 */
17public class ProductIndex {
18 static final double W_NAME = 3.0, W_BRAND = 2.5, W_CATEGORY = 2.0, W_DETAIL = 1.5, W_DESCRIPTION = 1.0;
19 static final double RELATED_WEIGHT = 0.5;
20 /** Below this score a product shares only a stray word with the query and is not shown. */
21 static final double MIN_SCORE = 0.05;
22
23 private record Doc(int productId, long price, Map<String, Double> vector, Map<String, String> fieldOf) { }
24
25 private final List<Doc> docs;
26 private final Map<String, Integer> df;
27 private final int n;

29 private ProductIndex(List<Doc> docs, Map<String, Integer> df) {
30 this.docs = docs;
31 this.df = df;
32 this.n = docs.size();
33 }

35 /** @param documents the ACTIVE products (AI01) as documents; the category text includes the parent category */
36 public static ProductIndex build(List<Document> documents) {
37 List<Map<String, Double>> tfs = new ArrayList<>();
38 List<Map<String, String>> fields = new ArrayList<>();
39 Map<String, Integer> df = new HashMap<>();
40 for (Document p : documents) {
41 Map<String, Double> tf = new HashMap<>();
42 Map<String, String> fieldOf = new HashMap<>();
43 Map<String, Double> best = new HashMap<>();
44 add(tf, fieldOf, best, p.name(), W_NAME, "name");
45 add(tf, fieldOf, best, p.brand(), W_BRAND, "brand");
46 add(tf, fieldOf, best, p.category(), W_CATEGORY, "category");
47 p.details().forEach((label, value) -> add(tf, fieldOf, best, value, W_DETAIL, label.toLowerCase(Locale.ROOT)));
48 add(tf, fieldOf, best, p.description(), W_DESCRIPTION, "description");
49 for (String t : tf.keySet()) df.merge(t, 1, Integer::sum);
50 tfs.add(tf);
51 fields.add(fieldOf);
52 }
53 int n = documents.size();
54 List<Doc> docs = new ArrayList<>();
55 for (int i = 0; i < n; i++) {
56 Map<String, Double> v = new HashMap<>();
57 for (Map.Entry<String, Double> e : tfs.get(i).entrySet())
58 v.put(e.getKey(), (1 + Math.log(e.getValue())) * idf(n, df.get(e.getKey())));
59 normalize(v);
60 docs.add(new Doc(documents.get(i).id(), documents.get(i).price(), v, fields.get(i)));
61 }
62 return new ProductIndex(docs, df);
63 }

65 private static void add(Map<String, Double> tf, Map<String, String> fieldOf, Map<String, Double> best,
66 String text, double weight, String field) {
67 for (String t : QueryInterpreter.tokens(text)) {
68 tf.merge(t, weight, Double::sum);
69 if (weight > best.getOrDefault(t, 0.0)) {
70 best.put(t, weight);
71 fieldOf.put(t, field);
72 }
73 }
74 }

76 private static double idf(int n, int df) {
77 return Math.log(1.0 + (double) n / df);
78 }

80 private static void normalize(Map<String, Double> v) {
81 double norm = Math.sqrt(v.values().stream().mapToDouble(x -> x * x).sum());
82 if (norm > 0) v.replaceAll((k, x) -> x / norm);
83 }

90 /** Up to k products, best first; ties keep the product id order, so the same query always ranks the same. */
91 public List<Hit> rank(SearchQuery q, int k) {
92 Map<String, Double> qv = new LinkedHashMap<>();
93 Map<String, String> via = new HashMap<>();
94 for (String t : q.terms()) if (df.containsKey(t)) qv.put(t, idf(n, df.get(t)));
95 q.related().forEach((t, rels) -> {
96 // weighted by the customer's word, not by the related word: a rare related word ("burgundy") must not
97 // outweigh the word the customer typed ("red")
98 double weight = RELATED_WEIGHT * idf(n, df.getOrDefault(t, 1));
99 for (String r : rels)
100 if (df.containsKey(r) && !qv.containsKey(r)) {
101 qv.put(r, weight);
102 via.put(r, t);
103 }
104 });
105 normalize(qv);
106 String priceReason = priceReason(q);
107 List<Doc> candidates = docs.stream()
108 .filter(d -> (q.minPrice() == null || d.price() >= q.minPrice()) && (q.maxPrice() == null || d.price() <= q.maxPrice()))
109 .toList();
110 List<Hit> hits = new ArrayList<>();
111 if (qv.isEmpty()) {
112 // only a price limit ("under 2 million"): every product in the range, cheapest first
113 if (priceReason == null || !q.terms().isEmpty()) return hits;
114 candidates.stream().sorted(Comparator.comparingLong(Doc::price).thenComparingInt(Doc::productId)).limit(k)
115 .forEach(d -> hits.add(new Hit(d.productId(), 1.0, List.of(priceReason))));
116 return hits;
117 }
118 long maxPrice = candidates.stream().mapToLong(Doc::price).max().orElse(1);
119 for (Doc d : candidates) {
120 // a related word stands in for its query term, it does not add to it: per query term the best of the
121 // term and its related words counts ("red sneakers": red canvas shoes beat burgundy boots)
122 Map<String, Double> best = new HashMap<>();
123 Map<String, String> bestWord = new HashMap<>();
124 for (Map.Entry<String, Double> e : qv.entrySet()) {
125 Double w = d.vector().get(e.getKey());
126 if (w == null) continue;
127 String origin = via.getOrDefault(e.getKey(), e.getKey());
128 double part = e.getValue() * w;
129 if (part > best.getOrDefault(origin, 0.0)) {
130 best.put(origin, part);
131 bestWord.put(origin, e.getKey());
132 }
133 }
134 double score = best.values().stream().mapToDouble(Double::doubleValue).sum();
135 if (score < MIN_SCORE) continue;
136 if (q.cheap()) score *= 1 + 0.25 * (1 - (double) d.price() / maxPrice);
137 List<String> reasons = new ArrayList<>();
138 best.entrySet().stream().sorted(Map.Entry.<String, Double>comparingByValue().reversed()).limit(3).forEach(e -> {
139 String word = bestWord.get(e.getKey());
140 String field = d.fieldOf().get(word);
141 reasons.add(field + ": “" + word + "”" + (word.equals(e.getKey()) ? "" : " (related to “" + e.getKey() + "”)"));
142 });
143 if (priceReason != null) reasons.add(priceReason);
144 if (q.cheap()) reasons.add("cheaper products ranked higher");
145 hits.add(new Hit(d.productId(), Math.round(score * 1000) / 1000.0, reasons));
146 }
147 hits.sort(Comparator.comparingDouble(Hit::score).reversed().thenComparingInt(Hit::productId));
148 return hits.size() > k ? new ArrayList<>(hits.subList(0, k)) : hits;
149 }
150
151 private static String priceReason(SearchQuery q) {
152 NumberFormat f = NumberFormat.getIntegerInstance(Locale.forLanguageTag("vi-VN"));
153 if (q.minPrice() != null && q.maxPrice() != null) return "price between " + f.format(q.minPrice()) + " and " + f.format(q.maxPrice()) + " ₫";
154 if (q.maxPrice() != null) return "price at most " + f.format(q.maxPrice()) + " ₫";
155 if (q.minPrice() != null) return "price at least " + f.format(q.minPrice()) + " ₫";
156 return null;
157 }
