27/**
28 * UC04 with the AI search of docs/SPEC.md section 6: text (typed or spoken) and photo queries. The indexes are built
29 * from the catalogue when the first search runs and again whenever the catalogue changed (FR-S7); the products of a
30 * result are always read fresh, so prices and stock are current.

32@Service

62 /** FR-S1, FR-S2, FR-S3, FR-S6. */
63 public SearchResult search(String text) {
64 long t0 = System.nanoTime();
65 if (text == null || text.isBlank()) throw new IllegalArgumentException("Type what you are looking for.");
66 List<Product> active = activeProducts();
67 Indexes ix = indexes(active);
68 SearchQuery q = ix.interpreter().interpret(text);
69 List<Hit> hits = ix.text().rank(q, TOP_K);
70 if (!hits.isEmpty()) return new SearchResult(Mode.TEXT, q, items(hits, active, true), false, ms(t0));
71 // AI04: nothing understood or nothing close enough, the keyword search of UC04 has the last word
72 List<Item> keyword = db.inTransaction(d -> d.products().search(text.trim(), null, ProductDAO.Sort.RELEVANCE)).stream()
73 .limit(TOP_K).map(p -> new Item(p, 0, List.of("keyword match in name, brand or category"))).toList();
74 return new SearchResult(Mode.KEYWORD, q, keyword, true, ms(t0));
75 }

77 /** FR-S5; AI05: the photo is decoded in memory and dropped after the search. */
78 public SearchResult searchByPhoto(byte[] photo) {
79 long t0 = System.nanoTime();
80 if (photo == null || photo.length == 0) throw new IllegalArgumentException("Choose a photo first.");
81 if (photo.length > MAX_PHOTO_BYTES) throw new IllegalArgumentException("The photo is larger than 5 MB.");
82 if (!isJpeg(photo) && !isPng(photo)) throw new IllegalArgumentException("The file is not a JPEG or PNG image.");
83 BufferedImage img;
84 try {
85 img = ImageIO.read(new ByteArrayInputStream(photo));
86 } catch (IOException e) {
87 img = null;
88 }
89 if (img == null) throw new IllegalArgumentException("The file is not a JPEG or PNG image.");
90 List<Product> active = activeProducts();
91 Indexes ix = indexes(active);
92 List<Hit> hits = ix.images().rank(img, TOP_K_PHOTO);
93 SearchQuery none = new SearchQuery("", List.of(), Map.of(), Map.of(), null, null, false);
94 return new SearchResult(Mode.PHOTO, none, items(hits, active, false), false, ms(t0));
95 }

101 /** AI07: in a text search an out-of-stock product counts 20 % less, so it stays in front only when clearly more relevant. */
102 static final double OUT_OF_STOCK_FACTOR = 0.8;
103
104 private static List<Item> items(List<Hit> hits, List<Product> active, boolean stockMatters) {
105 Map<Integer, Product> byId = active.stream().collect(Collectors.toMap(Product::getId, Function.identity()));
106 List<Item> out = new ArrayList<>();
107 for (Hit h : hits) {
108 Product p = byId.get(h.productId());
109 if (p == null) continue; // AI01: hidden since the index was built
110 if (!stockMatters || p.getStockQuantity() > 0) {
111 out.add(new Item(p, h.score(), h.reasons()));
112 } else {
113 List<String> reasons = new ArrayList<>(h.reasons());
114 reasons.add("out of stock: ranked lower");
115 out.add(new Item(p, Math.round(h.score() * OUT_OF_STOCK_FACTOR * 1000) / 1000.0, reasons));
116 }
117 }
118 out.sort(java.util.Comparator.comparingDouble(Item::score).reversed().thenComparingInt(i -> i.product().getId()));
119 return out;
120 }

122 /** The catalogue text in one string; when it differs from the indexed one, the indexes are rebuilt (FR-S7). */
123 private Indexes indexes(List<Product> active) {
124 String catalogue = active.stream().map(p -> p.getId() + "|" + p.getName() + "|" + p.getImageFile() + "|" + p.getPrice())
125 .collect(Collectors.joining("\n"));
126 Indexes ix = indexes;
127 if (ix != null && ix.catalogue().equals(catalogue)) return ix;
128 synchronized (this) {
129 if (indexes != null && indexes.catalogue().equals(catalogue)) return indexes;
130 Map<Integer, String> names = db.inTransaction(d -> d.products().categories()).stream()
131 .collect(Collectors.toMap(Category::id, Category::name));
132 ProductIndex text = ProductIndex.build(active.stream().map(p -> document(p, names)).toList());
133 Map<Integer, BufferedImage> pictures = new LinkedHashMap<>();
134 for (Product p : active) {
135 BufferedImage img = catalogueImage(p.getImageFile());
136 if (img != null) pictures.put(p.getId(), img);
137 }
138 indexes = new Indexes(catalogue, text, ImageIndex.build(pictures), new QueryInterpreter(text.vocabulary()));
139 return indexes;
140 }
141 }

143 /** The product as the AI component sees it; the category text carries the parent ("Shoes Fashion"). */
144 static Document document(Product p, Map<Integer, String> categoryNames) {
145 Integer parent = p.getCategory().parentId();
146 String category = p.getCategory().name() + (parent == null ? "" : " " + categoryNames.getOrDefault(parent, ""));
147 Map<String, String> details = new LinkedHashMap<>();
148 if (p.getDetail() != null) for (ProductDetail.Spec s : p.getDetail().specs()) details.put(s.label(), s.value());
149 return new Document(p.getId(), p.getPrice(), p.getName(), p.getBrand(), category, details, p.getDescription());
150 }

161 private static boolean isJpeg(byte[] b) {
162 return b.length > 3 && (b[0] & 0xFF) == 0xFF && (b[1] & 0xFF) == 0xD8 && (b[2] & 0xFF) == 0xFF;
163 }
164
165 private static boolean isPng(byte[] b) {
166 return b.length > 8 && (b[0] & 0xFF) == 0x89 && b[1] == 'P' && b[2] == 'N' && b[3] == 'G';
167 }
