14/**
15 * UC08 Make Payment. Two short transactions with the provider call between them: no database lock is held while the
16 * provider answers, and the result is recorded against the order only after it has come back (BR07).
17 */
18@Service
19public class PaymentService {
20 public record PaymentResult(boolean approved, Order order, String reason) { }
21
22 private final Database db;
23 private final CardProvider provider;
24
25 public PaymentService(Database db, CardProvider provider) {
26 this.db = db;
27 this.provider = provider;
28 }
29
30 public PaymentResult pay(int orderId, int customerId, CardData card) {
31 Order before = db.inTransaction(d -> OrderService.load(d, orderId, customerId));
32 if (before.getStatus() != OrderStatus.PENDING_PAYMENT)
33 throw new RuleViolation("BR07", "Order " + before.code() + " is not waiting for a card payment.");
34
35 CardProvider.Result result = provider.charge(before.code(), before.getTotalAmount(), card);
36
37 return db.inTransaction(d -> {
38 Order o = OrderService.load(d, orderId, customerId);
39 LocalDateTime now = LocalDateTime.now();
40 PayCredit payment = new PayCredit(null, o.getTotalAmount(), result.approved() ? Payment.Status.APPROVED
41 : Payment.Status.DECLINED, now, card.holder(), card.last4(), card.expiry(), result.token());
42 o.recordCardPayment(payment, now);
43 d.payments().saveAll(o.getId(), o.getPayments());
44 d.orders().saveProgress(o);
45 return new PaymentResult(result.approved(), o, result.reason());
46 });
47 }
