39 /** UC07 steps 1 to 4 and UC21: changed prices are taken over (BR03) and saved, blocking issues are reported. */
40 public Checkout checkout(int customerId) {
41 return db.inTransaction(d -> {
42 Customer customer = d.customers().findById(customerId).orElseThrow(() -> new NotFound("Customer not found."));
43 Cart cart = d.carts().findOrCreate(customerId);
44 List<CartIssue> issues = cart.validate();
45 d.carts().save(cart);
46 return new Checkout(customer, cart, issues, d.customers().addresses(customerId),
47 Order.Totals.of(cart.calcTotal(), customer.discountRate()));
48 });
49 }

51 /**
52 * UC07 steps 5 to 7: create the order from the cart (BR06 address, BR05 blocking issues, BR04 totals), refuse it when
53 * the total differs from the one the customer confirmed (BR03), reserve the stock of every line (BR05), store the
54 * order and empty the cart. A failure at any step rolls back all of them.
55 */
56 public Order placeOrder(int customerId, int addressId, PaymentMethod method, Long expectedTotal) {
57 return db.inTransaction(d -> {
58 Customer customer = d.customers().findById(customerId).orElseThrow(() -> new NotFound("Customer not found."));
59 Address address = d.customers().address(addressId)
60 .orElseThrow(() -> new RuleViolation("BR06", "Choose one of your delivery addresses."));
61 Cart cart = d.carts().findOrCreate(customerId);
62 Order order = Order.place(customer, cart, address, method, LocalDateTime.now());
63 if (expectedTotal != null && expectedTotal != order.getTotalAmount())
64 throw new RuleViolation("BR03", "Prices changed since you opened checkout. Please check the new total.",
65 List.of(Map.of("expectedTotal", expectedTotal, "newTotal", order.getTotalAmount())));
66 for (OrderItem i : order.getItems())
67 if (!d.products().reserveStock(i.productId(), i.quantity()))
68 throw new RuleViolation("BR05", i.productName() + " has just sold out in this quantity.");
69 d.orders().insert(order);
70 d.carts().clear(cart);
71 return order;
72 });
73 }

90 /** BR09: the customer may cancel before packing starts, sales staff until shipping; the stock is released (BR05). */
91 public OrderDetails cancel(int orderId, Integer customerId, Actor actor, String actorName) {
92 return db.inTransaction(d -> {
93 Order o = load(d, orderId, customerId);
94 o.cancel(actor, actorName, LocalDateTime.now());
95 for (OrderItem i : o.getItems()) d.products().releaseStock(i.productId(), i.quantity());
96 d.payments().saveAll(o.getId(), o.getPayments());
97 d.orders().saveProgress(o);
98 return details(d, o);
99 });
100 }
101
102 /** UC14: sales staff start packing (BR09). */
103 public OrderDetails process(int orderId, String staffName) {
104 return db.inTransaction(d -> {
105 Order o = load(d, orderId, null);
106 o.changeStatus(OrderStatus.PROCESSING, Actor.SALES, staffName, LocalDateTime.now(), "Packing started");
107 d.orders().saveProgress(o);
108 return details(d, o);
109 });
110 }

112 /** Loads an order the caller may see; another customer's order is reported as missing (BR08). */
113 static Order load(Database.Daos d, int orderId, Integer customerId) throws SQLException {
114 Order o = d.orders().findById(orderId).orElseThrow(() -> new NotFound("Order not found."));
115 if (customerId != null && o.getCustomerId() != customerId) throw new NotFound("Order not found.");
116 return o;
117 }
118
119 static OrderDetails details(Database.Daos d, Order o) throws SQLException {
120 Address a = d.customers().address(o.getAddressId()).orElse(null);
121 String customer = d.customers().findById(o.getCustomerId()).map(c -> c.getFullName().display()).orElse("");
122 return new OrderDetails(o, a, customer);
123 }
