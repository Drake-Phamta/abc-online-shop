12/**
13 * FR-S5: the photo side of the AI search. Every product image is described by three feature vectors: the colours
14 * of the object (the background is found from the border of the picture and left out), the directions of its edges
15 * in four regions, and its silhouette on a 6 x 6 grid. A photo is compared with every product by the cosine of
16 * each vector; the weighted sum is the similarity.
17 */
18public class ImageIndex {
19 static final int SIZE = 96;
20 static final double W_COLOUR = 0.55, W_EDGES = 0.25, W_SHAPE = 0.20;
21 /** Below this similarity the picture is not shown (failure analysis: unrelated photo). */
22 static final double MIN_SCORE = 0.6;
23
24 record Features(double[] colour, double[] edges, double[] shape) { }
25
26 private final Map<Integer, Features> features;
27
28 private ImageIndex(Map<Integer, Features> features) {
29 this.features = features;
30 }

32 /** @param images product id -> its catalogue image */
33 public static ImageIndex build(Map<Integer, BufferedImage> images) {
34 Map<Integer, Features> f = new LinkedHashMap<>();
35 images.forEach((id, img) -> f.put(id, features(img)));
36 return new ImageIndex(f);
37 }

39 public int size() {
40 return features.size();
41 }

43 /** Up to k products whose image looks most like the photo, best first. */
44 public List<Hit> rank(BufferedImage photo, int k) {
45 Features q = features(photo);
46 List<Hit> hits = new ArrayList<>();
47 features.forEach((id, f) -> {
48 double colour = dot(q.colour(), f.colour()), edges = dot(q.edges(), f.edges()), shape = dot(q.shape(), f.shape());
49 double score = W_COLOUR * colour + W_EDGES * edges + W_SHAPE * shape;
50 if (score < MIN_SCORE) return;
51 hits.add(new Hit(id, Math.round(score * 1000) / 1000.0, List.of(
52 "colours alike " + Math.round(colour * 100) + " %",
53 "outline alike " + Math.round((W_EDGES * edges + W_SHAPE * shape) / (W_EDGES + W_SHAPE) * 100) + " %")));
54 });
55 hits.sort(Comparator.comparingDouble(Hit::score).reversed().thenComparingInt(Hit::productId));
56 return hits.size() > k ? new ArrayList<>(hits.subList(0, k)) : hits;
57 }

59 static Features features(BufferedImage source) {
60 BufferedImage img = new BufferedImage(SIZE, SIZE, BufferedImage.TYPE_INT_RGB);
61 Graphics2D g = img.createGraphics();
62 g.setRenderingHint(RenderingHints.KEY_INTERPOLATION, RenderingHints.VALUE_INTERPOLATION_BILINEAR);
63 g.drawImage(source, 0, 0, SIZE, SIZE, null);
64 g.dispose();
65 int[] rgb = img.getRGB(0, 0, SIZE, SIZE, null, 0, SIZE);
66 boolean[] object = objectMask(rgb);
67 return new Features(colours(rgb, object), edges(rgb, object), silhouette(object));
68 }

70 /** Background = pixels close to the mean colour of a 3-pixel border; if that leaves almost nothing, all pixels. */
71 static boolean[] objectMask(int[] rgb) {
72 long r = 0, gr = 0, b = 0;
73 int count = 0;
74 for (int y = 0; y < SIZE; y++)
75 for (int x = 0; x < SIZE; x++)
76 if (x < 3 || y < 3 || x >= SIZE - 3 || y >= SIZE - 3) {
77 int p = rgb[y * SIZE + x];
78 r += (p >> 16) & 255; gr += (p >> 8) & 255; b += p & 255;
79 count++;
80 }
81 double br = (double) r / count, bg = (double) gr / count, bb = (double) b / count;
82 boolean[] mask = new boolean[rgb.length];
83 int on = 0;
84 for (int i = 0; i < rgb.length; i++) {
85 int p = rgb[i];
86 double d = Math.sqrt(sq(((p >> 16) & 255) - br) + sq(((p >> 8) & 255) - bg) + sq((p & 255) - bb));
87 mask[i] = d > 40;
88 if (mask[i]) on++;
89 }
90 if (on < rgb.length / 20) java.util.Arrays.fill(mask, true);
91 return mask;
92 }

94 /** 18 hue bins weighted by saturation, plus black, grey and white; square root of the shares (Hellinger). */
95 static double[] colours(int[] rgb, boolean[] object) {
96 double[] h = new double[21];
97 float[] hsb = new float[3];
98 for (int i = 0; i < rgb.length; i++) {
99 if (!object[i]) continue;
100 int p = rgb[i];
101 java.awt.Color.RGBtoHSB((p >> 16) & 255, (p >> 8) & 255, p & 255, hsb);
102 if (hsb[1] < 0.18 || hsb[2] < 0.18) h[18 + (hsb[2] < 0.3 ? 0 : hsb[2] < 0.75 ? 1 : 2)] += 1;
103 else h[Math.min(17, (int) (hsb[0] * 18))] += 0.5 + hsb[1] / 2;
104 }
105 double sum = 0;
106 for (double v : h) sum += v;
107 for (int i = 0; i < h.length; i++) h[i] = sum == 0 ? 0 : Math.sqrt(h[i] / sum);
108 return unit(h);
109 }
110
111 /** Sobel edge directions (8 bins over 180 degrees) in each quarter of the picture, weighted by edge strength. */
112 static double[] edges(int[] rgb, boolean[] object) {
113 double[] gray = new double[rgb.length];
114 for (int i = 0; i < rgb.length; i++) {
115 int p = rgb[i];
116 gray[i] = 0.299 * ((p >> 16) & 255) + 0.587 * ((p >> 8) & 255) + 0.114 * (p & 255);
117 }
118 double[] h = new double[32];
119 for (int y = 1; y < SIZE - 1; y++)
120 for (int x = 1; x < SIZE - 1; x++) {
121 double gx = gray[(y - 1) * SIZE + x + 1] + 2 * gray[y * SIZE + x + 1] + gray[(y + 1) * SIZE + x + 1]
122 - gray[(y - 1) * SIZE + x - 1] - 2 * gray[y * SIZE + x - 1] - gray[(y + 1) * SIZE + x - 1];
123 double gy = gray[(y + 1) * SIZE + x - 1] + 2 * gray[(y + 1) * SIZE + x] + gray[(y + 1) * SIZE + x + 1]
124 - gray[(y - 1) * SIZE + x - 1] - 2 * gray[(y - 1) * SIZE + x] - gray[(y - 1) * SIZE + x + 1];
125 double m = Math.hypot(gx, gy);
126 if (m < 40) continue;
127 double angle = Math.atan2(gy, gx);
128 if (angle < 0) angle += Math.PI;
129 int bin = Math.min(7, (int) (angle / Math.PI * 8));
130 int cell = (y < SIZE / 2 ? 0 : 2) + (x < SIZE / 2 ? 0 : 1);
131 h[cell * 8 + bin] += m;
132 }
133 return unit(h);
134 }
135
136 /** Share of object pixels in each cell of a 6 x 6 grid. */
137 static double[] silhouette(boolean[] object) {
138 double[] h = new double[36];
139 int cell = SIZE / 6;
140 for (int i = 0; i < object.length; i++)
141 if (object[i]) h[(i / SIZE / cell) * 6 + (i % SIZE) / cell] += 1;
142 return unit(h);
143 }

145 private static double[] unit(double[] v) {
146 double norm = 0;
147 for (double x : v) norm += x * x;
148 norm = Math.sqrt(norm);
149 if (norm > 0) for (int i = 0; i < v.length; i++) v[i] /= norm;
150 return v;
151 }

153 private static double dot(double[] a, double[] b) {
154 double s = 0;
155 for (int i = 0; i < a.length; i++) s += a[i] * b[i];
156 return s;
157 }

159 private static double sq(double x) {
160 return x * x;
