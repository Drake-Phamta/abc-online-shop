16/** UC02 Login for customers and staff. The session holds who is logged in; the account check is AuthService's. */
17@RestController
18@RequestMapping("/api/auth")
19public class AuthController {
20 record LoginRequest(String email, String password) { }
21
22 private final AuthService auth;
23
24 public AuthController(AuthService auth) {
25 this.auth = auth;
26 }
27
28 @PostMapping("/login")
29 public Map<String, Object> login(@RequestBody LoginRequest req, HttpServletRequest http) {
30 Account a = auth.login(req.email(), req.password())
31 .orElseThrow(() -> new ApiErrors.Unauthorized("Email or password is incorrect."));
32 HttpSession old = http.getSession(false);
33 if (old != null) old.invalidate(); // a new session id after login (no session fixation)
34 http.getSession(true).setAttribute(CurrentUser.KEY, new CurrentUser(a.role(), a.id(), a.displayName()));
35 return me(http.getSession(false));
36 }
37
38 @PostMapping("/logout")
39 public Map<String, Object> logout(HttpServletRequest http) {
40 HttpSession s = http.getSession(false);
41 if (s != null) s.invalidate();
42 return Map.of("loggedIn", false);
43 }
44
45 @GetMapping("/me")
46 public Map<String, Object> me(HttpSession session) {
47 CurrentUser u = CurrentUser.of(session);
48 Map<String, Object> out = new LinkedHashMap<>();
49 out.put("loggedIn", u != null);
50 if (u == null) return out;
51 out.put("role", u.role());
52 out.put("name", u.name());
53 if (u.role() == Role.CUSTOMER) {
54 AuthService.CustomerProfile p = auth.profile(u.id());
55 out.put("kind", p.customer().kindLabel());
56 out.put("discountRate", p.customer().discountRate());
57 out.put("cartCount", p.cartItems());
58 }
59 return out;
60 }
