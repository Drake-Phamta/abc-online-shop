11/** UC14 shipping steps with the data exchanged with the Shipping Provider (UC16): ship and deliver. */
12@RestController

28 @PostMapping("/api/staff/orders/{id}/deliver")
29 public Views.OrderView deliver(@PathVariable int id, HttpSession session) {
30 CurrentUser u = CurrentUser.require(session, Role.SALES);
31 return OrderController.view(shipping.deliver(id, u.name()), u);
32 }
