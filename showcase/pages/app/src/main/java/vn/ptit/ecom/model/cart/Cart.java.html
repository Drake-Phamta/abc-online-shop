10/** A customer's cart; its products are the current product rows, its prices the ones shown when added. */
11public class Cart {
12 private final int id;
13 private final int customerId;
14 private final List<CartItem> items;
15
16 public Cart(int id, int customerId, List<CartItem> items) {
17 this.id = id;
18 this.customerId = customerId;
19 this.items = new ArrayList<>(items);
20 }
21
22 /** UC06: BR01 active product, BR02 one line per product and quantity within stock. */
23 public CartItem addItem(Product product, int quantity) {
24 if (quantity < 1) throw new RuleViolation("BR02", "Quantity must be at least 1.");
25 if (product.getStatus() != Product.Status.ACTIVE)
26 throw new RuleViolation("BR01", product.getName() + " is not for sale.");
27 Optional<CartItem> line = find(product.getId());
28 int wanted = quantity + line.map(CartItem::getQuantity).orElse(0);
29 if (wanted > product.getStockQuantity())
30 throw new RuleViolation("BR02", "Only " + product.getStockQuantity() + " of " + product.getName() + " in stock.");
31 if (line.isPresent()) {
32 line.get().setQuantity(wanted);
33 return line.get();
34 }
35 CartItem item = new CartItem(product, quantity, product.getPrice());
36 items.add(item);
37 return item;
38 }
39
40 /** UC06 step 3. */
41 public void setQuantity(int productId, int quantity) {
42 CartItem line = find(productId).orElseThrow(() -> new RuleViolation("BR02", "This product is not in the cart."));
43 if (quantity < 1) throw new RuleViolation("BR02", "Quantity must be at least 1; remove the line instead.");
44 if (quantity > line.getProduct().getStockQuantity())
45 throw new RuleViolation("BR02", "Only " + line.getProduct().getStockQuantity() + " in stock.");
46 line.setQuantity(quantity);
47 }
48
49 public void removeItem(int productId) {
50 items.removeIf(i -> i.getProduct().getId() == productId);
51 }
52
53 /**
54 * UC21 Validate Cart (BR03): compares every line with the current product. Changed prices are taken over and
55 * reported; unavailable products and missing stock are reported as blocking.
56 */
57 public List<CartIssue> validate() {
58 List<CartIssue> issues = new ArrayList<>();
59 for (CartItem i : items) {
60 Product p = i.getProduct();
61 if (!p.isAvailable(i.getQuantity())) { // the product answers; the cart only names the reason for the customer
62 if (p.getStatus() != Product.Status.ACTIVE)
63 issues.add(new CartIssue(p.getId(), p.getName(), CartIssue.Kind.UNAVAILABLE, i.getCartPrice(), p.getPrice(), 0));
64 else
65 issues.add(new CartIssue(p.getId(), p.getName(), CartIssue.Kind.NOT_ENOUGH_STOCK, i.getCartPrice(),
66 p.getPrice(), p.getStockQuantity()));
67 }
68 if (i.getCartPrice() != p.getPrice()) {
69 issues.add(new CartIssue(p.getId(), p.getName(), CartIssue.Kind.PRICE_CHANGED, i.getCartPrice(), p.getPrice(),
70 p.getStockQuantity()));
71 i.setCartPrice(p.getPrice());
72 }
73 }
74 return issues;
75 }
76
77 public long calcTotal() {
78 return items.stream().mapToLong(CartItem::subTotal).sum();
79 }
80
81 public boolean isEmpty() { return items.isEmpty(); }
82
83 private Optional<CartItem> find(int productId) {
84 return items.stream().filter(i -> i.getProduct().getId() == productId).findFirst();
85 }
86
87 public int getId() { return id; }
88 public int getCustomerId() { return customerId; }
89 public List<CartItem> getItems() { return Collections.unmodifiableList(items); }
