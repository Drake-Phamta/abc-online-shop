55 /** The amounts of BR04 for a subtotal and a discount rate. */
56 public record Totals(long subtotal, long discount, long shippingFee, long total) {
57 public static Totals of(long subtotal, BigDecimal discountRate) {
58 long discount = BigDecimal.valueOf(subtotal).multiply(discountRate).setScale(0, RoundingMode.HALF_UP).longValue();
59 long afterDiscount = subtotal - discount;
60 long fee = afterDiscount >= FREE_SHIPPING_FROM ? 0 : SHIPPING_FEE;
61 return new Totals(subtotal, discount, fee, afterDiscount + fee);
62 }
63 }

65 /**
66 * UC07 step 6: builds the order from a validated cart. Blocking cart issues (UC07 A1) and a foreign address
67 * (BR06) are refused. The caller reserves the stock (BR05) and stores the order in the same transaction.
68 */
69 public static Order place(Customer customer, Cart cart, Address address, PaymentMethod method, LocalDateTime now) {
70 if (address.customerId() != customer.getId())
71 throw new RuleViolation("BR06", "The delivery address does not belong to this customer.");
72 if (cart.isEmpty()) throw new RuleViolation("BR02", "The cart is empty.");
73 List<CartIssue> blocking = cart.validate().stream().filter(i -> i.kind().isBlocking()).toList();
74 if (!blocking.isEmpty())
75 throw new RuleViolation("BR05", "Some products cannot be ordered in this quantity.", blocking);
76 List<OrderItem> lines = new ArrayList<>();
77 for (CartItem i : cart.getItems())
78 lines.add(new OrderItem(null, i.getProduct().getId(), i.getProduct().getName(), i.getQuantity(), i.getCartPrice()));
79 Totals t = Totals.of(cart.calcTotal(), customer.discountRate());
80 OrderStatus first = method == PaymentMethod.CARD ? OrderStatus.PENDING_PAYMENT : OrderStatus.PLACED;
81 String note = method == PaymentMethod.CARD ? "Order placed, waiting for card payment" : "Order placed, cash on delivery";
82 OrderEvent placed = new OrderEvent(null, first, now, customer.getFullName().display(), note);
83 return new Order(null, customer.getId(), address.id(), now, first, method, t.subtotal(), t.discount(),
84 t.shippingFee(), t.total(), lines, List.of(placed), List.of(), null);
85 }

87 /** BR09: the only way the status changes. */
88 public void changeStatus(OrderStatus to, Actor actor, String actorName, LocalDateTime now, String note) {
89 if (!status.canMoveTo(to, actor))
90 throw new RuleViolation("BR09", "Order " + code() + " cannot go from " + status + " to " + to + ".");
91 status = to;
92 events.add(new OrderEvent(null, to, now, actorName, note));

95 /** UC08: records a provider result; only an approved card payment makes the order PAID (BR07). */
96 public void recordCardPayment(PayCredit payment, LocalDateTime now) {
97 if (paymentMethod != PaymentMethod.CARD || status != OrderStatus.PENDING_PAYMENT)
98 throw new RuleViolation("BR07", "Order " + code() + " is not waiting for a card payment.");
99 if (payment.getAmount() != totalAmount)
100 throw new RuleViolation("BR07", "The payment amount does not match the order total.");
101 payments.add(payment);
102 String card = "card ending " + payment.getCardLast4();
103 if (payment.getStatus() == Payment.Status.APPROVED)
104 changeStatus(OrderStatus.PAID, Actor.PROVIDER, "Payment provider", now, "Paid by " + card);
105 else
106 events.add(new OrderEvent(null, status, now, "Payment provider", "Payment declined, " + card));
107 }

116 public Shipping ship(String carrier, String trackingNumber, String staffName, LocalDateTime now) {
117 if (carrier == null || carrier.isBlank() || trackingNumber == null || trackingNumber.isBlank())
118 throw new RuleViolation("BR10", "Carrier and tracking number are required to ship an order.");
119 changeStatus(OrderStatus.SHIPPED, Actor.SALES, staffName, now, "Handed to " + carrier + ", tracking " + trackingNumber);
120 shipping = new Shipping(null, addressId, carrier, trackingNumber, shippingFee, Shipping.Status.IN_TRANSIT, now, null);
121 return shipping;
122 }
