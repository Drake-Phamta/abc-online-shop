8/** UC14 shipping steps with the data exchanged with the Shipping Provider (UC16): ship and deliver. */
9@Service
10public class ShippingService {
11 private final Database db;
12
13 public ShippingService(Database db) {
14 this.db = db;
15 }
16
17 /** BR10: one shipment per order, created when a PROCESSING order is handed to the carrier. */
18 public OrderService.OrderDetails ship(int orderId, String carrier, String trackingNumber, String staffName) {
19 return db.inTransaction(d -> {
20 Order o = OrderService.load(d, orderId, null);
21 o.ship(carrier, trackingNumber, staffName, LocalDateTime.now());
22 d.shipping().save(o.getId(), o.getShipping());
23 d.orders().saveProgress(o);
24 return OrderService.details(d, o);
25 });
26 }
27
28 /** BR11: delivering a cash-on-delivery order records its cash payment. */
29 public OrderService.OrderDetails deliver(int orderId, String staffName) {
30 return db.inTransaction(d -> {
31 Order o = OrderService.load(d, orderId, null);
32 o.deliver(staffName, LocalDateTime.now());
33 d.shipping().save(o.getId(), o.getShipping());
34 d.payments().saveAll(o.getId(), o.getPayments());
35 d.orders().saveProgress(o);
36 return OrderService.details(d, o);
37 });
38 }
