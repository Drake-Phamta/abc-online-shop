62 @Override
63 public List<Product> search(String keyword, String categorySlug, Sort sort) throws SQLException {
64 StringBuilder sql = new StringBuilder(SELECT).append("WHERE p.status = 'ACTIVE' ");
65 List<Object> args = new ArrayList<>();
66 if (categorySlug != null && !categorySlug.isBlank()) {
67 sql.append("AND (c.slug = ? OR c.parent_id = (SELECT category_id FROM category WHERE slug = ?)) ");
68 args.add(categorySlug);
69 args.add(categorySlug);
70 }
71 String like = null;
72 if (keyword != null && !keyword.isBlank()) {
73 like = "%" + keyword.trim().toLowerCase() + "%";
74 sql.append("AND (lower(p.name) LIKE ? OR lower(coalesce(p.brand, '')) LIKE ? OR lower(c.name) LIKE ?) ");
75 args.add(like);
76 args.add(like);
77 args.add(like);
78 }
79 switch (sort == null ? Sort.RELEVANCE : sort) {
80 case PRICE_ASC -> sql.append("ORDER BY p.price ASC");
81 case PRICE_DESC -> sql.append("ORDER BY p.price DESC");
82 case NEWEST -> sql.append("ORDER BY p.created_at DESC, p.product_id DESC");
83 default -> {
84 if (like != null) {
85 sql.append("ORDER BY CASE WHEN lower(p.name) LIKE ? THEN 0 ELSE 1 END, p.name");
86 args.add(like);
87 } else {
88 sql.append("ORDER BY c.category_id, p.product_id");
89 }
90 }
91 }
92 List<Product> rows = new ArrayList<>();
93 try (PreparedStatement ps = Jdbc.prepare(connection, sql.toString(), args.toArray()); ResultSet rs = ps.executeQuery()) {
94 while (rs.next()) rows.add(map(rs, null));
95 }
96 List<Product> out = new ArrayList<>();
97 for (Product p : rows) out.add(withDetail(p));
98 return out;
99 }

101 /** The product cards show the first values of the detail (catalogue sizes are small, one query per card is fine). */
102 private Product withDetail(Product p) throws SQLException {
103 DetailKind kind = p.getCategory().detailKind();
104 if (kind == null) return p;
105 return new Product(p.getId(), p.getSku(), p.getName(), p.getBrand(), p.getPrice(), p.getStockQuantity(),
106 p.getDescription(), p.getImageFile(), p.getStatus(), p.getCategory(), detail(kind, p.getId()));
107 }

109 @Override
110 public Optional<Product> findById(int productId) throws SQLException {
111 try (PreparedStatement ps = Jdbc.prepare(connection, SELECT + "WHERE p.product_id = ?", productId);
112 ResultSet rs = ps.executeQuery()) {
113 if (!rs.next()) return Optional.empty();
114 DetailKind kind = kind(rs.getString("detail_kind"));
115 return Optional.of(map(rs, kind == null ? null : detail(kind, productId)));
116 }

119 @Override
120 public Map<Integer, Product> findByIds(Collection<Integer> productIds) throws SQLException {
121 Map<Integer, Product> out = new LinkedHashMap<>();
122 if (productIds.isEmpty()) return out;
123 String marks = String.join(",", java.util.Collections.nCopies(productIds.size(), "?"));
124 try (PreparedStatement ps = Jdbc.prepare(connection, SELECT + "WHERE p.product_id IN (" + marks + ")", productIds.toArray());
125 ResultSet rs = ps.executeQuery()) {
126 while (rs.next()) {
127 Product p = map(rs, null);
128 out.put(p.getId(), p);
129 }
130 }
131 return out;
132 }

160 private ProductDetail detail(DetailKind kind, int productId) throws SQLException {
161 try (PreparedStatement ps = Jdbc.prepare(connection, "SELECT * FROM " + kind.table() + " WHERE product_id = ?", productId);
162 ResultSet rs = ps.executeQuery()) {
163 return rs.next() ? DETAILS.get(kind).map(rs) : null;
164 }

167 static Product map(ResultSet rs, ProductDetail detail) throws SQLException {
168 Category category = new Category(rs.getInt("category_id"), Jdbc.intOrNull(rs, "parent_id"),
169 rs.getString("category_name"), rs.getString("slug"), kind(rs.getString("detail_kind")));
170 return new Product(rs.getInt("product_id"), rs.getString("sku"), rs.getString("name"), rs.getString("brand"),
171 rs.getLong("price"), rs.getInt("stock_quantity"), rs.getString("description"), rs.getString("image_file"),
172 Product.Status.valueOf(rs.getString("status")), category, detail);
173 }
