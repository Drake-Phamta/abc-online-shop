27 @Override
28 public void insert(Order o) throws SQLException {
29 int id = Jdbc.insert(connection, "INSERT INTO customer_order (customer_id, address_id, order_date, status, payment_method, "
30 + "subtotal, discount, shipping_fee, total_amount) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)",
31 o.getCustomerId(), o.getAddressId(), Jdbc.ts(o.getOrderDate()), o.getStatus().name(),
32 o.getPaymentMethod().name(), o.getSubtotal(), o.getDiscount(), o.getShippingFee(), o.getTotalAmount());
33 o.assignId(id);
34 for (OrderItem i : o.getItems())
35 Jdbc.update(connection, "INSERT INTO order_item (order_id, product_id, product_name, quantity, unit_price) "
36 + "VALUES (?, ?, ?, ?, ?)", id, i.productId(), i.productName(), i.quantity(), i.unitPrice());
37 insertNewEvents(o);
38 }

69 private void insertNewEvents(Order o) throws SQLException {
70 for (OrderEvent e : o.getEvents())
71 if (e.id() == null)
72 Jdbc.update(connection, "INSERT INTO order_event (order_id, status, occurred_at, actor, note) VALUES (?, ?, ?, ?, ?)",

76 private List<OrderSummary> summaries(String where, Object... args) throws SQLException {
77 String sql = "SELECT o.*, c.first_name, c.middle_name, c.last_name, "
78 + "(SELECT sum(quantity) FROM order_item i WHERE i.order_id = o.order_id) AS item_count, "
79 + "(SELECT product_name FROM order_item i WHERE i.order_id = o.order_id ORDER BY order_item_id LIMIT 1) AS first_item "
80 + "FROM customer_order o JOIN customer c ON c.customer_id = o.customer_id " + where
81 + " ORDER BY o.order_date DESC, o.order_id DESC";
82 List<OrderSummary> out = new ArrayList<>();
83 try (PreparedStatement ps = Jdbc.prepare(connection, sql, args); ResultSet rs = ps.executeQuery()) {
84 while (rs.next()) {
85 String middle = rs.getString("middle_name");
86 String name = rs.getString("last_name") + (middle == null ? "" : " " + middle) + " " + rs.getString("first_name");
87 int id = rs.getInt("order_id");
88 out.add(new OrderSummary(id, String.format("ORD-%06d", id), Jdbc.ts(rs.getString("order_date")),
89 OrderStatus.valueOf(rs.getString("status")), rs.getString("payment_method"),
90 rs.getLong("total_amount"), rs.getInt("item_count"), rs.getString("first_item"),
91 rs.getInt("customer_id"), name));
92 }
93 }
94 return out;
95 }

119 private List<Payment> payments(int orderId) throws SQLException {
120 String sql = "SELECT p.*, cr.card_holder, cr.card_last4, cr.expiry, cr.provider_token, ca.cash_received, ca.change_amount "
121 + "FROM payment p LEFT JOIN pay_credit cr ON cr.payment_id = p.payment_id "
122 + "LEFT JOIN pay_cash ca ON ca.payment_id = p.payment_id WHERE p.order_id = ? ORDER BY p.payment_id";
123 List<Payment> out = new ArrayList<>();
124 try (PreparedStatement ps = Jdbc.prepare(connection, sql, orderId); ResultSet rs = ps.executeQuery()) {
125 while (rs.next()) {
126 Payment.Status status = Payment.Status.valueOf(rs.getString("status"));
127 if ("CARD".equals(rs.getString("method")))
128 out.add(new PayCredit(rs.getInt("payment_id"), rs.getLong("amount"), status, Jdbc.ts(rs.getString("payment_date")),
129 rs.getString("card_holder"), rs.getString("card_last4"), rs.getString("expiry"),
130 rs.getString("provider_token")));
131 else
132 out.add(new PayCash(rs.getInt("payment_id"), rs.getLong("amount"), status, Jdbc.ts(rs.getString("payment_date")),
133 rs.getLong("cash_received"), rs.getLong("change_amount")));
134 }
135 }
136 return out;
137 }
