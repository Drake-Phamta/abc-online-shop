15/** UC04 search with the AI search (docs/SPEC.md section 6): typed or spoken text, or a photo. Guests may search. */

18 /** AI02: how the query was understood, in words the result page can show. */
19 record Understood(List<String> terms, List<String> related, List<String> corrections, Long minPrice, Long maxPrice,
20 boolean cheap) {
21 static Understood of(SearchQuery q) {
22 List<String> related = new ArrayList<>();
23 q.related().forEach((t, rels) -> related.add(t + " → " + String.join(", ", rels)));
24 List<String> corrections = new ArrayList<>();
25 q.corrections().forEach((from, to) -> corrections.add(from + " → " + to));
26 return new Understood(q.terms(), related, corrections, q.minPrice(), q.maxPrice(), q.cheap());
27 }
28 }
29
30 record Found(Views.ProductCard product, double score, List<String> reasons) { }
31
32 record SearchView(String mode, String query, Understood understood, boolean fallback, long tookMs, List<Found> results) {
33 static SearchView of(SearchService.SearchResult r) {
34 return new SearchView(r.mode().name(), r.understood().text(), Understood.of(r.understood()), r.fallback(), r.tookMs(),
35 r.items().stream().map(i -> new Found(Views.ProductCard.of(i.product()), i.score(), i.reasons())).toList());
36 }
37 }
