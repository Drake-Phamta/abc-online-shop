13/** UC08 Make Payment. The card data is checked here (format, Luhn, expiry: UC08 A2); the payment itself is PaymentService's. */
14@RestController
15public class PaymentController {
16 record CardPayment(String cardHolder, String cardNumber, String expiry, String cvv) { }
17
18 private final PaymentService payments;
19
20 public PaymentController(PaymentService payments) {
21 this.payments = payments;
22 }
23
24 @PostMapping("/api/orders/{id}/payments")
25 public Map<String, Object> pay(@PathVariable int id, @RequestBody CardPayment req, HttpSession session) {
26 CurrentUser u = CurrentUser.require(session, Role.CUSTOMER);
27 CardData card = new CardData(req.cardHolder(), req.cardNumber(), req.expiry(), req.cvv());
28 PaymentService.PaymentResult r = payments.pay(id, u.id(), card);
29 return Map.of("approved", r.approved(), "status", r.order().getStatus(),
30 "message", r.approved() ? "Payment approved." : r.reason() + ". Your order is kept; try another card.");
31 }
