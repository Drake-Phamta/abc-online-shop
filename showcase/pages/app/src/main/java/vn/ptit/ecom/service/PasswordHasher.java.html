3import java.security.GeneralSecurityException;
4import java.security.MessageDigest;
5import java.util.Base64;
6import javax.crypto.SecretKeyFactory;
7import javax.crypto.spec.PBEKeySpec;

9/** Checks stored hashes "pbkdf2$iterations$salt$hash" (PBKDF2-HMAC-SHA256, Base64); passwords are never stored. */
10final class PasswordHasher {
11 private PasswordHasher() { }
12
13 static boolean matches(String password, String stored) {
14 String[] parts = stored == null ? new String[0] : stored.split("\\$");
15 if (parts.length != 4 || !parts[0].equals("pbkdf2")) return false;
16 byte[] salt = Base64.getDecoder().decode(parts[2]);
17 byte[] expected = Base64.getDecoder().decode(parts[3]);
18 return MessageDigest.isEqual(expected, derive(password, salt, Integer.parseInt(parts[1]), expected.length * 8));
19 }
20
21 static byte[] derive(String password, byte[] salt, int iterations, int bits) {
22 try {
23 PBEKeySpec spec = new PBEKeySpec(password.toCharArray(), salt, iterations, bits);
24 return SecretKeyFactory.getInstance("PBKDF2WithHmacSHA256").generateSecret(spec).getEncoded();
25 } catch (GeneralSecurityException e) {
26 throw new IllegalStateException(e);
27 }
28 }
