3import java.util.EnumMap;
4import java.util.EnumSet;
5import java.util.Map;
6import java.util.Set;

8/** Order lifecycle of docs/SPEC.md section 3 (BR09): one transition table instead of scattered checks. */
9public enum OrderStatus {
10 PENDING_PAYMENT, PLACED, PAID, PROCESSING, SHIPPED, DELIVERED, CANCELLED;
11
12 /** Who causes a transition. PROVIDER is the payment provider's approval (UC08). */
13 public enum Actor { CUSTOMER, SALES, PROVIDER }
14
15 private static final Map<OrderStatus, Map<OrderStatus, Set<Actor>>> TRANSITIONS = new EnumMap<>(OrderStatus.class);
16
17 static {
18 allow(PENDING_PAYMENT, PAID, Actor.PROVIDER);
19 allow(PENDING_PAYMENT, CANCELLED, Actor.CUSTOMER);
20 allow(PLACED, PROCESSING, Actor.SALES);
21 allow(PLACED, CANCELLED, Actor.CUSTOMER, Actor.SALES);
22 allow(PAID, PROCESSING, Actor.SALES);
23 allow(PAID, CANCELLED, Actor.SALES);
24 allow(PROCESSING, SHIPPED, Actor.SALES);
25 allow(PROCESSING, CANCELLED, Actor.SALES);
26 allow(SHIPPED, DELIVERED, Actor.SALES);
27 }
28
29 private static void allow(OrderStatus from, OrderStatus to, Actor... actors) {
30 TRANSITIONS.computeIfAbsent(from, k -> new EnumMap<>(OrderStatus.class)).put(to, Set.of(actors));
31 }
32
33 public boolean canMoveTo(OrderStatus to, Actor actor) {
34 return TRANSITIONS.getOrDefault(this, Map.of()).getOrDefault(to, Set.of()).contains(actor);
35 }
36
37 /** The statuses this actor may move the order to next (drives the buttons in the UI). */
38 public Set<OrderStatus> next(Actor actor) {
39 Set<OrderStatus> out = EnumSet.noneOf(OrderStatus.class);
40 TRANSITIONS.getOrDefault(this, Map.of()).forEach((to, actors) -> { if (actors.contains(actor)) out.add(to); });
41 return out;
42 }
