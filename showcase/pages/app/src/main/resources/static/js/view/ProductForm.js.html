61 async showCatalogue(category, sort, q) {
62 const nav = this.app.nav;
63 sort = sort || "RELEVANCE";
64 mount(this.app.view, loading());
65 try {
66 const params = new URLSearchParams({ sort });
67 if (category && category !== "all") params.set("category", category);
68 if (q) params.set("q", q);
69 const [tree, products] = await Promise.all([api("GET", "/api/categories"), api("GET", `${this.controller}?${params}`)]);
70 if (!this.app.current(nav)) return; // a newer navigation owns the page
71 const all = tree.flatMap((r) => [r, ...r.children]);
72 const current = all.find((c) => c.slug === category);
73 const title = q ? html`Results for “${q}”` : current ? current.name : "All products";
74 const base = q ? `search?q=${encodeURIComponent(q)}&` : `c/${category}?`;
75 mount(this.app.view, html`
76 <div class="catalogue">
77 <aside class="tree" data-trace="GET /api/categories · ProductController.categories · CatalogService.categories · ProductDAO.categories · category (tree)">
78 <a href="#/c/all" class="${!current && !q ? "on" : ""}">All products</a>
79 ${tree.map((r) => html`<h4>${r.name}</h4>
80 <a href="#/c/${r.slug}" class="parent ${current?.slug === r.slug ? "on" : ""}">All ${r.name.toLowerCase()}</a>
81 ${r.children.map((c) => html`<a href="#/c/${c.slug}" class="${current?.slug === c.slug ? "on" : ""}">${c.name}</a>`)}`)}
82 </aside>
83 <section>
84 <div class="page-head"><div><p class="eyebrow">${products.length} products</p><h1>${title}</h1></div></div>
85 <div class="toolbar">
86 <div class="sorts">${SORTS.map(([k, label]) => html`<button data-sort="${k}" class="${k === sort ? "on" : ""}">${label}</button>`)}</div>
87 </div>
88 ${products.length === 0
89 ? html`<div class="empty"><h3>No products found</h3><p>${q ? html`Nothing matches “${q}”. Try another word or ` : "This category is empty. "}<a href="#/c/all">see all products</a>.</p></div>`
90 : html`<div class="grid" data-trace="ProductForm.showCatalogue → GET /api/products · ProductController.search · CatalogService.search · ProductDAO.search · product ⋈ category + *_detail (chips)">${products.map((p) => this.card(p))}</div>`}
91 </section>
92 </div>`);
93 this.app.view.querySelectorAll("[data-sort]").forEach((b) => b.addEventListener("click", () => this.app.go(`${base}sort=${b.dataset.sort}`)));
94 } catch (err) {
95 if (this.app.current(nav)) mount(this.app.view, failure(err));
96 }
97 }

99 async showProduct(id) {
100 const nav = this.app.nav;
101 mount(this.app.view, loading());
102 let page;
103 try {
104 page = await api("GET", `${this.controller}/${id}`);
105 if (!this.app.current(nav)) return; // a newer navigation owns the page
106 } catch (err) {
107 return this.app.current(nav) && mount(this.app.view, failure(err));
108 }
109 const p = page.product;
110 const canBuy = p.stockState !== "OUT";
111 mount(this.app.view, html`
112 <p class="crumbs"><a href="#/">Shop</a> / ${page.parentSlug ? html`<a href="#/c/${page.parentSlug}">${page.parentCategory}</a> / ` : ""}<a href="#/c/${p.categorySlug}">${p.category}</a></p>
113 <section class="product">
114 <div class="pic">${p.image ? html`<img src="${p.image}" alt="${p.name}">` : ""}</div>
115 <div data-trace="ProductForm.showProduct → GET /api/products/${p.id} · ProductController.product · CatalogService.product · ProductDAO.findById · product ⋈ category + ${KIND_TABLE[p.detailKind] || "-"}">
116 <span class="eyebrow">${p.byline || p.category}</span>
117 <h1>${p.name}</h1>
118 ${stockBadge(p.stockState, p.stock)}
119 <div class="price-big">${vnd(p.price)}</div>
120 <p class="lead">${page.description}</p>
121 <div class="buy" data-trace="CartForm.submitAddItem → POST /api/cart/items · CartController.addItem · CartService.addItem · Cart.addItem (BR01, BR02) · CartDAO.save · cart_item">
122 <div class="stepper"><button type="button" data-step="-1" aria-label="Fewer">−</button><input id="qty" type="number" min="1" max="${p.stock}" value="1" aria-label="Quantity"><button type="button" data-step="1" aria-label="More">+</button></div>
123 <button class="btn accent" id="add" ${canBuy ? "" : "disabled"}>${canBuy ? "Add to cart" : "Out of stock"}</button>
124 </div>
125 <h3>${KIND_TITLE[p.detailKind] || "Details"}</h3>
126 <table class="specs" data-trace="ProductDetail.specs() · ${KIND_TABLE[p.detailKind] || "-"} (chosen by category.detail_kind)">
127 ${page.specs.map((s) => html`<tr><th>${s.label}</th><td>${s.value}</td></tr>`)}
128 <tr><th>Category</th><td>${page.parentCategory ? `${page.parentCategory} › ` : ""}${p.category}</td></tr>
129 <tr><th>SKU</th><td class="code">${p.sku}</td></tr>
130 </table>
131 </div>
132 </section>`);
133 const qty = document.getElementById("qty");
134 this.app.view.querySelectorAll("[data-step]").forEach((b) => b.addEventListener("click", () => {
135 qty.value = Math.max(1, Math.min(p.stock || 1, Number(qty.value) + Number(b.dataset.step)));
136 }));
137 document.getElementById("add").addEventListener("click", () => this.app.cartForm.submitAddItem(p.id, Number(qty.value)));
138 }
