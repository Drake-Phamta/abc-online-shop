1// Entry point: routes a URL hash to a view class and keeps the header in sync with the session.
2import { api } from "./api.js";
3import { html, mount, vnd, failure, toast } from "./ui.js";
4import { CustomerForm } from "./view/CustomerForm.js";
5import { ProductForm } from "./view/ProductForm.js";
6import { CartForm } from "./view/CartForm.js";
7import { OrderForm } from "./view/OrderForm.js";
8import { PaymentForm } from "./view/PaymentForm.js";
9import { StaffOrderForm } from "./view/StaffOrderForm.js";
10import { StockForm } from "./view/StockForm.js";
11import { SearchForm } from "./view/SearchForm.js";

13class App {
14 constructor() {
15 this.view = document.getElementById("view");
16 this.me = { loggedIn: false };
17 this.nav = 0; // number of the current navigation; a view drops a load that a newer navigation overtook
18 this.customerForm = new CustomerForm(this);
19 this.productForm = new ProductForm(this);
20 this.cartForm = new CartForm(this);
21 this.orderForm = new OrderForm(this);
22 this.paymentForm = new PaymentForm(this);
23 this.staffOrderForm = new StaffOrderForm(this);
24 this.stockForm = new StockForm(this);
25 this.searchForm = new SearchForm(this);
26 }
27
28 async start() {
29 document.getElementById("search").addEventListener("submit", (e) => {
30 e.preventDefault();
31 const q = new FormData(e.target).get("q").trim();
32 this.go(q ? `search?q=${encodeURIComponent(q)}` : "c/all");
33 });
34 document.getElementById("voice").addEventListener("click", () => this.searchForm.startVoice());
35 const photo = document.getElementById("photo");
36 document.getElementById("camera").addEventListener("click", () => photo.click());
37 photo.addEventListener("change", () => { this.searchForm.submitPhoto(photo.files[0]); photo.value = ""; });
38 const lens = document.getElementById("lens");
39 const setLens = (on) => { document.body.classList.toggle("lens", on); lens.checked = on; sessionStorage.setItem("lens", on ? "1" : ""); };
40 lens.addEventListener("change", () => setLens(lens.checked));
41 document.addEventListener("keydown", (e) => { if (e.altKey && e.key.toLowerCase() === "l") setLens(!lens.checked); });
42 try { setLens(sessionStorage.getItem("lens") === "1"); } catch { /* storage may be blocked */ }
43 window.addEventListener("hashchange", () => this.route());
44 await this.refreshMe();
45 this.route();
46 }
47
48 async refreshMe() {
49 try { this.me = await api("GET", "/api/auth/me"); } catch { this.me = { loggedIn: false }; }
50 this.renderHeader();
51 }
52
53 renderHeader() {
54 const me = this.me;
55 const account = document.getElementById("account");
56 if (!me.loggedIn) {
57 mount(account, html`<a href="#/login">Log in</a>`);
58 } else if (me.role === "CUSTOMER") {
59 mount(account, html`
60 <span class="who">Hello, <b>${me.name.split(" ").pop()}</b>${me.kind && me.kind !== "Customer" ? html` · ${me.kind}` : ""}</span>
61 <a href="#/orders">My orders</a>
62 <a class="cart-link" href="#/cart" data-trace="CartForm · GET /api/cart · CartController.viewCart · CartService.cart">Cart <span class="count">${me.cartCount || 0}</span></a>
63 <button class="link-btn" id="logout">Log out</button>`);
64 } else {
65 mount(account, html`<span class="who"><b>${me.name}</b> · ${me.role === "SALES" ? "Sales staff" : "Warehouse staff"}</span>
66 <button class="link-btn" id="logout">Log out</button>`);
67 }
68 const bar = document.getElementById("staff-bar");
69 mount(bar, me.loggedIn && me.role !== "CUSTOMER" ? html`<div class="staff-bar"><div class="wrap">
70 <b>Staff area</b>
71 ${me.role === "SALES" ? html`<a href="#/staff/orders">Order board (UC14)</a>` : html`<a href="#/staff/stock">Inventory (UC15)</a>`}
72 <a href="#/">Shop front</a></div></div>` : html``);
73 const out = document.getElementById("logout");
74 if (out) out.addEventListener("click", () => this.customerForm.submitLogout());
75 }
76
77 setCartCount(n) {
78 this.me.cartCount = n;
79 const el = document.querySelector(".cart-link .count");
80 if (el) el.textContent = n;
81 }
82
83 go(path) {
84 location.hash = "#/" + path;
85 }
86
87 /** Customer pages need a customer session; staff pages need the staff role (BR13). */
88 needs(role) {
89 if (!this.me.loggedIn) {
90 this.go(`login?next=${encodeURIComponent(location.hash.slice(2))}`);
91 return false;
92 }
93 if (this.me.role !== role) {
94 mount(this.view, failure(new Error("Your account cannot open this page.")));
95 return false;
96 }
97 return true;
98 }
99
100 current(nav) {
101 return nav === this.nav;
102 }
103
104 route() {
105 this.nav += 1;
106 const [path, query] = location.hash.replace(/^#\/?/, "").split("?");
107 const parts = path.split("/").filter(Boolean);
108 const params = new URLSearchParams(query || "");
109 document.querySelectorAll("#nav a").forEach((a) => a.classList.toggle("on", parts[0] === "c" && parts[1] === a.dataset.dept));
110 window.scrollTo({ top: 0 });
111 switch (parts[0]) {
112 case undefined: return this.productForm.showHome();
113 case "c": return this.productForm.showCatalogue(parts[1] || "all", params.get("sort"));
114 case "search": return parts[1] === "photo" ? this.searchForm.showPhotoResults() : this.searchForm.showResults(params.get("q"));
115 case "p": return this.productForm.showProduct(Number(parts[1]));
116 case "login": return this.customerForm.showLogin(params.get("next"));
117 case "cart": return this.needs("CUSTOMER") && this.cartForm.showCart();
118 case "checkout": return this.needs("CUSTOMER") && this.orderForm.showCheckout();
119 case "orders": return this.needs("CUSTOMER") && (parts[1] ? this.orderForm.showOrder(Number(parts[1])) : this.orderForm.showOrders());
120 case "pay": return this.needs("CUSTOMER") && this.paymentForm.showPayment(Number(parts[1]));
121 case "staff":
122 if (parts[1] === "stock") return this.needs("WAREHOUSE") && this.stockForm.showStock();
123 if (parts[1] === "orders" && parts[2]) return this.needs("SALES") && this.staffOrderForm.showOrder(Number(parts[2]));
124 return this.needs("SALES") && this.staffOrderForm.showBoard();
125 default: mount(this.view, failure(new Error("This page does not exist.")));
126 }
127 }
128}

130export const app = new App();
