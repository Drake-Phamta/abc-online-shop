1// View class StaffOrderForm (UC14 Process Orders). Controllers: OrderController and ShippingController, /api/staff/orders.
2import { api } from "../api.js";
3import { html, mount, vnd, when, statusBadge, loading, failure, toast } from "../ui.js";
4import { orderBody } from "./OrderForm.js";
5
6const COLUMNS = [
7 ["To process", ["PLACED", "PAID"]],
8 ["Processing", ["PROCESSING"]],
9 ["On the way", ["SHIPPED"]],
10 ["Closed", ["DELIVERED", "CANCELLED", "PENDING_PAYMENT"]],
11];
12const CARRIERS = ["GHN", "Viettel Post", "GHTK", "J&T Express"];
13
14export class StaffOrderForm {
15 constructor(app) {
16 this.app = app;
17 this.controller = "/api/staff/orders";
18 }
19
20 async showBoard() {
21 const nav = this.app.nav;
22 mount(this.app.view, loading());
23 let orders;
24 try { orders = await api("GET", this.controller); } catch (err) { return this.app.current(nav) && mount(this.app.view, failure(err)); }
25 if (!this.app.current(nav)) return; // a newer navigation owns the page
26 mount(this.app.view, html`
27 <div class="page-head"><div><p class="eyebrow">Sales staff · UC14</p><h1 class="two-tone">Order board. <span class="muted">Move each order along its lifecycle.</span></h1></div></div>
28 <div class="board" data-trace="StaffOrderForm.showBoard ← GET /api/staff/orders · OrderController.board · OrderService.allOrders · OrderDAO.findAll · customer_order ⋈ customer">
29 ${COLUMNS.map(([title, statuses]) => {
30 const list = orders.filter((o) => statuses.includes(o.status));
31 return html`<div class="column"><h4>${title}<span>${list.length}</span></h4>
32 ${list.map((o) => html`<div class="ticket">
33 <div class="row"><a class="code" href="#/staff/orders/${o.id}">${o.code}</a>${statusBadge(o.status)}</div>
34 <div style="margin:8px 0 2px"><b>${o.customerName}</b></div>
35 <div class="small">${o.firstItem}${o.itemCount > 1 ? ` + ${o.itemCount - 1} more` : ""}</div>
36 <div class="row" style="margin-top:6px"><span class="small">${when(o.orderDate)} · ${o.paymentMethod === "COD" ? "cash on delivery" : "card"}</span><span class="money">${vnd(o.totalAmount)}</span></div>
37 <div class="acts">${this.actions(o)}</div>
38 </div>`)}
39 </div>`;
40 })}
41 </div>`);
42 this.bind(() => this.showBoard());
43 }
44
45 /** One button per transition the server allows from the order's status (OrderStatus.next, BR09); no rule here. */
46 actions(o) {
47 const next = o.next || [];
48 const a = [];
49 if (next.includes("PROCESSING")) a.push(html`<button class="btn small" data-act="process" data-id="${o.id}">Start processing</button>`);
50 if (next.includes("SHIPPED")) a.push(html`<button class="btn small" data-act="ship" data-id="${o.id}" data-code="${o.code}">Ship…</button>`);
51 if (next.includes("DELIVERED")) a.push(html`<button class="btn small" data-act="deliver" data-id="${o.id}">Mark delivered</button>`);
52 if (next.includes("CANCELLED")) a.push(html`<button class="btn small ghost" data-act="cancel" data-id="${o.id}">Cancel</button>`);
53 return a;
54 }
55
56 async showOrder(id) {
57 const nav = this.app.nav;
58 mount(this.app.view, loading());
59 let o;
60 try { o = await api("GET", `/api/orders/${id}`); } catch (err) { return this.app.current(nav) && mount(this.app.view, failure(err)); }
61 if (!this.app.current(nav)) return; // a newer navigation owns the page
62 mount(this.app.view, html`
63 <p class="crumbs"><a href="#/staff/orders">Order board</a> / <span class="code">${o.code}</span></p>
64 <div class="page-head"><div><p class="eyebrow">${o.customerName} · placed ${when(o.orderDate)}</p><h1>Order <span class="code" style="font-size:.8em">${o.code}</span></h1></div>
65 <div class="acts" style="display:flex;gap:8px">${this.actions(o)}</div></div>
66 ${orderBody(o)}`);
67 this.bind(() => this.showOrder(id));
68 }
69
70 bind(refresh) {
71 this.app.view.querySelectorAll("[data-act]").forEach((b) => b.addEventListener("click", async () => {
72 const id = b.dataset.id;
73 try {
74 if (b.dataset.act === "ship") {
75 const data = await this.askShipping(b.dataset.code);
76 if (!data) return;
77 await api("POST", `${this.controller}/${id}/ship`, data);
78 toast("Order shipped, the customer can now track it");
79 } else if (b.dataset.act === "cancel") {
80 if (!confirm("Cancel this order? Stock is returned and an approved card payment is refunded.")) return;
81 await api("POST", `${this.controller}/${id}/cancel`);
82 toast("Order cancelled");
83 } else {
84 await api("POST", `${this.controller}/${id}/${b.dataset.act}`);
85 toast(b.dataset.act === "process" ? "Packing started" : "Delivery recorded");
86 }
87 refresh();
88 } catch (err) {
89 toast(err.message, "error");
90 }
91 }));
92 }
93
94 askShipping(code) {
95 return new Promise((resolve) => {
96 const dlg = document.createElement("dialog");
97 const tracking = "VN" + Math.floor(100000000 + Math.random() * 899999999);
98 dlg.innerHTML = html`<form method="dialog" data-trace="ShippingController.ship · ShippingService.ship · Order.ship (BR10) · ShippingDAO.save · shipping">
99 <h3 style="margin-bottom:6px">Ship <span class="code">${code}</span></h3>
100 <p class="small" style="margin-bottom:16px">The carrier and tracking number are shown to the customer (UC09).</p>
101 <div class="field"><label>Carrier</label><select name="carrier">${CARRIERS.map((c) => html`<option>${c}</option>`)}</select></div>
102 <div class="field"><label>Tracking number</label><input name="tracking" value="${tracking}" required></div>
103 <div style="display:flex;gap:10px;justify-content:flex-end"><button class="btn ghost" value="cancel">Back</button><button class="btn accent" value="ok">Hand over to carrier</button></div>
104 </form>`.__raw;
105 document.body.appendChild(dlg);
106 dlg.addEventListener("close", () => {
107 const f = dlg.querySelector("form");
108 resolve(dlg.returnValue === "ok" ? { carrier: f.carrier.value, trackingNumber: f.tracking.value } : null);
109 dlg.remove();
110 });
111 dlg.showModal();
112 });
113 }
