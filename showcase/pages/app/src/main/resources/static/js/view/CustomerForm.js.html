51 async submitLogin(email, password) {
52 const error = document.getElementById("login-error");
53 error.textContent = "";
54 try {
55 const me = await api("POST", `${this.controller}/login`, { email, password });
56 await this.app.refreshMe();
57 toast(`Logged in as ${me.name}`);
58 const start = me.role === "SALES" ? "staff/orders" : me.role === "WAREHOUSE" ? "staff/stock" : this.next || "";
59 this.app.go(start);
60 } catch (err) {
61 error.textContent = err.message;
62 error.style.color = "var(--danger)";
63 }
64 }
65
66 async submitLogout() {
67 await api("POST", `${this.controller}/logout`);
68 await this.app.refreshMe();
69 this.app.go("");
70 }
