1// View class StockForm (UC15 Manage Inventory). Controller: ProductController, /api/staff/stock.
2import { api } from "../api.js";
3import { html, mount, vnd, stockBadge, loading, failure, toast } from "../ui.js";
4
5export class StockForm {
6 constructor(app) {
7 this.app = app;
8 this.controller = "/api/staff/stock";
9 }
10
11 async showStock() {
12 const nav = this.app.nav;
13 mount(this.app.view, loading());
14 let rows;
15 try { rows = await api("GET", this.controller); } catch (err) { return this.app.current(nav) && mount(this.app.view, failure(err)); }
16 if (!this.app.current(nav)) return; // a newer navigation owns the page
17 const low = rows.filter((r) => r.stockState !== "IN").length;
18 mount(this.app.view, html`
19 <div class="page-head"><div><p class="eyebrow">Warehouse staff · UC15</p><h1 class="two-tone">Inventory. <span class="muted">${low} product${low === 1 ? "" : "s"} low or out of stock.</span></h1></div></div>
20 <table class="orders-table" data-trace="StockForm.showStock ← GET /api/staff/stock · ProductController.stock · CatalogService.stockList · ProductDAO.listForStock · product">
21 <thead><tr><th></th><th>Product</th><th>Category</th><th>State</th><th style="text-align:right">Stock</th><th></th></tr></thead>
22 <tbody>${rows.map((r) => html`<tr class="${r.stockState === "OUT" ? "out" : r.stockState === "LOW" ? "low" : ""}">
23 <td style="width:56px">${r.image ? html`<img src="${r.image}" alt="" style="width:44px;height:44px;object-fit:contain;mix-blend-mode:multiply">` : ""}</td>
24 <td><b>${r.name}</b><br><span class="code small">${r.sku}</span></td>
25 <td>${r.category}</td><td>${stockBadge(r.stockState, r.stock)}</td>
26 <td style="text-align:right"><input class="stock-input" type="number" min="0" value="${r.stock}" data-id="${r.id}" aria-label="Stock of ${r.name}"></td>
27 <td><button class="btn small ghost" data-save="${r.id}" data-trace="StockForm.submitStock → PUT /api/staff/stock/{id} · ProductController.updateStock · CatalogService.updateStock · Product.updateStock (BR12) · ProductDAO.updateStock">Save</button></td>
28 </tr>`)}</tbody>
29 </table>`);
30 this.app.view.querySelectorAll("[data-save]").forEach((b) => b.addEventListener("click", () => {
31 const input = this.app.view.querySelector(`.stock-input[data-id="${b.dataset.save}"]`);
32 this.submitStock(Number(b.dataset.save), input.value === "" ? null : Number(input.value));
33 }));
34 }
35
36 async submitStock(productId, quantity) {
37 try {
38 const row = await api("PUT", `${this.controller}/${productId}`, { quantity });
39 toast(`${row.name}: stock set to ${row.stock}`);
40 this.showStock();
41 } catch (err) {
42 toast(err.message, "error");
43 }
44 }
45}
