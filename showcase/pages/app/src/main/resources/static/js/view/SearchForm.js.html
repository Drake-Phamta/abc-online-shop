17 /** The understood query as chips (AI02). */
18 understood(r) {
19 const u = r.understood;
20 const chips = [
21 ...u.terms.map((t) => html`<span class="chip">${t}</span>`),
22 ...u.related.map((t) => html`<span class="chip soft" title="related words of the shop's lexicon">${t}</span>`),
23 ...u.corrections.map((t) => html`<span class="chip fix" title="spelling corrected to a catalogue word">${t}</span>`),
24 ];
25 if (u.maxPrice !== null && u.minPrice !== null) chips.push(html`<span class="chip price">${vnd(u.minPrice)} to ${vnd(u.maxPrice)}</span>`);
26 else if (u.maxPrice !== null) chips.push(html`<span class="chip price">at most ${vnd(u.maxPrice)}</span>`);
27 else if (u.minPrice !== null) chips.push(html`<span class="chip price">at least ${vnd(u.minPrice)}</span>`);
28 if (u.cheap) chips.push(html`<span class="chip price">cheaper first</span>`);
29 return html`<div class="understood"><span class="label">Understood as</span>${chips}</div>`;
30 }

32 card(f, max) {
33 const p = f.product;
34 const share = max > 0 ? Math.max(8, Math.round((f.score / max) * 100)) : 0;
35 return html`<a class="card found" href="#/p/${p.id}">
36 <div class="pic">${p.image ? html`<img src="${p.image}" alt="${p.name}" loading="lazy">` : ""}</div>
37 <span class="brand-line">${p.byline || p.category}</span>
38 <span class="name">${p.name}</span>
39 <div class="foot"><span class="money">${vnd(p.price)}</span>${p.stockState !== "IN" ? stockBadge(p.stockState, p.stock) : ""}</div>
40 <div class="why">
41 ${max > 0 ? html`<div class="meter" title="relevance ${f.score}"><i style="width:${share}%"></i></div>` : ""}
42 <ul>${f.reasons.map((x) => html`<li>${x}</li>`)}</ul>
43 </div>
44 </a>`;
45 }

47 results(r) {
48 const max = r.results.reduce((m, f) => Math.max(m, f.score), 0);
49 return html`<div class="grid" data-trace="${r.mode === "PHOTO" ? TRACE_PHOTO : TRACE_TEXT}">${r.results.map((f) => this.card(f, max))}</div>`;
50 }

52 /** FR-S1 to FR-S3 and FR-S6: typed or spoken text. */
53 async showResults(q) {
54 const nav = this.app.nav;
55 if (!q) return this.app.go("c/all");
56 mount(this.app.view, loading());
57 let r;
58 try {
59 r = await api("GET", `${this.controller}?q=${encodeURIComponent(q)}`);
60 } catch (err) {
61 return this.app.current(nav) && mount(this.app.view, failure(err));
62 }
63 if (!this.app.current(nav)) return; // a newer navigation owns the page
64 const how = r.fallback ? "Keyword search" : "AI search";
65 mount(this.app.view, html`
66 <div class="page-head"><div>
67 <p class="eyebrow">${how} · ${r.results.length} products · ${r.tookMs} ms</p>
68 <h1>Results for “${q}”</h1>
69 </div></div>
70 ${this.understood(r)}
71 ${r.fallback && r.results.length ? html`<p class="note">The AI search found nothing close enough, so these are plain keyword matches (AI04).</p>` : ""}
72 ${r.results.length === 0
73 ? html`<div class="empty"><h3>No products found</h3><p>Neither the AI search nor the keyword search found anything for “${q}”.
74 Try other words, a photo, or <a href="#/c/all">see all products</a>.</p></div>`
75 : this.results(r)}`);
76 }

78 /** FR-S4: speech to text in the browser; the customer can correct the text before it is searched. */
79 startVoice() {
80 const Recognition = window.SpeechRecognition || window.webkitSpeechRecognition;
81 const box = document.createElement("div");
82 box.className = "voice";
83 box.setAttribute("role", "dialog");
84 box.setAttribute("aria-label", "Search by voice");
85 mount(box, html`<form class="voice-card">
86 <div class="voice-head"><span class="dot ${Recognition ? "on" : ""}"></span><b id="voice-state">${Recognition ? "Listening…" : "Voice input is not available in this browser"}</b>
87 <select id="voice-lang" aria-label="Language"><option value="en-US">English</option><option value="vi-VN">Tiếng Việt</option></select></div>
88 <p class="small">${Recognition ? "Say what you are looking for, for example “red sneakers under three million”. You can correct the text before searching."
89 : "Type your search instead; the AI search treats it the same way."}</p>
90 <input id="voice-text" name="q" autocomplete="off" placeholder="Your words appear here" aria-label="Spoken text">
91 <div class="voice-actions"><button type="button" class="btn ghost small" id="voice-cancel">Cancel</button>
92 <button class="btn accent small" id="voice-go">Search</button></div>
93 </form>`);
94 document.body.appendChild(box);
95 const text = box.querySelector("#voice-text");
96 let rec = null;
97 const close = () => { if (rec) rec.abort(); box.remove(); };
98 const listen = () => {
99 if (!Recognition) { text.focus(); return; }
100 if (rec) rec.abort();
101 rec = new Recognition();
102 rec.lang = box.querySelector("#voice-lang").value;
103 rec.interimResults = true;
104 rec.onresult = (e) => { text.value = Array.from(e.results).map((x) => x[0].transcript).join(" "); };
105 rec.onerror = (e) => { box.querySelector("#voice-state").textContent = e.error === "not-allowed" ? "Microphone blocked: type instead" : "Could not hear you: type or try again"; box.querySelector(".dot").classList.remove("on"); };
106 rec.onend = () => { box.querySelector(".dot").classList.remove("on"); if (box.querySelector("#voice-state").textContent === "Listening…") box.querySelector("#voice-state").textContent = "Check the text, then search"; };
107 box.querySelector(".dot").classList.add("on");
108 box.querySelector("#voice-state").textContent = "Listening…";
109 try {
110 rec.start();
111 } catch { // no microphone or speech service: the text box still works
112 box.querySelector("#voice-state").textContent = "Voice input is not available here: type instead";
113 box.querySelector(".dot").classList.remove("on");
114 }
115 };
116 box.querySelector("#voice-lang").addEventListener("change", listen);
117 box.querySelector("#voice-cancel").addEventListener("click", close);
118 box.addEventListener("click", (e) => { if (e.target === box) close(); });
119 box.querySelector("form").addEventListener("submit", (e) => {
120 e.preventDefault();
121 const q = text.value.trim();
122 if (!q) { toast("Say or type what you are looking for.", "error"); return; }
123 close();
124 this.app.go(`search?q=${encodeURIComponent(q)}`);
125 });
126 listen();
127 }

129 /** FR-S5: the photo goes to the server once, is compared and dropped there (AI05). */
130 async submitPhoto(file) {
131 if (!file) return;
132 if (!/^image\/(jpeg|png)$/.test(file.type)) { toast("Choose a JPEG or PNG photo.", "error"); return; }
133 if (file.size > 5 * 1024 * 1024) { toast("The photo is larger than 5 MB.", "error"); return; }
134 mount(this.app.view, loading());
135 const form = new FormData();
136 form.append("photo", file);
137 try {
138 const res = await fetch(`${this.controller}/photo`, { method: "POST", body: form, credentials: "same-origin" });
139 const r = await res.json();
140 if (!res.ok) throw new ApiError(res.status, r);
141 if (this.lastPhoto) URL.revokeObjectURL(this.lastPhoto.url);
142 this.lastPhoto = { url: URL.createObjectURL(file), name: file.name, result: r };
143 if (location.hash === "#/search/photo") this.showPhotoResults(); else this.app.go("search/photo");
144 } catch (err) {
145 mount(this.app.view, failure(err));
146 }
147 }

165 : this.results(r)}`);
166 }
