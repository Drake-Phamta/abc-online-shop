1// View class CartForm (UC06 Manage Shopping Cart). Controller: CartController, /api/cart.
2import { api } from "../api.js";
3import { html, mount, vnd, loading, failure, toast } from "../ui.js";
4
5export class CartForm {
6 constructor(app) {
7 this.app = app;
8 this.controller = "/api/cart";
9 }
10
11 async submitAddItem(productId, quantity) {
12 if (!this.app.me.loggedIn) {
13 toast("Log in to add products to your cart.");
14 return this.app.go(`login?next=${encodeURIComponent(`p/${productId}`)}`);
15 }
16 if (this.app.me.role !== "CUSTOMER") return toast("Staff accounts cannot buy.", "error");
17 try {
18 const cart = await api("POST", `${this.controller}/items`, { productId, quantity });
19 this.app.setCartCount(cart.count);
20 toast(`Added to your cart · ${cart.count} item${cart.count === 1 ? "" : "s"}`);
21 } catch (err) {
22 toast(err.message, "error");
23 }
24 }
25
26 async showCart(cart) {
27 const nav = this.app.nav;
28 if (!cart) {
29 mount(this.app.view, loading());
30 try { cart = await api("GET", this.controller); } catch (err) { return this.app.current(nav) && mount(this.app.view, failure(err)); }
31 if (!this.app.current(nav)) return; // a newer navigation owns the page
32 }
33 this.app.setCartCount(cart.count);
34 if (cart.lines.length === 0) {
35 return mount(this.app.view, html`<div class="empty"><h2>Your cart is empty</h2>
36 <p class="lead" style="margin:10px auto 20px">Find something you like in the catalogue.</p>
37 <a class="btn" href="#/c/all">Browse products</a></div>`);
38 }
39 mount(this.app.view, html`
40 <div class="page-head"><div><p class="eyebrow">Shopping cart</p><h1>${cart.count} item${cart.count === 1 ? "" : "s"} in your cart</h1></div></div>
41 <div class="two-col">
42 <section class="lines" data-trace="CartForm.showCart ← GET /api/cart · CartController.viewCart · CartService.cart · CartDAO.findOrCreate · cart, cart_item ⋈ product">
43 ${cart.lines.map((l) => html`
44 <div class="line">
45 <a class="thumb" href="#/p/${l.productId}">${l.image ? html`<img src="${l.image}" alt="">` : ""}</a>
46 <div class="meta">
47 <a href="#/p/${l.productId}">${l.name}</a>
48 <span class="small">${vnd(l.cartPrice)} each${l.cartPrice !== l.currentPrice ? html` · <b style="color:var(--warn)">now ${vnd(l.currentPrice)}, updated at checkout</b>` : ""}</span>
49 <div class="stepper" data-product="${l.productId}">
50 <button type="button" data-step="-1" aria-label="Fewer">−</button><input type="number" min="1" max="${l.stock}" value="${l.quantity}" aria-label="Quantity"><button type="button" data-step="1" aria-label="More">+</button>
51 </div>
52 </div>
53 <div class="right"><span class="money">${vnd(l.subTotal)}</span><button class="link-btn" data-remove="${l.productId}">Remove</button></div>
54 </div>`)}
55 </section>
56 <aside class="panel sticky">
57 <h3>Summary</h3>
58 <div class="sum-row"><span>Subtotal</span><b>${vnd(cart.subtotal)}</b></div>
59 <p class="small">VIP discount and delivery fee are calculated at checkout. Delivery is free from 500.000 ₫.</p>
60 <a class="btn accent wide" href="#/checkout">Go to checkout</a>
61 </aside>
62 </div>`);
63 this.app.view.querySelectorAll(".stepper[data-product]").forEach((s) => {
64 const id = Number(s.dataset.product);
65 const input = s.querySelector("input");
66 s.querySelectorAll("[data-step]").forEach((b) => b.addEventListener("click", () => this.changeQuantity(id, Number(input.value) + Number(b.dataset.step))));
67 input.addEventListener("change", () => this.changeQuantity(id, Number(input.value)));
68 });
69 this.app.view.querySelectorAll("[data-remove]").forEach((b) => b.addEventListener("click", () => this.removeItem(Number(b.dataset.remove))));
70 }
71
72 async changeQuantity(productId, quantity) {
73 if (quantity < 1) return this.removeItem(productId);
74 try {
75 this.showCart(await api("PUT", `${this.controller}/items/${productId}`, { quantity }));
76 } catch (err) {
77 toast(err.message, "error");
78 this.showCart();
79 }
80 }
81
82 async removeItem(productId) {
83 try {
84 this.showCart(await api("DELETE", `${this.controller}/items/${productId}`));
85 toast("Removed from your cart");
86 } catch (err) {
87 toast(err.message, "error");
88 }
89 }
90}
