12export function statusSteps(order) {
13 const flow = FLOW[order.paymentMethod];
14 if (order.status === "CANCELLED") return html`<div class="notice danger"><b>This order was cancelled.</b>Reserved stock was returned${order.payments.some((p) => p.status === "REFUNDED") ? " and the card payment was refunded" : ""}.</div>`;
15 const at = flow.indexOf(order.status);
16 return html`<div class="steps" style="grid-template-columns:repeat(${flow.length},1fr)" data-trace="OrderStatus transition table (BR09) · order_event">
17 ${flow.map((s, i) => html`<div class="step ${i < at ? "done" : ""} ${i === at ? "done now" : ""}">${STATUS[s].label}</div>`)}</div>`;
18}

105 async submitOrder(addressId, paymentMethod, expectedTotal) {
106 const btn = document.getElementById("place");
107 btn.disabled = true;
108 btn.textContent = "Placing order…";
109 try {
110 const res = await api("POST", this.controller, { addressId, paymentMethod, expectedTotal });
111 this.app.setCartCount(0);
112 if (res.status === "PENDING_PAYMENT") {
113 toast(`Order ${res.code} created. Complete the card payment.`);
114 this.app.go(`pay/${res.orderId}`);
115 } else {
116 toast(`Order ${res.code} placed. Pay the courier on delivery.`);
117 this.app.go(`orders/${res.orderId}`);
118 }
119 } catch (err) {
120 toast(err.message, "error");
121 this.showCheckout();
122 }
123 }

142 async showOrder(id) {

153 <div style="display:flex;gap:10px">

155 ${canCancel ? html`<button class="btn ghost" id="cancel">Cancel order</button>` : ""}

158 ${orderBody(o)}`);
159 const c = document.getElementById("cancel");
160 if (c) c.addEventListener("click", () => this.submitCancel(o));
