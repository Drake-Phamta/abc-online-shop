2"""Builds db/schema.sql, the schema of the running shop, from the DDL that Visual Paradigm generated from the data model.
3
4Input
5 db/vp/vp_ddl.sql written by VP (Tools > DB > Generate Database, SQLite) from the ERD that Synchronize to Entity
6 Relationship Diagram generated from the class diagram of the Data Model file
7 RULES below the constraints VP cannot write because the data model holds no such rule: UNIQUE, CHECK and
8 ON DELETE CASCADE (business rules of Chapter 1)
9Output
10 db/schema.sql the same tables, columns, types, NOT NULL, primary keys and foreign keys as the VP DDL, one column
11 per line, with the hand-written constraints added and marked with a comment
12
13Run: python db/make_schema.py (db/compare_schema.py then checks the result against the VP DDL)
14"""

58def split_top(s):
59 """Split the body of a CREATE TABLE at the commas that are not inside parentheses."""
60 parts, depth, cur = [], 0, ""
61 for ch in s:
62 if ch == "(":
63 depth += 1
64 elif ch == ")":
65 depth -= 1
66 if ch == "," and depth == 0:
67 parts.append(cur.strip())
68 cur = ""
69 else:
70 cur += ch
71 if cur.strip():
72 parts.append(cur.strip())
73 return parts

76def main():
77 ddl = VP.read_text(encoding="utf-8")
78 out = ["-- ABC Online Shop: the schema of the running shop.",
79 "-- Tables, columns, types, NOT NULL, primary keys and foreign keys are copied from db/vp/vp_ddl.sql, the DDL that",
80 "-- Visual Paradigm generated (Tools > DB > Generate Database, SQLite) from the data model of vp/2_DataModel.vpp.",
81 "-- Added by hand (marked \"-- rule\"): UNIQUE, CHECK and ON DELETE CASCADE, which the data model cannot express.",
82 "-- Built by db/make_schema.py. Money is integer VND. Run with PRAGMA foreign_keys = ON.", ""]
83 n = 0
84 for m in re.finditer(r"CREATE TABLE (\w+) \((.*?)\);", ddl, re.S):
85 table, body = m.group(1), m.group(2)
86 lines = []
87 for part in split_top(body):
88 fk = re.match(r"FOREIGN KEY\((\w+)\) REFERENCES (\w+)\((\w+)\)", part)
89 if fk:
90 col = fk.group(1)
91 text = f"FOREIGN KEY ({col}) REFERENCES {fk.group(2)} ({fk.group(3)})"
92 if (table, col) in CASCADE:
93 text += " ON DELETE CASCADE -- rule"
94 lines.append(text)
95 continue
96 if part.startswith("PRIMARY KEY"):
97 lines.append(part)
98 continue
99 col = part.split()[0]
100 rule = CHECKS.get(table, {}).get(col)
101 lines.append(part + (f" {rule} -- rule" if rule else ""))
102 if table in UNIQUES:
103 lines.append(UNIQUES[table] + " -- rule")
104 # a trailing "-- comment" must not swallow the comma: put the comma before the comment
105 rendered = []
106 for i, l in enumerate(lines):
107 comma = "," if i < len(lines) - 1 else ""
108 if " -- " in l:
109 code, note = l.split(" -- ", 1)
110 rendered.append(f" {code}{comma} -- {note}")
111 else:
112 rendered.append(f" {l}{comma}")
113 out.append(f"CREATE TABLE {table} (")
114 out += rendered
115 out.append(");")
116 out.append("")
117 n += 1
118 OUT.write_text("\n".join(out), encoding="utf-8")
119 used = sum(len(v) for v in CHECKS.values()) + len(UNIQUES) + len(CASCADE)
120 print(f"{OUT.name}: {n} tables from the VP DDL, {used} hand-written constraints")
