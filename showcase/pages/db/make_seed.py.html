48def sql(v):
49 if v is None:
50 return "NULL"
51 if isinstance(v, (int, float, Decimal)):
52 return str(v)
53 return "'" + str(v).replace("'", "''") + "'"

56def row(table, cols, values):
57 return f"INSERT INTO {table} ({', '.join(cols)}) VALUES ({', '.join(sql(v) for v in values)});"

60def pbkdf2(password, label):
61 salt = hashlib.sha256(("abc-shop-seed-" + label).encode()).digest()[:16]
62 digest = hashlib.pbkdf2_hmac("sha256", password.encode(), salt, 120_000, 32)
63 return f"pbkdf2$120000${base64.b64encode(salt).decode()}${base64.b64encode(digest).decode()}"

66def totals(lines, rate):
67 subtotal = sum(q * p for _, _, q, p in lines)
68 discount = int((Decimal(subtotal) * Decimal(rate)).quantize(Decimal(1), ROUND_HALF_UP))
69 fee = 0 if subtotal - discount >= FREE_FROM else SHIPPING_FEE
70 return subtotal, discount, fee, subtotal - discount + fee
