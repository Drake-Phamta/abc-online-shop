50def sql(v):
51 if v is None:
52 return "NULL"
53 if isinstance(v, (int, float, Decimal)):
54 return str(v)
55 return "'" + str(v).replace("'", "''") + "'"

58def pascal(key):
59 """ram_gb -> RamGb: the column of db/schema.sql for a key of products.json."""
60 return "".join(part[:1].upper() + part[1:] for part in key.split("_"))

63def row(table, cols, values):
64 return f"INSERT INTO {table} ({', '.join(cols)}) VALUES ({', '.join(sql(v) for v in values)});"

67def pbkdf2(password, label):
68 salt = hashlib.sha256(("abc-shop-seed-" + label).encode()).digest()[:16]
69 digest = hashlib.pbkdf2_hmac("sha256", password.encode(), salt, 120_000, 32)
70 return f"pbkdf2$120000${base64.b64encode(salt).decode()}${base64.b64encode(digest).decode()}"

73def totals(lines, rate):
74 subtotal = sum(q * p for _, _, q, p in lines)
75 discount = int((Decimal(subtotal) * Decimal(rate)).quantize(Decimal(1), ROUND_HALF_UP))
76 fee = 0 if subtotal - discount >= FREE_FROM else SHIPPING_FEE
77 return subtotal, discount, fee, subtotal - discount + fee
