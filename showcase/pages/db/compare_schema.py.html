1# -*- coding: utf-8 -*-
2"""Checks that the shop's schema (db/schema.sql) is the database Visual Paradigm generated from the data model.
3
4db/vp/vp_ddl.sql is written by VP (Tools > DB > Generate Database, SQLite) from the ERD that Synchronize to Entity
5Relationship Diagram generated from the class diagram of vp/2_DataModel.vpp. Both scripts are run in an in-memory
6SQLite database and compared table by table:
7 same tables; same columns with the same declared type, NOT NULL and primary key; same foreign keys.
8The constraints that exist only in schema.sql (UNIQUE, CHECK, ON DELETE CASCADE) are listed: they are the business
9rules added by hand. Finally the seed data (db/seed.sql) is loaded into both, which proves the generated tables
10hold the shop's data. Exit code = number of differences.
11"""
12import re
13import sqlite3
14import sys
15from pathlib import Path
16
17ROOT = Path(__file__).resolve().parent
18VP_DDL = ROOT / "vp" / "vp_ddl.sql"
19SCHEMA = ROOT / "schema.sql"
20SEED = ROOT / "seed.sql"
21
22
23def load(script):
24 con = sqlite3.connect(":memory:")
25 con.execute("PRAGMA foreign_keys = ON")
26 con.executescript(script)
27 return con
28
29
30def describe(con):
31 out = {}
32 for (t,) in con.execute("SELECT name FROM sqlite_master WHERE type = 'table' AND name NOT LIKE 'sqlite_%'"):
33 cols = [(r[1], r[2].lower(), bool(r[3]), r[5] > 0) for r in con.execute(f"PRAGMA table_info({t})")]
34 fks = sorted((r[3], r[2], r[4]) for r in con.execute(f"PRAGMA foreign_key_list({t})"))
35 out[t] = (sorted(cols), fks)
36 return out
37
38
39def main():
40 vp_sql, app_sql = VP_DDL.read_text(encoding="utf-8"), SCHEMA.read_text(encoding="utf-8")
41 vp, app = describe(load(vp_sql)), describe(load(app_sql))
42 problems = 0
43 if sorted(vp) != sorted(app):
44 problems += 1
45 print("DIFFERENT TABLES: only VP", sorted(set(vp) - set(app)), "only schema.sql", sorted(set(app) - set(vp)))
46 for t in sorted(set(vp) & set(app)):
47 if vp[t][0] != app[t][0]:
48 problems += 1
49 print(f"{t}: columns differ: {sorted(set(vp[t][0]) ^ set(app[t][0]))}")
50 if vp[t][1] != app[t][1]:
51 problems += 1
52 print(f"{t}: foreign keys differ: VP {vp[t][1]} schema.sql {app[t][1]}")
53 rules = [l.split(" -- ")[0].strip().rstrip(",") for l in app_sql.splitlines() if l.rstrip().endswith("-- rule")]
54 n_cols = sum(len(v[0]) for v in vp.values())
55 n_fk = sum(len(v[1]) for v in vp.values())
56 print(f"generated by Visual Paradigm: {len(vp)} tables, {n_cols} columns, {n_fk} foreign keys")
57 print(f"schema.sql: same tables, columns, types, NOT NULL, primary and foreign keys: {problems == 0}")
58 print(f"added by hand in schema.sql: {sum(len(re.findall(r'CHECK \(', r)) for r in rules)} CHECK, "
59 f"{sum(r.count('UNIQUE') for r in rules)} UNIQUE, {sum(r.count('ON DELETE CASCADE') for r in rules)} ON DELETE CASCADE")
60 seed = SEED.read_text(encoding="utf-8")
61 for name, script in (("VP DDL", vp_sql), ("schema.sql", app_sql)):
62 con = load(script)
63 con.executescript(seed)
64 rows = sum(con.execute(f"SELECT COUNT(*) FROM {t}").fetchone()[0] for t in vp)
65 bad = con.execute("PRAGMA foreign_key_check").fetchall()
66 print(f"seed data in the {name} database: {rows} rows, foreign key violations: {len(bad)}")
67 problems += len(bad)
68 sys.exit(problems)
69
