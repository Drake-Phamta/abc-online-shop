2"""Checks that the DDL exported from the Visual Paradigm physical ERD (db/vp/vp_ddl.sql) defines exactly the database
3of db/schema.sql: same statements after removing comments, and the same structure when both are run in SQLite
4(columns, types, NOT NULL, defaults, primary keys, foreign keys with ON DELETE, unique indexes, CHECK clauses).
5
6Usage: python db/compare_schema.py [vp_ddl.sql] [schema.sql] exit code 0 = identical
7"""
8import re
9import sqlite3
10import sys
11from pathlib import Path
12
13HERE = Path(__file__).resolve().parent
14VP = Path(sys.argv[1]) if len(sys.argv) > 1 else HERE / "vp" / "vp_ddl.sql"
15REF = Path(sys.argv[2]) if len(sys.argv) > 2 else HERE / "schema.sql"

18def statements(path):
19 text = "\n".join(l for l in path.read_text(encoding="utf-8").splitlines() if not l.strip().startswith("--"))
20 return [re.sub(r"\s+", " ", s).strip() for s in text.split(";") if s.strip()]
21
22
23def structure(path):
24 con = sqlite3.connect(":memory:")
25 con.executescript(path.read_text(encoding="utf-8"))
26 out = {}
27 for (name, sql) in con.execute("SELECT name, sql FROM sqlite_master WHERE type = 'table' ORDER BY name"):
28 checks = sorted(re.findall(r"CHECK \((.*?)\)(?:,|\n|$)", sql))
29 out[name] = {
30 "columns": con.execute(f"PRAGMA table_info({name})").fetchall(),
31 "foreign_keys": sorted(con.execute(f"PRAGMA foreign_key_list({name})").fetchall(), key=lambda r: r[3]),
32 "unique": sorted(tuple(c[2] for c in con.execute(f"PRAGMA index_info({ix[1]})"))
33 for ix in con.execute(f"PRAGMA index_list({name})") if ix[2]),
34 "checks": checks,
35 }
36 return out

39a, b = statements(VP), statements(REF)
40print(f"{VP.name}: {len(a)} statements, {REF.name}: {len(b)} statements")
41text_same = a == b
42if not text_same:
43 for i, (x, y) in enumerate(zip(a, b)):
44 if x != y:
45 print(f"first textual difference in statement {i + 1}:\n VP : {x[:300]}\n REF: {y[:300]}")
46 break
47sa, sb = structure(VP), structure(REF)
48diffs = [t for t in sorted(set(sa) | set(sb)) if sa.get(t) != sb.get(t)]
49for t in diffs:
50 print(f"structure differs in table {t}")
51 for k in ("columns", "foreign_keys", "unique", "checks"):
52 if sa.get(t, {}).get(k) != sb.get(t, {}).get(k):
53 print(f" {k}:\n VP : {sa.get(t, {}).get(k)}\n REF: {sb.get(t, {}).get(k)}")
54fk = sum(len(v["foreign_keys"]) for v in sa.values())
55print(f"tables {len(sa)}, foreign keys {fk}, textually identical: {text_same}, structurally identical: {not diffs}")

57# The database that Visual Paradigm itself generated (Tools > DB > Generate Database, SQLite, db/vp/abc_shop_vp.db,
58# DDL in db/vp/vp_generated.ddl): same tables and columns as schema.sql? VP's SQLite generator writes no foreign keys
59# (SQLite accepts them only inside CREATE TABLE), so they are counted, not compared.
60GEN = HERE / "vp" / "abc_shop_vp.db"
61gen_ok = True
62if GEN.exists():
63 con = sqlite3.connect(f"file:{GEN.as_posix()}?mode=ro", uri=True)

67 def cols(c, t): # name, declared type, not null, primary key; INTEGER(10) and INTEGER are the same type in SQLite
68 return [(r[1], re.sub(r"\(.*", "", r[2]).upper() if r[2].upper().startswith("INTEGER") else r[2].upper().replace(" ", ""),
69 r[3] or r[5] > 0, r[5] > 0) for r in c.execute(f"PRAGMA table_info('{t}')")]
70
71 def checks(c, t):
72 sql = c.execute("SELECT sql FROM sqlite_master WHERE name = ?", (t,)).fetchone()[0]
73 return sorted(re.sub(r"\s+", "", x).lower() for x in re.findall(r"CHECK\s*\((.*?)\)\s*[,)]", sql, re.S))
74
75 def uniques(c, t): # without the index SQLite adds for a PRIMARY KEY clause (origin "pk"): the key is compared above
76 return sorted(tuple(x[2] for x in c.execute(f"PRAGMA index_info('{ix[1]}')")) for ix in c.execute(f"PRAGMA index_list('{t}')")
77 if ix[2] and ix[3] != "pk")
78
79 tables = [r[0] for r in con.execute("SELECT name FROM sqlite_master WHERE type = 'table' AND name NOT LIKE 'sqlite_%' ORDER BY name")]
80 ref_tables = [r[0] for r in ref.execute("SELECT name FROM sqlite_master WHERE type = 'table' AND name NOT LIKE 'sqlite_%' ORDER BY name")]
81 gen_ok = tables == ref_tables
82 n_cols = n_chk = n_uq = gen_fk = 0
83 for t in tables:
84 for kind, f in (("columns", cols), ("checks", checks), ("unique keys", uniques)):
85 x, y = f(con, t), f(ref, t)
86 if x != y:
87 gen_ok = False
88 print(f"VP database differs in {t} {kind}:\n VP : {x}\n REF: {y}")
89 n_cols += len(cols(con, t)); n_chk += len(checks(con, t)); n_uq += len(uniques(con, t))
90 gen_fk += len(con.execute(f"PRAGMA foreign_key_list('{t}')").fetchall())
91 print(f"database generated by Visual Paradigm: tables {len(tables)}, columns {n_cols}, CHECK {n_chk}, unique keys {n_uq}, "
92 f"foreign keys {gen_fk}; same tables, columns, types, NOT NULL, keys, CHECK and UNIQUE as schema.sql: {gen_ok}")
93sys.exit(0 if text_same and not diffs and gen_ok else 1)
