142CREATE TABLE Orders (
143 OrderId INTEGER NOT NULL PRIMARY KEY AUTOINCREMENT,
144 AddressId integer(10) NOT NULL,
145 CustomerId integer(10) NOT NULL,
146 OrderDate date NOT NULL,
147 Status varchar(255) NOT NULL CHECK (Status IN ('PENDING_PAYMENT', 'PLACED', 'PAID', 'PROCESSING', 'SHIPPED', 'DELIVERED', 'CANCELLED')), -- rule
148 PaymentMethod varchar(255) NOT NULL CHECK (PaymentMethod IN ('CARD', 'COD')), -- rule
149 Subtotal integer(10) NOT NULL CHECK (Subtotal > 0), -- rule
150 Discount integer(10) NOT NULL CHECK (Discount >= 0), -- rule
151 ShippingFee integer(10) NOT NULL CHECK (ShippingFee >= 0), -- rule
152 TotalAmount integer(10) NOT NULL CHECK (TotalAmount = Subtotal - Discount + ShippingFee), -- rule
153 FOREIGN KEY (CustomerId) REFERENCES Customer (CustomerId),
154 FOREIGN KEY (AddressId) REFERENCES Address (AddressId)
