1CREATE TABLE Category (CategoryId INTEGER NOT NULL PRIMARY KEY AUTOINCREMENT, ParentId integer(10), Name varchar(255) NOT NULL, Slug varchar(255) NOT NULL, DetailKind varchar(255), FOREIGN KEY(ParentId) REFERENCES Category(CategoryId));
2CREATE TABLE Product (ProductId INTEGER NOT NULL PRIMARY KEY AUTOINCREMENT, CategoryId integer(10) NOT NULL, Sku varchar(255) NOT NULL, Name varchar(255) NOT NULL, Brand varchar(255), Price integer(10) NOT NULL, StockQuantity integer(10) NOT NULL, Description varchar(255), ImageFile varchar(255), Status varchar(255) NOT NULL, CreatedAt date NOT NULL, FOREIGN KEY(CategoryId) REFERENCES Category(CategoryId));
3CREATE TABLE BookDetail (Author varchar(255) NOT NULL, Publisher varchar(255), Pages integer(10), Language varchar(255), ProductId integer(10) NOT NULL, PRIMARY KEY (ProductId), FOREIGN KEY(ProductId) REFERENCES Product(ProductId));
4CREATE TABLE LaptopDetail (Cpu varchar(255) NOT NULL, RamGb integer(10) NOT NULL, StorageGb integer(10) NOT NULL, ScreenInch double(10) NOT NULL, WarrantyMonths integer(10) NOT NULL, ProductId integer(10) NOT NULL, PRIMARY KEY (ProductId), FOREIGN KEY(ProductId) REFERENCES Product(ProductId));
5CREATE TABLE PhoneDetail (Os varchar(255) NOT NULL, StorageGb integer(10) NOT NULL, CameraMp integer(10), ScreenInch double(10) NOT NULL, BatteryMah integer(10), WarrantyMonths integer(10) NOT NULL, ProductId integer(10) NOT NULL, PRIMARY KEY (ProductId), FOREIGN KEY(ProductId) REFERENCES Product(ProductId));
6CREATE TABLE ApparelDetail (SizeLabel varchar(255) NOT NULL, Color varchar(255) NOT NULL, Material varchar(255), ProductId integer(10) NOT NULL, PRIMARY KEY (ProductId), FOREIGN KEY(ProductId) REFERENCES Product(ProductId));
7CREATE TABLE Customer (CustomerId INTEGER NOT NULL PRIMARY KEY AUTOINCREMENT, FirstName varchar(255) NOT NULL, MiddleName varchar(255), LastName varchar(255) NOT NULL, Email varchar(255) NOT NULL, Phone varchar(255), DateOfBirth date, PasswordHash varchar(255) NOT NULL, CreatedAt date NOT NULL);
8CREATE TABLE Address (AddressId INTEGER NOT NULL PRIMARY KEY AUTOINCREMENT, CustomerId integer(10) NOT NULL, RecipientName varchar(255) NOT NULL, Phone varchar(255) NOT NULL, HouseNumber varchar(255), Street varchar(255) NOT NULL, District varchar(255) NOT NULL, City varchar(255) NOT NULL, IsDefault integer(1) NOT NULL, FOREIGN KEY(CustomerId) REFERENCES Customer(CustomerId));
9CREATE TABLE CustomerVIP (VipLevel integer(10) NOT NULL, DiscountRate double(10) NOT NULL, RewardPoints integer(10) NOT NULL, CustomerId integer(10) NOT NULL, PRIMARY KEY (CustomerId), FOREIGN KEY(CustomerId) REFERENCES Customer(CustomerId));
10CREATE TABLE CustomerNew (RegisteredDate date NOT NULL, WelcomeVoucher varchar(255), CustomerId integer(10) NOT NULL, PRIMARY KEY (CustomerId), FOREIGN KEY(CustomerId) REFERENCES Customer(CustomerId));
11CREATE TABLE Staff (StaffId INTEGER NOT NULL PRIMARY KEY AUTOINCREMENT, FullName varchar(255) NOT NULL, Email varchar(255) NOT NULL, PasswordHash varchar(255) NOT NULL, Role varchar(255) NOT NULL);
12CREATE TABLE Cart (CartId INTEGER NOT NULL PRIMARY KEY AUTOINCREMENT, CustomerId integer(10) NOT NULL, CreatedAt date NOT NULL, UpdatedAt date NOT NULL, FOREIGN KEY(CustomerId) REFERENCES Customer(CustomerId));
13CREATE TABLE CartItem (CartItemId INTEGER NOT NULL PRIMARY KEY AUTOINCREMENT, ProductId integer(10) NOT NULL, CartId integer(10) NOT NULL, Quantity integer(10) NOT NULL, CartPrice integer(10) NOT NULL, FOREIGN KEY(CartId) REFERENCES Cart(CartId), FOREIGN KEY(ProductId) REFERENCES Product(ProductId));
14CREATE TABLE Orders (OrderId INTEGER NOT NULL PRIMARY KEY AUTOINCREMENT, AddressId integer(10) NOT NULL, CustomerId integer(10) NOT NULL, OrderDate date NOT NULL, Status varchar(255) NOT NULL, PaymentMethod varchar(255) NOT NULL, Subtotal integer(10) NOT NULL, Discount integer(10) NOT NULL, ShippingFee integer(10) NOT NULL, TotalAmount integer(10) NOT NULL, FOREIGN KEY(CustomerId) REFERENCES Customer(CustomerId), FOREIGN KEY(AddressId) REFERENCES Address(AddressId));
15CREATE TABLE OrderItem (OrderItemId INTEGER NOT NULL PRIMARY KEY AUTOINCREMENT, ProductId integer(10) NOT NULL, OrderId integer(10) NOT NULL, ProductName varchar(255) NOT NULL, Quantity integer(10) NOT NULL, UnitPrice integer(10) NOT NULL, FOREIGN KEY(OrderId) REFERENCES Orders(OrderId), FOREIGN KEY(ProductId) REFERENCES Product(ProductId));
16CREATE TABLE OrderEvent (EventId INTEGER NOT NULL PRIMARY KEY AUTOINCREMENT, OrderId integer(10) NOT NULL, Status varchar(255) NOT NULL, OccurredAt date NOT NULL, Actor varchar(255) NOT NULL, Note varchar(255), FOREIGN KEY(OrderId) REFERENCES Orders(OrderId));
17CREATE TABLE Payment (PaymentId INTEGER NOT NULL PRIMARY KEY AUTOINCREMENT, OrderId integer(10) NOT NULL, Method varchar(255) NOT NULL, Amount integer(10) NOT NULL, Status varchar(255) NOT NULL, PaymentDate date NOT NULL, FOREIGN KEY(OrderId) REFERENCES Orders(OrderId));
18CREATE TABLE PayCash (CashReceived integer(10) NOT NULL, ChangeAmount integer(10) NOT NULL, PaymentId integer(10) NOT NULL, PRIMARY KEY (PaymentId), FOREIGN KEY(PaymentId) REFERENCES Payment(PaymentId));
19CREATE TABLE PayCredit (CardHolder varchar(255) NOT NULL, CardLast4 varchar(255) NOT NULL, Expiry varchar(255) NOT NULL, ProviderToken varchar(255), PaymentId integer(10) NOT NULL, PRIMARY KEY (PaymentId), FOREIGN KEY(PaymentId) REFERENCES Payment(PaymentId));
20CREATE TABLE Shipping (ShippingId INTEGER NOT NULL PRIMARY KEY AUTOINCREMENT, AddressId integer(10) NOT NULL, OrderId integer(10) NOT NULL, Carrier varchar(255) NOT NULL, TrackingNumber varchar(255) NOT NULL, ShippingFee integer(10) NOT NULL, Status varchar(255) NOT NULL, ShippedDate date NOT NULL, DeliveredDate date, FOREIGN KEY(OrderId) REFERENCES Orders(OrderId), FOREIGN KEY(AddressId) REFERENCES Address(AddressId));
