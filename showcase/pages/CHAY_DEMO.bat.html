1@echo off
2rem ABC Online Shop demo. Double-click: starts the server on a fresh demo database and opens the shop in the browser.
3rem Close this window (or press Ctrl+C) to stop the server.
4rem "CHAY_DEMO.bat keep" keeps the data of the previous run instead of recreating the database.
5setlocal
6title ABC Online Shop - demo server (close this window to stop)
7set "URL=http://localhost:8080/"
8set "JAR=target\abc-online-shop-1.0.0.jar"
9set "RESET=--abc.db.reset=true"
10if /i "%~1"=="keep" set "RESET="
11cd /d "%~dp0app"
12
13rem 0. Register the link type abcshop: for this Windows user (no admin rights, nothing else changes). The showcase
14rem pages use it to run this file when the server is not running yet ("Open the shop") and to show a file in
15rem File Explorer. It is written again on every run, so it always points to this copy of the project.
16rem Details in showcase\abcshop.ps1. To remove it: reg delete "HKCU\Software\Classes\abcshop" /f
17powershell -NoProfile -ExecutionPolicy RemoteSigned -File "%~dp0showcase\abcshop.ps1" -Register >nul 2>&1
18
19rem 1. Already running: only open the browser.
20powershell -NoProfile -Command "try { Invoke-WebRequest -UseBasicParsing -TimeoutSec 2 '%URL%' | Out-Null; exit 0 } catch { exit 1 }"
21if errorlevel 1 goto findjava
22echo The shop is already running at %URL%
23start "" "%URL%"
24timeout /t 3 >nul
25exit /b 0
26
27rem 2. Java 17 or newer: JAVA_HOME first, then the usual JDK folders. The "java" on PATH may be an older Java 8.
28:findjava
29set "JAVA="
30if defined JAVA_HOME call :tryjava "%JAVA_HOME%\bin\java.exe"
31if not defined JAVA for /d %%d in ("%ProgramFiles%\Eclipse Adoptium\jdk-*" "%ProgramFiles%\Java\jdk-*" "%ProgramFiles%\Microsoft\jdk-*") do if not defined JAVA call :tryjava "%%~d\bin\java.exe"
32if not defined JAVA for /f "delims=" %%j in ('where java 2^>nul') do if not defined JAVA call :tryjava "%%~j"
33if defined JAVA goto build
34echo.
35echo Java 17 or newer was not found. Install a JDK 17 (for example Eclipse Temurin 17) and run this file again.
36pause
37exit /b 1
38
39rem 3. Build the application once if the jar is missing.
40:build
41for %%j in ("%JAVA%") do set "JAVA_HOME=%%~dpj.."
42if exist "%JAR%" goto run
43echo Building the application (first run only, about one minute)...
44call mvn -q package -DskipTests
45if exist "%JAR%" goto run
46echo.
47echo The build failed. Run "mvn package" in the app folder to see the error.
48pause
49exit /b 1
50
51rem 4. Open the browser as soon as the server answers, and run the server in this window.
52:run
53start "" /b powershell -NoProfile -WindowStyle Hidden -Command "for ($i = 0; $i -lt 120; $i++) { try { Invoke-WebRequest -UseBasicParsing -TimeoutSec 2 '%URL%' | Out-Null; Start-Process '%URL%'; break } catch { Start-Sleep -Seconds 1 } }"
54echo.
55echo ABC Online Shop - %URL%
56if defined RESET (echo Database: fresh demo data) else (echo Database: kept from the previous run)
57echo.
58echo Accounts, password demo1234:
59echo an.nguyen@example.com VIP customer
60echo lan.tran@example.com new customer
61echo sales@example.com sales staff, order board
62echo warehouse@example.com warehouse staff, stock
63echo Cards: 4242 4242 4242 4242 approved, 4000 0000 0000 0002 declined. Alt+L: design lens.
64echo.
65echo The browser opens when the server is ready. Close this window to stop the server.
66echo.
67"%JAVA%" -jar "%JAR%" %RESET%
68echo.
69echo The server has stopped.
70pause
71exit /b 0
72
73:tryjava
74if not exist "%~1" exit /b 0
75"%~1" -XshowSettings:properties -version 2>&1 | findstr /r /c:"java.specification.version = 1[7-9]" /c:"java.specification.version = [2-9][0-9]" >nul
76if not errorlevel 1 set "JAVA=%~1"
77exit /b 0
