2"""Offline evaluation of the AI search (docs/SPEC.md 6.2) against a running server with the demo data.
3
4Text: the 30 labelled queries of tests/search/queries.json through the keyword search of UC04 (GET /api/products?q=)
5and through the AI search (GET /api/search?q=). Measures per method:
6 hit@1 share of queries whose first result is relevant
7 recall@5 share of the relevant products found in the first five results (averaged over queries)
8 MRR mean of 1 / rank of the first relevant result (0 when none)
9 zero result share of queries with no result at all
10 latency median and maximum server time (AI search) or round-trip time (keyword search)
11Photo: the 84 photos of tests/search/photos (python tests/search/make_photos.py) through POST /api/search/photo;
12top-1 and top-3 accuracy of the product the photo shows, per kind of photo.
13Usage: python tests/search_eval.py [base-url] writes docs/evidence/search_eval.txt; exit code 1 when the AI search
14does not beat the keyword search on MRR or when photo top-1 accuracy is under 0.8.
15"""
16import json
17import statistics
18import sys
19import time
20import urllib.parse
21import urllib.request
22import uuid
23from collections import defaultdict
24from pathlib import Path
25
26ROOT = Path(__file__).resolve().parent.parent
27BASE = sys.argv[1] if len(sys.argv) > 1 else "http://localhost:8080"
28QUERIES = json.loads((ROOT / "tests" / "search" / "queries.json").read_text(encoding="utf-8"))["queries"]
29PHOTOS = ROOT / "tests" / "search" / "photos"
30OUT = ROOT / "docs" / "evidence" / "search_eval.txt"
31
32
33def get(path):
34 t0 = time.perf_counter()
35 with urllib.request.urlopen(BASE + path) as r:
36 data = json.load(r)
37 return data, (time.perf_counter() - t0) * 1000
38
39
40def post_photo(path):
41 boundary = uuid.uuid4().hex
42 body = (f"--{boundary}\r\nContent-Disposition: form-data; name=\"photo\"; filename=\"{path.name}\"\r\n"
43 f"Content-Type: image/jpeg\r\n\r\n").encode() + path.read_bytes() + f"\r\n--{boundary}--\r\n".encode()
44 req = urllib.request.Request(BASE + "/api/search/photo", data=body, method="POST",
45 headers={"Content-Type": f"multipart/form-data; boundary={boundary}"})
46 with urllib.request.urlopen(req) as r:
47 return json.load(r)

49
50def score(ranked, relevant):
51 rel = set(relevant)
52 first = next((i + 1 for i, s in enumerate(ranked) if s in rel), None)
53 return {
54 "hit1": 1.0 if ranked[:1] and ranked[0] in rel else 0.0,
55 "recall5": len(rel & set(ranked[:5])) / len(rel),
56 "rr": 1.0 / first if first else 0.0,
57 "zero": 1.0 if not ranked else 0.0,
58 }

61def main():
62 lines, rows = [], []
63 totals = {"keyword": defaultdict(float), "ai": defaultdict(float)}
64 times = {"keyword": [], "ai": []}
65 for item in QUERIES:
66 q = urllib.parse.quote(item["q"])
67 kw, t_kw = get(f"/api/products?q={q}")
68 ai, _ = get(f"/api/search?q={q}")
69 kw_ranked = [p["sku"] for p in kw]
70 ai_ranked = [r["product"]["sku"] for r in ai["results"]] if not ai["fallback"] else [r["product"]["sku"] for r in ai["results"]]
71 times["keyword"].append(t_kw)
72 times["ai"].append(ai["tookMs"])
73 s_kw, s_ai = score(kw_ranked, item["relevant"]), score(ai_ranked, item["relevant"])
74 for k, v in s_kw.items():
75 totals["keyword"][k] += v
76 for k, v in s_ai.items():
77 totals["ai"][k] += v
78 rows.append((item["q"], item["type"], s_kw, s_ai, ai_ranked[:3], ai["mode"]))
79 n = len(QUERIES)
80 lines.append(f"AI search evaluation, {time.strftime('%Y-%m-%d %H:%M')}, server {BASE}")
81 lines.append(f"Text: {n} labelled queries (tests/search/queries.json)")
82 lines.append("")
83 lines.append(f"{'method':10} {'hit@1':>7} {'recall@5':>9} {'MRR':>6} {'zero result':>12} {'median ms':>10} {'max ms':>7}")
84 for m, label in (("keyword", "keyword"), ("ai", "AI search")):
85 t = totals[m]
86 lines.append(f"{label:10} {t['hit1'] / n:7.2f} {t['recall5'] / n:9.2f} {t['rr'] / n:6.2f} {t['zero'] / n:12.2f} "
87 f"{statistics.median(times[m]):10.0f} {max(times[m]):7.0f}")
88 lines.append("")
89 lines.append("Per query (RR = 1 / rank of the first relevant result; AI top 3 by SKU):")
90 for q, typ, s_kw, s_ai, top, mode in rows:
91 lines.append(f" {q[:34]:34} {typ:11} keyword RR {s_kw['rr']:.2f} | AI RR {s_ai['rr']:.2f} {'(fallback)' if mode == 'KEYWORD' else '':10} {', '.join(top)}")
92
93 by_kind = defaultdict(lambda: [0, 0, 0])
94 photo_ms = []
95 misses = []
96 photos = sorted(PHOTOS.glob("*.jpg"))
97 for f in photos:
98 sku, kind = f.stem.rsplit("_", 1)
99 r = post_photo(f)
100 photo_ms.append(r["tookMs"])
101 ranked = [x["product"]["sku"] for x in r["results"]]
102 by_kind[kind][0] += 1
103 by_kind[kind][1] += 1 if ranked[:1] == [sku] else 0
104 by_kind[kind][2] += 1 if sku in ranked[:3] else 0
105 if ranked[:1] != [sku]:
106 misses.append(f"{f.name}: first {ranked[0] if ranked else 'none'}, true product at {ranked.index(sku) + 1 if sku in ranked else '-'}")
107 lines.append("")
108 lines.append(f"Photo: {len(photos)} photos made from the catalogue images (tests/search/make_photos.py)")
109 lines.append(f"{'kind':12} {'photos':>6} {'top-1':>6} {'top-3':>6}")
110 all_n = all_1 = all_3 = 0
111 for kind, (cnt, t1, t3) in sorted(by_kind.items()):
112 lines.append(f"{kind:12} {cnt:6} {t1 / cnt:6.2f} {t3 / cnt:6.2f}")
113 all_n, all_1, all_3 = all_n + cnt, all_1 + t1, all_3 + t3
114 lines.append(f"{'all':12} {all_n:6} {all_1 / all_n:6.2f} {all_3 / all_n:6.2f} median {statistics.median(photo_ms):.0f} ms, max {max(photo_ms)} ms")
115 if misses:
116 lines.append("Photos whose product was not first:")
117 lines.extend(" " + m for m in misses)
118 OUT.write_text("\n".join(lines) + "\n", encoding="utf-8")
119 print("\n".join(lines))
120 ok = totals["ai"]["rr"] > totals["keyword"]["rr"] and all_1 / all_n >= 0.8
121 return 0 if ok else 1

124if __name__ == "__main__":
125 sys.exit(main())
