1# -*- coding: utf-8 -*-
2"""Code generated by Visual Paradigm (Tools > Code > Instant Generator, Java, model folder "4 Design") against the
3implemented code.
4
5Input
6 docs/evidence/vp_codegen/**/*.java the skeleton VP wrote from the design model of section 4.4 (bodies throw
7 UnsupportedOperationException); VP prefixes the packages with "__Design"
8 because the model folder is named "4 Design"
9 app/target/classes compiled classes (run "mvn -q compile" first), read with javap
10 app/src/main/resources/static/js/view/*.js the view classes (the design's view package runs in the browser)
11
12Checks: every generated class exists in the same package, every generated field is a field (or a derived getter) of
13the same type, every generated method exists with the same parameter types and return type; for view classes the
14method exists in the JS class with the same number of parameters. Association ends that VP writes as Vector<T> are
15compared as List<T>. Exit code = number of problems.
16"""
17import re
18import sys
19from collections import defaultdict
20from pathlib import Path
21
22sys.path.insert(0, str(Path(__file__).resolve().parent))
23from design_vs_code import ROOT, javap_package, js_views, same_type, split_params # noqa: E402
24
25GEN = ROOT / "docs" / "evidence" / "vp_codegen"
26MOD = r"(?:public |protected |private |static |final |abstract )*"
27
28
29def norm(t):
30 """Undo what the Instant Generator does to type names: the comma of Map<Integer, Product> becomes "__", the
31 wildcard of List<?> becomes "_" (both are not Java identifiers), association ends are Vector<T>."""
32 t = re.sub(r"\bVector<", "List<", t.strip())
33 t = t.replace("__", ",")
34 return re.sub(r"(?<=[<,])_(?=[>,])", "?", t)
35
36
37def same_param(gen, code):
38 # a varargs parameter of the model (Role...) is written as Object by the Instant Generator
39 return same_type(gen, code) or gen == "Object" and code.endswith("...")
40
41
42def parse(f):
43 text = f.read_text(encoding="utf-8")
44 pkg = re.search(r"^package ([\w.]+);", text, re.M).group(1).replace("__Design.", "")
45 cm = re.search(r"^public (?:abstract )?(class|interface|enum) (\w+)", text, re.M)
46 kind, name = cm.group(1), cm.group(2)
47 fields, methods = {}, defaultdict(list)
48 body = text[cm.end():]
49 if kind == "enum":
50 literals = body.split(";", 1)[0]
51 for lit in re.findall(r"\b([A-Z][A-Z0-9_]*)\b", literals):
52 fields[lit] = name
53 for line in body.splitlines():
54 if line.startswith("\t\t") or not line.startswith("\t"):
55 continue # VP indents declarations with one tab; two tabs are method bodies
56 s = line.strip()
57 mm = re.match(MOD + r"(?:<.*?> )?([\w<>\[\], .?]+?) (\w+)\((.*)\)\s*(?:\{|;)$", s)
58 if mm and mm.group(2) != name:
59 params = [norm(" ".join(p.split()[:-1])) for p in split_params(mm.group(3))]
60 methods[mm.group(2)].append((params, norm(mm.group(1))))
61 continue
62 mf = re.match(MOD + r"([\w<>\[\], .?]+?) (\w+)(?: = .*)?;$", s)
63 if mf and "(" not in s:
64 fields[mf.group(2)] = norm(mf.group(1))
65 return pkg, kind, name, fields, methods

68def main():
69 problems, checked = [], defaultdict(int)
70 code, views = {}, js_views()
71 files = sorted(GEN.rglob("*.java"))
72 for f in files:
73 pkg, kind, cls, fields, methods = parse(f)
74 rel = f.relative_to(GEN).as_posix()
75 if pkg.endswith(".view"):
76 js = views.get(cls)
77 checked["view classes"] += 1
78 if js is None:
79 problems.append(f"{rel}: view class {cls} has no JS module")
80 continue
81 for op, sigs in methods.items():
82 for params, _ in sigs:
83 checked["view operations"] += 1
84 if op not in js:
85 problems.append(f"{cls}.{op}() generated, missing in the JS view")
86 elif len(js[op][0]) != len(params):
87 problems.append(f"{cls}.{op}: generated with {len(params)} parameters, {js[op][1]} has {len(js[op][0])}")
88 continue
89 if pkg not in code:
90 code[pkg] = javap_package(pkg)
91 c = code[pkg].get(cls)
92 checked["classes"] += 1
93 if c is None:
94 problems.append(f"{rel}: class {pkg}.{cls} not found in the compiled code")
95 continue
96 if kind != c["kind"] and not (kind == "class" and c["kind"] == "record"):
97 problems.append(f"{cls}: generated as {kind}, the code has a {c['kind']}")
98 for a, typ in fields.items():
99 checked["fields"] += 1
100 if a in c["fields"]:
101 if typ and not same_type(typ, c["fields"][a]):
102 problems.append(f"{cls}.{a}: generated type {typ}, code type {c['fields'][a]}")
103 elif not (a in c["methods"] or ("get" + a[0].upper() + a[1:]) in c["methods"] or ("is" + a[0].upper() + a[1:]) in c["methods"]):
104 problems.append(f"{cls}.{a}: generated field not found in the code")
105 for op, sigs in methods.items():
106 for params, ret in sigs:
107 checked["methods"] += 1
108 if op not in c["methods"]:
109 problems.append(f"{cls}.{op}({', '.join(params)}): generated method not found in the code")
110 continue
111 match = [m for m in c["methods"][op] if len(m[0]) == len(params) and all(same_param(w, g) for w, g in zip(params, m[0]))]
112 if not match:
113 got = " | ".join(f"({', '.join(p)}) : {rt}" for p, rt in c["methods"][op])
114 problems.append(f"{cls}.{op}({', '.join(params)}): parameters differ, code has {got}")
115 elif ret and ret != "void" and not any(same_type(ret, m[1]) for m in match):
116 problems.append(f"{cls}.{op}: generated return {ret}, code returns {match[0][1]}")
117 for p in problems:
118 print("FAIL " + p)
119 print(f"{len(files)} generated files; checked " + ", ".join(f"{v} {k}" for k, v in checked.items()))
120 print(f"{len(problems)} problems" if problems else "the code implements every class, field and method that VP generated")
121 return len(problems)

124if __name__ == "__main__":
125 sys.exit(min(main(), 100))
