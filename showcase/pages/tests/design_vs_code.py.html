1# -*- coding: utf-8 -*-
2"""Design model (exported from the Visual Paradigm project) against the code.
3
4Inputs
5 docs/evidence/design_model.tsv package, class, kind, name, parameter types, type[, parameter names]
6 (written by the VP plugin from the design classes of the project)
7 docs/evidence/seq_messages.tsv diagram, number, sender class, receiver class, label, is return
8 (the messages of the design sequence diagrams), optional
9 app/target/classes compiled Java classes (run "mvn -q compile" first), read with javap
10 app/src/main/resources/static/js/view/*.js the view classes
11
12Checks
13 every design class exists in its package (nested classes count: OrderService$Checkout is "Checkout")
14 every attribute exists as a field of the same type, every operation as a method with the same name,
15 parameter types and return type (simple names; generic arguments are compared too)
16 every view operation exists in the JS class with the same number of parameters
17 every call message of a design sequence names an operation of the receiver's design class and of the code
18Prints one line per problem and a summary; exit code = number of problems.
19"""

30CLASSES = ROOT / "app" / "target" / "classes"
31VIEWS = ROOT / "app" / "src" / "main" / "resources" / "static" / "js" / "view"

34def simple(t):
35 """'java.util.List<vn.ptit.ecom.dao.OrderDAO$OrderSummary>' -> 'List<OrderSummary>'."""
36 t = re.sub(r"[\w.]+\$", "", t) # outer classes of nested ones
37 t = re.sub(r"\b(?:[a-z_]\w*\.)+", "", t) # package prefixes
38 return t.replace(" ", "")

41def raw(t):
42 return simple(t).split("<")[0]

45def javap_package(pkg):
46 """{simple class name: {"kind", "fields": {name: type}, "methods": {name: [(params, return)]}}}"""
47 folder = CLASSES / Path(*pkg.split("."))
48 # relative paths from the classes folder: the Java tools read arguments in the ANSI code page, which cannot hold
49 # the Vietnamese letters of the course folder name
50 files = sorted(str(p.relative_to(CLASSES)) for p in folder.glob("*.class"))
51 out = subprocess.run(["javap", "-p"] + files, capture_output=True, text=True, encoding="utf-8", cwd=CLASSES).stdout
52 classes, cur = {}, None
53 for line in out.splitlines():
54 line = line.strip()
55 m = re.match(r"(?:public |protected |private )?(?:abstract |static |final |sealed )*(class|interface|enum) ([\w.$]+)", line)
56 if m:
57 name = m.group(2).split(".")[-1].split("$")[-1]
58 cur = classes.setdefault(name, {"kind": m.group(1), "fields": {}, "methods": defaultdict(list)})
59 if "extends java.lang.Record" in line:
60 cur["kind"] = "record"
61 if "extends java.lang.Enum" in line:
62 cur["kind"] = "enum"
63 continue
64 if cur is None or not line.endswith(";") or "lambda$" in line or "access$" in line:
65 continue
66 sig = re.sub(r" throws .*;$", ";", line)
67 mm = re.match(r"(?:(?:public|protected|private|static|final|abstract|synchronized|native|default|transient|volatile) )*"
68 r"(?:<.*?> )?(.+?) ([\w$]+)\((.*)\);$", sig)
69 if mm:
70 params = [simple(p) for p in split_params(mm.group(3))]
71 cur["methods"][mm.group(2)].append((params, simple(mm.group(1))))
72 continue
73 mf = re.match(r"(?:(?:public|protected|private|static|final|transient|volatile) )*(.+?) ([\w$]+);$", sig)
74 if mf:
75 cur["fields"][mf.group(2)] = simple(mf.group(1))
76 return classes

79def split_params(s):
80 """split on commas outside <...>."""
81 parts, depth, buf = [], 0, ""
82 for ch in s:
83 if ch == "<":
84 depth += 1
85 elif ch == ">":
86 depth -= 1
87 if ch == "," and depth == 0:
88 parts.append(buf.strip())
89 buf = ""
90 else:
91 buf += ch
92 if buf.strip():
93 parts.append(buf.strip())
94 return parts

97def same_type(design, code):
98 if not design:
99 return True # the design leaves the type open (e.g. void operations)
100 d, c = simple(design), simple(code)
101 if d == c or raw(d) == raw(c) and ("<" not in d or "<?>" in d):
102 return True
103 return d.replace("...", "[]") == c.replace("...", "[]")

106def js_views():
107 views = {}
108 for f in sorted(VIEWS.glob("*.js")):
109 text = f.read_text(encoding="utf-8")
110 cls = re.search(r"export class (\w+)", text)
111 if not cls:
112 continue
113 methods = {}
114 for i, line in enumerate(text.splitlines(), 1):
115 m = re.match(r"\s{2,4}(?:async\s+)?(\w+)\s*\(([^)]*)\)\s*\{", line)
116 if m and m.group(1) not in ("if", "for", "while", "switch", "catch", "constructor"):
117 params = [p.strip() for p in m.group(2).split(",") if p.strip()]
118 methods[m.group(1)] = (params, f"{f.relative_to(ROOT).as_posix()}:{i}")
119 views[cls.group(1)] = methods
120 return views

123def main():
124 rows = [r for r in csv.reader(TSV.read_text(encoding="utf-8").splitlines(), delimiter="\t") if r and not r[0].startswith("#")]
125 problems, checked = [], defaultdict(int)
126 design = defaultdict(lambda: {"kind": "", "attributes": {}, "operations": defaultdict(list)})
127 for r in rows:
128 r += [""] * (7 - len(r))
129 pkg, cls, kind, name, ptypes, typ, pnames = r[:7]
130 d = design[(pkg, cls)]
131 if kind == "class":
132 d["kind"] = name or "class"
133 elif kind == "attribute":
134 d["attributes"][name] = typ
135 elif kind == "operation":
136 d["operations"][name].append((ptypes, typ, pnames))
137 code = {}
138 views = js_views()
139 for (pkg, cls), d in design.items():
140 if pkg.endswith(".view"):
141 js = views.get(cls)
142 if js is None:
143 problems.append(f"view class {cls} missing in {VIEWS.relative_to(ROOT).as_posix()}")
144 continue
145 for op, sigs in d["operations"].items():
146 checked["view operations"] += 1
147 if op not in js:
148 problems.append(f"{cls}.{op}() missing in the JS view")
149 continue
150 for ptypes, _, pnames in sigs:
151 n_design = len(pnames.split(",")) if pnames else (ptypes.count(",") + 1 if ptypes else None)
152 n_code = len(js[op][0])
153 if n_design is not None and n_design != n_code:
154 problems.append(f"{cls}.{op}: design has {n_design} parameters, {js[op][1]} has {n_code}")
155 continue
156 if pkg not in code:
157 code[pkg] = javap_package(pkg)
158 c = code[pkg].get(cls)
159 checked["classes"] += 1
160 if c is None:
161 problems.append(f"class {pkg}.{cls} not found in the compiled code")
162 continue
163 if d["kind"] == "interface" and c["kind"] != "interface":
164 problems.append(f"{cls} is an interface in the design but a {c['kind']} in the code")
165 if d["kind"] == "enumeration" and c["kind"] != "enum":
166 problems.append(f"{cls} is an enumeration in the design but a {c['kind']} in the code")
167 for a, typ in d["attributes"].items():
168 checked["attributes"] += 1
169 if a not in c["fields"]:
170 # enum literals and record components are fields too; an attribute may also be a derived getter
171 if a in c["methods"] or ("get" + a[0].upper() + a[1:]) in c["methods"] or ("is" + a[0].upper() + a[1:]) in c["methods"]:
172 continue
173 problems.append(f"{cls}.{a} : attribute not found in the code")
174 elif typ and not same_type(typ, c["fields"][a]):
175 problems.append(f"{cls}.{a} : design type {typ}, code type {c['fields'][a]}")
176 for op, sigs in d["operations"].items():
177 for ptypes, typ, _ in sigs:
178 checked["operations"] += 1
179 if op not in c["methods"]:
180 problems.append(f"{cls}.{op}({ptypes}) : operation not found in the code")
181 continue
182 want = split_params(ptypes) if ptypes else [] # "Map<Integer,BufferedImage>" is one parameter
183 match = [m for m in c["methods"][op] if len(m[0]) == len(want) and all(same_type(w, g) for w, g in zip(want, m[0]))]
184 if not match:
185 got = " | ".join(f"({', '.join(p)}) : {rt}" for p, rt in c["methods"][op])
186 problems.append(f"{cls}.{op}({ptypes}) : parameters differ, code has {got}")
187 elif typ and not any(same_type(typ, m[1]) for m in match):
188 problems.append(f"{cls}.{op}({ptypes}) : design returns {typ}, code returns {match[0][1]}")
189 # sequence messages: the receiver's design class (and its code) must have the called operation
190 if SEQ.exists():
191 by_class = defaultdict(set)
192 for (pkg, cls), d in design.items():
193 by_class[cls] |= set(d["operations"])
194 for r in csv.reader(SEQ.read_text(encoding="utf-8").splitlines(), delimiter="\t"):
195 if not r or r[0].startswith("#"):
196 continue
197 diagram, number, sender, receiver, label, is_return = r[:6]
198 if is_return == "true" or receiver not in by_class:
199 continue # returns and messages to actors / external systems
200 m = re.search(r"(?:^|: )(\w+)\(", label)
201 if not m:
202 continue # plain event, e.g. "click Add to cart" sent to a view
203 checked["sequence calls"] += 1
204 if m.group(1) == receiver:
205 continue # constructor call, e.g. CardData(holder, number, expiry, cvv)
206 if m.group(1) not in by_class[receiver] and m.group(1) in by_class.get("Work", set()):
207 # Database.inTransaction(work) calls back Work.run(daos); the Work is the lambda written in the
208 # service, so the diagrams send run(daos) to the service lifeline. Accept it only when the service
209 # really passes a lambda to inTransaction.
210 src = next((ROOT / "app" / "src" / "main" / "java").rglob(f"{receiver}.java"), None)
211 if src and "inTransaction(" in src.read_text(encoding="utf-8"):
212 continue
213 if m.group(1) not in by_class[receiver]:
214 problems.append(f"{diagram} message {number} '{label}': {receiver} has no operation {m.group(1)}")
215 # the AI package depends on nothing else of the application (G0, G1, G8): Document in, Hit out
216 for f in sorted((ROOT / "app" / "src" / "main" / "java" / "vn" / "ptit" / "ecom" / "ai").glob("*.java")):
217 checked["ai files without application imports"] += 1
218 for line in f.read_text(encoding="utf-8").splitlines():
219 if line.startswith("import vn.ptit.ecom."):
220 problems.append(f"ai/{f.name} imports {line[7:].rstrip(';')}: the AI package must not depend on the application")
221 for p in problems:
222 print("FAIL " + p)
223 print("checked " + ", ".join(f"{v} {k}" for k, v in checked.items()))
224 print(f"{len(problems)} problems" if problems else "design and code agree")
225 return len(problems)

228if __name__ == "__main__":
229 sys.exit(min(main(), 100))
