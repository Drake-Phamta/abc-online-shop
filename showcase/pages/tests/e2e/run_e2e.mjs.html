1// End-to-end run of the implemented use cases in the installed Microsoft Edge (Playwright, channel "msedge").
2// The server must run on http://localhost:8080 with a fresh database (java -jar target/abc-online-shop-1.0.0.jar --abc.db.reset=true).
3// Every step is checked on the page and, where a use case changes data, through the API; a screenshot of each step
4// goes to docs/evidence/e2e/. Exit code = number of failed checks.
5import { chromium } from "playwright";
6import { mkdirSync, writeFileSync } from "node:fs";
7import { dirname, join } from "node:path";
8import { fileURLToPath } from "node:url";
9
10const BASE = process.env.BASE || "http://localhost:8080";
11const OUT = join(dirname(fileURLToPath(import.meta.url)), "..", "..", "docs", "evidence", "e2e");
12mkdirSync(OUT, { recursive: true });

14const results = [];
15let shot = 0;
16function check(name, ok, detail = "") {
17 results.push({ name, ok: !!ok, detail });
18 console.log(`${ok ? "PASS" : "FAIL"} ${name}${detail ? " (" + detail + ")" : ""}`);
19}

21const browser = await chromium.launch({ channel: "msedge", headless: true });
22const context = await browser.newContext({ viewport: { width: 1440, height: 900 }, deviceScaleFactor: 1 });
23const page = await context.newPage();
24const consoleErrors = [];
25// the browser logs every refused request (e.g. 409 for BR12) as "Failed to load resource"; those are expected answers
26page.on("console", (m) => { if (m.type() === "error" && !m.text().startsWith("Failed to load resource")) consoleErrors.push(m.text()); });
27page.on("pageerror", (e) => consoleErrors.push(e.message));

29async function snap(name, fullPage = false, keepToast = false) {
30 shot += 1;
31 if (!keepToast) await clearToast(); // a toast left from an earlier step would hide part of the page
32 const file = `${String(shot).padStart(2, "0")}_${name}.png`;
33 await page.waitForTimeout(250); // let the toast and transitions settle
34 await page.screenshot({ path: join(OUT, file), fullPage });
35 return file;
36}
37async function go(hash, ready) {
38 // first visit loads the page; later moves change the hash like the app's own links (same document)
39 if (!page.url().startsWith(BASE)) await page.goto(`${BASE}/#/${hash}`);
40 else await page.evaluate((h) => { location.hash = "#/" + h; }, hash);
41 await page.waitForLoadState("networkidle");
42 if (ready) await page.waitForSelector(ready); // views render after their fetch, wait for the content
43}
44async function clearToast() {
45 await page.evaluate(() => document.querySelectorAll(".toast").forEach((t) => t.remove()));
46}
47async function apiGet(path) {
48 return page.evaluate(async (p) => { const r = await fetch(p, { credentials: "same-origin" }); return { status: r.status, body: await r.json().catch(() => null) }; }, path);
49}
50async function login(email) {
51 await go("login", "#email");

53 await page.fill("#password", "demo1234");
54 await page.click("#login-form button[type=submit]");
55 await page.waitForFunction(() => !location.hash.startsWith("#/login"));
56 await page.waitForLoadState("networkidle");
57}
58async function logout() {
59 await clearToast();
60 await page.click("#logout");
61 await page.waitForSelector("#account a[href='#/login']");
62 await page.waitForFunction(() => location.hash === "#/" || location.hash === ""); // the app goes home after logout
63 await page.waitForSelector("#view a.dept"); // home rendered: no late render can replace the next page
64 await page.waitForLoadState("networkidle");
65}
66async function toastText() {
67 const t = page.locator(".toast");
68 await t.waitFor({ state: "visible", timeout: 5000 }).catch(() => {});
69 return (await t.textContent().catch(() => "")) || "";
70}
