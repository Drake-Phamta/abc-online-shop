2"""End-to-end smoke run over HTTP against a freshly seeded server (java -jar ... --abc.db.reset=true).
3
4Walks UC02 -> UC04/05 -> UC06 -> UC07 -> UC08 (declined, then approved) -> UC14 (process, ship, deliver) -> UC09 and
5UC15, and checks the key business rules (BR02, BR05, BR08, BR09, BR13). Prints one line per check
6and exits non-zero on the first failure. Usage: python tests/smoke_api.py [base_url]
7"""

15BASE = sys.argv[1] if len(sys.argv) > 1 else "http://localhost:8080"
16DB = sys.argv[2] if len(sys.argv) > 2 else "app/data/abc_shop.db"

19class Client:
20 def __init__(self):
21 self.opener = urllib.request.build_opener(urllib.request.HTTPCookieProcessor(http.cookiejar.CookieJar()))
22
23 def call(self, method, path, body=None):
24 data = None if body is None else json.dumps(body).encode()
25 req = urllib.request.Request(BASE + path, data=data, method=method, headers={"Content-Type": "application/json"})
26 try:
27 with self.opener.open(req) as r:
28 return r.status, json.loads(r.read() or b"null")
29 except urllib.error.HTTPError as e:
30 return e.code, json.loads(e.read() or b"null")
31
32
33def check(name, ok, detail=""):
34 print(("PASS " if ok else "FAIL ") + name + (f" [{detail}]" if detail else ""))
35 if not ok:
36 sys.exit(1)

39def stock(product_id):
40 with sqlite3.connect(DB) as c:
41 return c.execute("SELECT stock_quantity FROM product WHERE product_id = ?", (product_id,)).fetchone()[0]
