1"""Marks what changed since the previous submission with a change bar: a blue line in the left margin next to every
2paragraph, heading and caption of the new essay that contains a sentence the old essay does not have. A table or a
3figure is marked at its caption (and the picture beside it) when it is new or mostly changed. The text stays black.
4Figure, table, section and chapter numbers are ignored in the comparison, because a renumbering alone is no change.
5Code listings and the table of contents are not marked.
6
7usage: python mark_changes.py <new.docx> <old.docx> [--dry] (the new file is changed in place unless --dry)
8"""
9import re
10import sys
11
12from docx import Document
13from docx.oxml import OxmlElement
14from docx.oxml.ns import qn
15from docx.table import Table
16from docx.text.paragraph import Paragraph
17
18BAR_COLOR = "1F5FA8"
19SENT = re.compile(r"(?<=[.!?:;])\s+(?=[A-Z0-9\"(«])")
20AFTER_PBDR = ("w:shd", "w:tabs", "w:suppressAutoHyphens", "w:kinsoku", "w:wordWrap", "w:overflowPunct",
21 "w:topLinePunct", "w:autoSpaceDE", "w:autoSpaceDN", "w:bidi", "w:adjustRightInd", "w:snapToGrid",
22 "w:spacing", "w:ind", "w:contextualSpacing", "w:mirrorIndents", "w:suppressOverlap", "w:jc",
23 "w:textDirection", "w:textAlignment", "w:textboxTightWrap", "w:outlineLvl", "w:divId", "w:cnfStyle",
24 "w:rPr", "w:sectPr", "w:pPrChange")
25
26
27def norm(s):
28 s = s.lower()
29 s = re.sub(r"\b(figures?|tables?|sections?|chapters?)\s+\d+(\.\d+)*(\s+(to|and)\s+\d+(\.\d+)*)?", r"\1 #", s)
30 s = re.sub(r"[“”\"'’‘]", "", s)
31 s = re.sub(r"\s+", " ", s)
32 return s.strip(" .;:,")
33
34
35def all_text(doc):
36 parts = [p.text for p in doc.paragraphs]
37 for t in doc.tables:
38 for row in t.rows:
39 parts += [c.text for c in row.cells]
40 return " ".join(parts)
41
42
43def is_skipped(p):
44 fonts = {r.font.name for r in p.runs if r.text.strip()}
45 return (bool(fonts) and fonts <= {"Consolas", "Courier New"}) or (p.style is not None and "TOC" in p.style.name)
46
47
48def changed_chars(text, old):
49 total = changed = 0
50 for sent in SENT.split(text):
51 n = norm(sent)
52 total += len(sent)
53 if n and len(n) > 3 and n not in old:
54 changed += len(sent)
55 return changed, total
56
57
58def bar(p):
59 pPr = p._p.get_or_add_pPr()
60 if pPr.find(qn("w:pBdr")) is not None:
61 return
62 pbdr = OxmlElement("w:pBdr")
63 left = OxmlElement("w:left")
64 for k, v in (("w:val", "single"), ("w:sz", "18"), ("w:space", "10"), ("w:color", BAR_COLOR)):
65 left.set(qn(k), v)
66 pbdr.append(left)
67 pPr.insert_element_before(pbdr, *AFTER_PBDR)
68
69
70def main():
71 new_path, old_path = sys.argv[1], sys.argv[2]
72 dry = "--dry" in sys.argv
73 old = norm(all_text(Document(old_path)))
74 doc = Document(new_path)
75 body = doc.element.body
76 marked = paragraphs = 0
77 total_chars = changed_total = 0
78 blocks = list(body.iterchildren())
79 for i, el in enumerate(blocks):
80 if el.tag == qn("w:p"):
81 p = Paragraph(el, doc)
82 text = p.text
83 if not text.strip():
84 # a picture: marked when the caption under it is marked
85 if el.xpath(".//w:drawing") and i + 1 < len(blocks) and blocks[i + 1].tag == qn("w:p"):
86 cap = Paragraph(blocks[i + 1], doc)
87 c, t = changed_chars(cap.text, old)
88 if t and c and not dry:
89 bar(p)
90 continue
91 if is_skipped(p):
92 continue
93 c, t = changed_chars(text, old)
94 total_chars += t
95 changed_total += c
96 if c:
97 paragraphs += 1
98 if not dry:
99 bar(p)
100 marked += 1
101 elif el.tag == qn("w:tbl"):
102 table = Table(el, doc)
103 c = t = 0
104 for row in table.rows:
105 for cell in row.cells:
106 cc, tt = changed_chars(cell.text, old)
107 c, t = c + cc, t + tt
108 total_chars += t
109 changed_total += c
110 if t and c / t >= 0.3:
111 prev = el.getprevious()
112 if prev is not None and prev.tag == qn("w:p") and not dry:
113 bar(Paragraph(prev, doc))
114 marked += 1
115 print(f"changed text: {changed_total} of {total_chars} characters ({100 * changed_total / max(total_chars, 1):.0f} %), "
116 f"{paragraphs} paragraphs")
117 if not dry:
118 doc.save(new_path)
119 print(f"change bars on {marked} paragraphs and captions:", new_path)
120
121
122if __name__ == "__main__":
123 main()
