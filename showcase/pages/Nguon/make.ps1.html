1# usage: make.ps1 [-Root <07_Tieu_luan folder>] [-Name <file name without extension>]
2# build docx -> Word (in a background job: Word COM hangs when driven from this host directly):
3# update TOC, save docx, export PDF -> page sheet + free space per page. Word works in C:\a3cap\tieuluan_out
4# (plain ASCII path); a successful build is then copied to <Root>\Bai_nop.
5param([string]$Root = (Split-Path $PSScriptRoot -Parent), [string]$Name = "tieuluan_V4_04_10_phamtuananh_018")
6$env:PYTHONUTF8 = "1"
7$env:PYTHONIOENCODING = "utf-8"
8$out = "C:\a3cap\tieuluan_out"
9New-Item -ItemType Directory -Path $out -Force | Out-Null
10$name = $Name
11$docx = Join-Path $out "$name.docx"
12$pdf = Join-Path $out "$name.pdf"
13$started = Get-Date
14python (Join-Path $Root "Nguon\build_tieuluan.py") $Root $docx
15if ($LASTEXITCODE -ne 0) { "BUILD FAILED"; exit 1 }
16$job = Start-Job -ScriptBlock {
17 param($docx, $pdf)
18 $word = New-Object -ComObject Word.Application
19 $word.Visible = $false
20 $word.DisplayAlerts = 0
21 try {
22 $doc = $word.Documents.Open($docx, $false, $false, $false)
23 if ($doc.TablesOfContents.Count -ge 1) { $doc.TablesOfContents.Item(1).Update() }
24 $doc.Save()
25 # ExportAsFixedFormat2 with OptimizeForImageQuality keeps the diagrams at full resolution (the older call
26 # lowers every picture above about 220 ppi to 200 ppi, and the small text of large diagrams blurs on zoom)
27 try { $doc.ExportAsFixedFormat2($pdf, 17, $false, 0, 0, 1, 1, 0, $true, $true, 1, $true, $true, $false, $true); "pdf: full image quality" }
28 catch { $doc.ExportAsFixedFormat($pdf, 17, $false, 0, 0, 1, 1, 0, $true); "pdf: standard export ($_)" }
29 "pages: " + $doc.ComputeStatistics(2)
30 $doc.Close($false)
31 } catch { "WORD ERROR: $_" } finally { $word.Quit() }
32} -ArgumentList $docx, $pdf
33if (-not (Wait-Job $job -Timeout 240)) { "WORD TIMEOUT"; Stop-Job $job }
34$result = Receive-Job $job
35Remove-Job $job -Force
36$result
37python C:\a3cap\toolkit\report\pages.py $pdf (Join-Path $out "_pages.png") | Out-Null
38python C:\a3cap\toolkit\report\free.py $pdf
39if (($result -match "^pages: ") -and (Test-Path $pdf) -and ((Get-Item $pdf).LastWriteTime -gt $started)) {
40 $dest = Join-Path $Root "Bai_nop"
41 Copy-Item $docx, $pdf $dest -Force
42 "copied to Bai_nop: $name.docx, $name.pdf"
43} else { "NOT copied to Bai_nop (build incomplete)" }
