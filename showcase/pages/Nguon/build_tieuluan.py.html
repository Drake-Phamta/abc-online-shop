87def runs(para, text, bold=False, italic=False, size=None, mono=False):
88 text = KEEP_NUM.sub(lambda m: m.group(1) + NB + m.group(2), text)
89 text = text.replace(" → ", NB + "→ ")
90 # **bold** spans, then S_{x} subscripts inside each span
91 spans, pos = [], 0
92 for m in BOLD.finditer(text):
93 spans.append((text[pos:m.start()], bold))
94 spans.append((m.group(1), True))
95 pos = m.end()
96 spans.append((text[pos:], bold))
97 for span, b in spans:
98 pieces, pos = [], 0
99 for m in MARK.finditer(span):
100 pieces.append((span[pos:m.start()], italic, False))
101 pieces.append((m.group(1), True, False))
102 pieces.append((m.group(2), len(m.group(2)) == 1, True))
103 pos = m.end()
104 pieces.append((span[pos:], italic, False))
105 for chunk, ital, is_sub in pieces:
106 if not chunk:
107 continue
108 run = para.add_run(chunk)
109 run.bold, run.italic = b, ital
110 run.font.subscript = is_sub or None
111 if size:
112 run.font.size = Pt(size)
113 if mono:
114 run.font.name = "Consolas"
115 run.font.size = Pt(11)
116 run._element.rPr.rFonts.set(qn("w:eastAsia"), "Consolas")

192def figure(name, caption, width_cm=16, max_h_cm=21):
193 # "vp/png/R2.png", "docs/evidence/e2e/..": the project's own files, one copy for the essay and the project;
194 # a plain name is an older figure kept in Anh_bieu_do
195 path = os.path.join(ROOT, *name.split("/")) if "/" in name else os.path.join(FIGDIR, name)
196 with Image.open(path) as im:
197 w, h = im.size
198 if h / w * width_cm > max_h_cm:
199 width_cm = max_h_cm * w / h
200 # Word's PDF export keeps a picture up to about 270 ppi of its printed size and lowers a denser one to 200 ppi
201 # (measured: 270 kept, 290 lowered). A large diagram therefore goes in at 265 ppi, scaled here with Lanczos, which
202 # keeps a third more detail than Word's 200 ppi for the small text when the reader zooms in.
203 max_px = int(width_cm / 2.54 * 265)
204 if w > max_px:
205 import hashlib
206 cache = os.path.join(os.path.dirname(os.path.abspath(OUT)), "_img290")
207 os.makedirs(cache, exist_ok=True)
208 scaled = os.path.join(cache, hashlib.md5(f"{path}|{max_px}|{os.path.getmtime(path)}".encode()).hexdigest() + ".png")
209 if not os.path.exists(scaled):
210 with Image.open(path) as im:
211 im.convert("RGB").resize((max_px, round(h * max_px / w)), Image.LANCZOS).save(scaled, optimize=True)
212 path = scaled
213 para = doc.add_paragraph()
214 para.alignment = C
215 take_pending(para.paragraph_format, 4)
216 para.paragraph_format.space_after = Pt(2)
217 para.paragraph_format.keep_with_next = True
218 para.add_run().add_picture(path, width=Cm(width_cm))
219 lab = _label("fig")
220 FIGURES.append((lab, caption))
221 p(f"Figure {lab}. {caption}", italic=True, size=11, align=C, after=10)
222 return lab

225def table(caption, rows, widths, size=11, center_cols=(), merge_first=False, header=True):
226 lab = None
227 if caption:
228 lab = _label("tab")
229 TABLES.append((lab, caption))
230 p(f"Table {lab}. {caption}", italic=True, size=11, align=C, after=2, keep=True, before=4)
231 t = doc.add_table(rows=len(rows), cols=len(rows[0]))
232 t.style = "Table Grid"
233 t.alignment = WD_TABLE_ALIGNMENT.CENTER
234 for i, row in enumerate(rows):
235 for j, val in enumerate(row):
236 cell = t.cell(i, j)
237 cell.width = Cm(widths[j])
238 cell.text = ""
239 para = cell.paragraphs[0]
240 para.paragraph_format.space_after = Pt(0)
241 para.paragraph_format.line_spacing = 1.0
242 if j in center_cols or (header and i == 0) or (merge_first and j == 0):
243 para.alignment = C
244 runs(para, val, bold=(header and i == 0), size=size)
245 if header and i == 0:
246 shd = OxmlElement("w:shd")
247 shd.set(qn("w:val"), "clear")
248 shd.set(qn("w:fill"), "E7ECF3")
249 cell._tc.get_or_add_tcPr().append(shd)
250 if i < len(rows) - 1 and len(rows) <= 6:
251 for cell in t.rows[i].cells:
252 for para in cell.paragraphs:
253 para.paragraph_format.keep_with_next = True
254 tr_pr = t.rows[i]._tr.get_or_add_trPr()
255 tr_pr.append(tr_pr.makeelement(qn("w:cantSplit"), {}))
256 if header and i == 0:
257 tr_pr.append(tr_pr.makeelement(qn("w:tblHeader"), {}))
258 if merge_first:
259 i = 1
260 while i < len(rows):
261 j = i
262 while j + 1 < len(rows) and rows[j + 1][0] == "":
263 j += 1
264 if j > i:
265 merged = t.cell(i, 0).merge(t.cell(j, 0))
266 for extra in merged.paragraphs[1:]:
267 extra._element.getparent().remove(extra._element)
268 merged.vertical_alignment = WD_ALIGN_VERTICAL.CENTER
269 i = j + 1
270 PENDING[0] = 8
271 return lab

328def field(para, instr, size=11):
329 run = para.add_run()
330 b = run._r.makeelement(qn("w:fldChar"), {qn("w:fldCharType"): "begin"})
331 t = run._r.makeelement(qn("w:instrText"), {})
332 t.text = instr
333 t.set(qn("xml:space"), "preserve")
334 sep = run._r.makeelement(qn("w:fldChar"), {qn("w:fldCharType"): "separate"})
335 e = run._r.makeelement(qn("w:fldChar"), {qn("w:fldCharType"): "end"})
336 for el in (b, t, sep):
337 run._r.append(el)
338 holder = para.add_run(" ")
339 holder.font.size = Pt(size)
340 run2 = para.add_run()
341 run2._r.append(e)
342 run.font.size = Pt(size)
